1use crate::parquet::error::ParquetError;
2
3const MAX_ELEMENT_SIZE: usize = 8;
4
5/// Decodes using the [Byte Stream Split](https://github.com/apache/parquet-format/blob/master/Encodings.md#byte-stream-split-byte_stream_split--9) encoding.
6/// # Implementation
7/// A fixed size buffer is stored inline to support reading types of up to 8 bytes in size.
8#[derive(Debug)]
9pub struct Decoder<'a> {
10    values: &'a [u8],
11    buffer: [u8; MAX_ELEMENT_SIZE],
12    num_elements: usize,
13    position: usize,
14    element_size: usize,
15}
16
17impl<'a> Decoder<'a> {
18    pub fn try_new(values: &'a [u8], element_size: usize) -> Result<Self, ParquetError> {
19        if element_size > MAX_ELEMENT_SIZE {
20            // Since Parquet format version 2.11 it's valid to use byte stream split for fixed-length byte array data,
21            // which could be larger than 8 bytes, but Polars doesn't yet support reading byte stream split encoded FLBA data.
22            return Err(ParquetError::oos(format!(
23                "Byte stream split decoding only supports up to {} byte element sizes",
24                MAX_ELEMENT_SIZE
25            )));
26        }
27
28        let values_size = values.len();
29        if values_size % element_size != 0 {
30            return Err(ParquetError::oos(format!(
31                "Values array length ({}) is not a multiple of the element size ({})",
32                values_size, element_size
33            )));
34        }
35        let num_elements = values.len() / element_size;
36
37        Ok(Self {
38            values,
39            buffer: [0; MAX_ELEMENT_SIZE],
40            num_elements,
41            position: 0,
42            element_size,
43        })
44    }
45
46    pub fn move_next(&mut self) -> bool {
47        if self.position >= self.num_elements {
48            return false;
49        }
50
51        debug_assert!(self.element_size <= MAX_ELEMENT_SIZE);
52        debug_assert!(self.values.len() >= self.num_elements * self.element_size);
53        for n in 0..self.element_size {
54            unsafe {
55                // SAFETY:
56                // We have the invariants that element_size <= MAX_ELEMENT_SIZE,
57                // buffer.len() == MAX_ELEMENT_SIZE,
58                // position < num_elements and
59                // values.len() >= num_elements * element_size.
60                *self.buffer.get_unchecked_mut(n) = *self
61                    .values
62                    .get_unchecked((self.num_elements * n) + self.position)
63            }
64        }
65
66        self.position += 1;
67        true
68    }
69
70    /// The number of remaining values
71    pub fn len(&self) -> usize {
72        self.num_elements - self.position
73    }
74
75    pub fn current_value(&self) -> &[u8] {
76        &self.buffer[0..self.element_size]
77    }
78
79    pub fn iter_converted<'b, T, F>(&'b mut self, converter: F) -> DecoderIterator<'a, 'b, T, F>
80    where
81        F: Copy + Fn(&[u8]) -> T,
82    {
83        DecoderIterator {
84            decoder: self,
85            converter,
86        }
87    }
88}
89
90#[derive(Debug)]
91pub struct DecoderIterator<'a, 'b, T, F>
92where
93    F: Copy + Fn(&[u8]) -> T,
94{
95    decoder: &'b mut Decoder<'a>,
96    converter: F,
97}
98
99impl<T, F> Iterator for DecoderIterator<'_, '_, T, F>
100where
101    F: Copy + Fn(&[u8]) -> T,
102{
103    type Item = T;
104
105    #[inline]
106    fn next(&mut self) -> Option<Self::Item> {
107        if self.decoder.move_next() {
108            Some((self.converter)(self.decoder.current_value()))
109        } else {
110            None
111        }
112    }
113
114    #[inline]
115    fn size_hint(&self) -> (usize, Option<usize>) {
116        (self.decoder.len(), Some(self.decoder.len()))
117    }
118}

13pub fn columns_to_iter_recursive(
14    mut columns: Vec<BasicDecompressor>,
15    mut types: Vec<&PrimitiveType>,
16    field: Field,
17    mut init: Vec<InitNested>,
18    filter: Option<Filter>,
19) -> PolarsResult<(NestedState, Box<dyn Array>)> {
20    use arrow::datatypes::PhysicalType::*;
21    use arrow::datatypes::PrimitiveType::*;
22
23    Ok(match field.dtype().to_physical_type() {
24        Null => {
25            // physical type is i32
26            init.push(InitNested::Primitive(field.is_nullable));
27            types.pop();
28            PageNestedDecoder::new(
29                columns.pop().unwrap(),
30                field.dtype().clone(),
31                null::NullDecoder,
32                init,
33            )?
34            .collect_n(filter)
35            .map(|(s, a)| (s, Box::new(a) as Box<_>))?
36        },
37        Boolean => {
38            init.push(InitNested::Primitive(field.is_nullable));
39            types.pop();
40            PageNestedDecoder::new(
41                columns.pop().unwrap(),
42                ArrowDataType::Boolean,
43                boolean::BooleanDecoder,
44                init,
45            )?
46            .collect_n(filter)
47            .map(|(s, a)| (s, Box::new(a) as Box<_>))?
48        },
49        Primitive(Int8) => {
50            init.push(InitNested::Primitive(field.is_nullable));
51            types.pop();
52            PageNestedDecoder::new(
53                columns.pop().unwrap(),
54                field.dtype().clone(),
55                primitive::IntDecoder::<i32, i8, _>::cast_as(),
56                init,
57            )?
58            .collect_n(filter)
59            .map(|(s, a)| (s, Box::new(a) as Box<_>))?
60        },
61        Primitive(Int16) => {
62            init.push(InitNested::Primitive(field.is_nullable));
63            types.pop();
64            PageNestedDecoder::new(
65                columns.pop().unwrap(),
66                field.dtype().clone(),
67                primitive::IntDecoder::<i32, i16, _>::cast_as(),
68                init,
69            )?
70            .collect_n(filter)
71            .map(|(s, a)| (s, Box::new(a) as Box<_>))?
72        },
73        Primitive(Int32) => {
74            init.push(InitNested::Primitive(field.is_nullable));
75            types.pop();
76            PageNestedDecoder::new(
77                columns.pop().unwrap(),
78                field.dtype().clone(),
79                primitive::IntDecoder::<i32, _, _>::unit(),
80                init,
81            )?
82            .collect_n(filter)
83            .map(|(s, a)| (s, Box::new(a) as Box<_>))?
84        },
85        Primitive(Int64) => {
86            init.push(InitNested::Primitive(field.is_nullable));
87            types.pop();
88            PageNestedDecoder::new(
89                columns.pop().unwrap(),
90                field.dtype().clone(),
91                primitive::IntDecoder::<i64, _, _>::unit(),
92                init,
93            )?
94            .collect_n(filter)
95            .map(|(s, a)| (s, Box::new(a) as Box<_>))?
96        },
97        Primitive(UInt8) => {
98            init.push(InitNested::Primitive(field.is_nullable));
99            types.pop();
100            PageNestedDecoder::new(
101                columns.pop().unwrap(),
102                field.dtype().clone(),
103                primitive::IntDecoder::<i32, u8, _>::cast_as(),
104                init,
105            )?
106            .collect_n(filter)
107            .map(|(s, a)| (s, Box::new(a) as Box<_>))?
108        },
109        Primitive(UInt16) => {
110            init.push(InitNested::Primitive(field.is_nullable));
111            types.pop();
112            PageNestedDecoder::new(
113                columns.pop().unwrap(),
114                field.dtype().clone(),
115                primitive::IntDecoder::<i32, u16, _>::cast_as(),
116                init,
117            )?
118            .collect_n(filter)
119            .map(|(s, a)| (s, Box::new(a) as Box<_>))?
120        },
121        Primitive(UInt32) => {
122            init.push(InitNested::Primitive(field.is_nullable));
123            let type_ = types.pop().unwrap();
124            match type_.physical_type {
125                PhysicalType::Int32 => PageNestedDecoder::new(
126                    columns.pop().unwrap(),
127                    field.dtype().clone(),
128                    primitive::IntDecoder::<i32, u32, _>::cast_as(),
129                    init,
130                )?
131                .collect_n(filter)
132                .map(|(s, a)| (s, Box::new(a) as Box<_>))?,
133                // some implementations of parquet write arrow's u32 into i64.
134                PhysicalType::Int64 => PageNestedDecoder::new(
135                    columns.pop().unwrap(),
136                    field.dtype().clone(),
137                    primitive::IntDecoder::<i64, u32, _>::cast_as(),
138                    init,
139                )?
140                .collect_n(filter)
141                .map(|(s, a)| (s, Box::new(a) as Box<_>))?,
142                other => {
143                    polars_bail!(ComputeError:
144                        "deserializing UInt32 from {other:?}'s parquet"
145                    )
146                },
147            }
148        },
149        Primitive(UInt64) => {
150            init.push(InitNested::Primitive(field.is_nullable));
151            types.pop();
152            PageNestedDecoder::new(
153                columns.pop().unwrap(),
154                field.dtype().clone(),
155                primitive::IntDecoder::<i64, u64, _>::cast_as(),
156                init,
157            )?
158            .collect_n(filter)
159            .map(|(s, a)| (s, Box::new(a) as Box<_>))?
160        },
161        Primitive(Float32) => {
162            init.push(InitNested::Primitive(field.is_nullable));
163            types.pop();
164            PageNestedDecoder::new(
165                columns.pop().unwrap(),
166                field.dtype().clone(),
167                primitive::FloatDecoder::<f32, _, _>::unit(),
168                init,
169            )?
170            .collect_n(filter)
171            .map(|(s, a)| (s, Box::new(a) as Box<_>))?
172        },
173        Primitive(Float64) => {
174            init.push(InitNested::Primitive(field.is_nullable));
175            types.pop();
176            PageNestedDecoder::new(
177                columns.pop().unwrap(),
178                field.dtype().clone(),
179                primitive::FloatDecoder::<f64, _, _>::unit(),
180                init,
181            )?
182            .collect_n(filter)
183            .map(|(s, a)| (s, Box::new(a) as Box<_>))?
184        },
185        BinaryView | Utf8View => {
186            init.push(InitNested::Primitive(field.is_nullable));
187            types.pop();
188            PageNestedDecoder::new(
189                columns.pop().unwrap(),
190                field.dtype().clone(),
191                binview::BinViewDecoder::default(),
192                init,
193            )?
194            .collect_n(filter)?
195        },
196        // These are all converted to View variants before.
197        LargeBinary | LargeUtf8 | Binary | Utf8 => unreachable!(),
198        _ => match field.dtype().to_logical_type() {
199            ArrowDataType::Dictionary(key_type, value_type, _) => {
200                // @note: this should only hit in two cases:
201                // - polars enum's and categorical's
202                // - int -> string which can be turned into categoricals
203                assert!(matches!(value_type.as_ref(), ArrowDataType::Utf8View));
204
205                init.push(InitNested::Primitive(field.is_nullable));
206
207                if field.metadata.as_ref().is_none_or(|md| {
208                    !md.contains_key(DTYPE_ENUM_VALUES) && !md.contains_key(DTYPE_CATEGORICAL)
209                }) {
210                    let (nested, arr) = PageNestedDecoder::new(
211                        columns.pop().unwrap(),
212                        ArrowDataType::Utf8View,
213                        binview::BinViewDecoder::default(),
214                        init,
215                    )?
216                    .collect_n(filter)?;
217
218                    let arr = polars_compute::cast::cast(
219                        arr.as_ref(),
220                        field.dtype(),
221                        CastOptionsImpl::default(),
222                    )
223                    .unwrap();
224
225                    (nested, arr)
226                } else {
227                    assert!(matches!(key_type, IntegerType::UInt32));
228
229                    PageNestedDecoder::new(
230                        columns.pop().unwrap(),
231                        field.dtype().clone(),
232                        CategoricalDecoder::new(),
233                        init,
234                    )?
235                    .collect_n(filter)
236                    .map(|(nested, arr)| (nested, arr.to_boxed()))?
237                }
238            },
239            ArrowDataType::List(inner) | ArrowDataType::LargeList(inner) => {
240                init.push(InitNested::List(field.is_nullable));
241                let (mut nested, array) = columns_to_iter_recursive(
242                    columns,
243                    types,
244                    inner.as_ref().clone(),
245                    init,
246                    filter,
247                )?;
248                let array = create_list(field.dtype().clone(), &mut nested, array);
249                (nested, array)
250            },
251            ArrowDataType::FixedSizeList(inner, width) => {
252                init.push(InitNested::FixedSizeList(field.is_nullable, *width));
253                let (mut nested, array) = columns_to_iter_recursive(
254                    columns,
255                    types,
256                    inner.as_ref().clone(),
257                    init,
258                    filter,
259                )?;
260                let array = create_list(field.dtype().clone(), &mut nested, array);
261                (nested, array)
262            },
263            ArrowDataType::Decimal(_, _) => {
264                init.push(InitNested::Primitive(field.is_nullable));
265                let type_ = types.pop().unwrap();
266                match type_.physical_type {
267                    PhysicalType::Int32 => PageNestedDecoder::new(
268                        columns.pop().unwrap(),
269                        field.dtype.clone(),
270                        primitive::IntDecoder::<i32, i128, _>::cast_into(),
271                        init,
272                    )?
273                    .collect_n(filter)
274                    .map(|(s, a)| (s, Box::new(a) as Box<_>))?,
275                    PhysicalType::Int64 => PageNestedDecoder::new(
276                        columns.pop().unwrap(),
277                        field.dtype.clone(),
278                        primitive::IntDecoder::<i64, i128, _>::cast_into(),
279                        init,
280                    )?
281                    .collect_n(filter)
282                    .map(|(s, a)| (s, Box::new(a) as Box<_>))?,
283                    PhysicalType::FixedLenByteArray(n) if n > 16 => {
284                        polars_bail!(
285                            ComputeError: "Can't decode Decimal128 type from `FixedLenByteArray` of len {n}"
286                        )
287                    },
288                    PhysicalType::FixedLenByteArray(size) => {
289                        let (nested, array) = PageNestedDecoder::new(
290                            columns.pop().unwrap(),
291                            ArrowDataType::FixedSizeBinary(size),
292                            fixed_size_binary::BinaryDecoder { size },
293                            init,
294                        )?
295                        .collect_n(filter)?;
296
297                        // Convert the fixed length byte array to Decimal.
298                        let values = array
299                            .values()
300                            .chunks_exact(size)
301                            .map(|value: &[u8]| super::super::convert_i128(value, size))
302                            .collect::<Vec<_>>();
303                        let validity = array.validity().cloned();
304
305                        let array: Box<dyn Array> = Box::new(PrimitiveArray::<i128>::try_new(
306                            field.dtype.clone(),
307                            values.into(),
308                            validity,
309                        )?);
310
311                        (nested, array)
312                    },
313                    _ => {
314                        polars_bail!(ComputeError:
315                            "Deserializing type for Decimal {:?} from parquet",
316                            type_.physical_type
317                        )
318                    },
319                }
320            },
321            ArrowDataType::Decimal256(_, _) => {
322                init.push(InitNested::Primitive(field.is_nullable));
323                let type_ = types.pop().unwrap();
324                match type_.physical_type {
325                    PhysicalType::Int32 => PageNestedDecoder::new(
326                        columns.pop().unwrap(),
327                        field.dtype.clone(),
328                        primitive::IntDecoder::closure(|x: i32| i256(I256::new(x as i128))),
329                        init,
330                    )?
331                    .collect_n(filter)
332                    .map(|(s, a)| (s, Box::new(a) as Box<_>))?,
333                    PhysicalType::Int64 => PageNestedDecoder::new(
334                        columns.pop().unwrap(),
335                        field.dtype.clone(),
336                        primitive::IntDecoder::closure(|x: i64| i256(I256::new(x as i128))),
337                        init,
338                    )?
339                    .collect_n(filter)
340                    .map(|(s, a)| (s, Box::new(a) as Box<_>))?,
341                    PhysicalType::FixedLenByteArray(size) if size <= 16 => {
342                        let (nested, array) = PageNestedDecoder::new(
343                            columns.pop().unwrap(),
344                            ArrowDataType::FixedSizeBinary(size),
345                            fixed_size_binary::BinaryDecoder { size },
346                            init,
347                        )?
348                        .collect_n(filter)?;
349
350                        // Convert the fixed length byte array to Decimal.
351                        let values = array
352                            .values()
353                            .chunks_exact(size)
354                            .map(|value| i256(I256::new(super::super::convert_i128(value, size))))
355                            .collect::<Vec<_>>();
356                        let validity = array.validity().cloned();
357
358                        let array: Box<dyn Array> = Box::new(PrimitiveArray::<i256>::try_new(
359                            field.dtype.clone(),
360                            values.into(),
361                            validity,
362                        )?);
363
364                        (nested, array)
365                    },
366
367                    PhysicalType::FixedLenByteArray(size) if size <= 32 => {
368                        let (nested, array) = PageNestedDecoder::new(
369                            columns.pop().unwrap(),
370                            ArrowDataType::FixedSizeBinary(size),
371                            fixed_size_binary::BinaryDecoder { size },
372                            init,
373                        )?
374                        .collect_n(filter)?;
375
376                        // Convert the fixed length byte array to Decimal.
377                        let values = array
378                            .values()
379                            .chunks_exact(size)
380                            .map(super::super::convert_i256)
381                            .collect::<Vec<_>>();
382                        let validity = array.validity().cloned();
383
384                        let array: Box<dyn Array> = Box::new(PrimitiveArray::<i256>::try_new(
385                            field.dtype.clone(),
386                            values.into(),
387                            validity,
388                        )?);
389
390                        (nested, array)
391                    },
392                    PhysicalType::FixedLenByteArray(n) => {
393                        polars_bail!(ComputeError:
394                            "Can't decode Decimal256 type from `FixedLenByteArray` of len {n}"
395                        )
396                    },
397                    _ => {
398                        polars_bail!(ComputeError:
399                            "Deserializing type for Decimal {:?} from parquet",
400                            type_.physical_type
401                        )
402                    },
403                }
404            },
405            ArrowDataType::Struct(fields) => {
406                // @NOTE:
407                // We go back to front here, because we constantly split off the end of the array
408                // to grab the relevant columns and types.
409                //
410                // Is this inefficient? Yes. Is this how we are going to do it for now? Yes.
411
412                let Some(last_field) = fields.last() else {
413                    return Err(ParquetError::not_supported("Struct has zero fields").into());
414                };
415
416                let field_to_nested_array =
417                    |mut init: Vec<InitNested>,
418                     columns: &mut Vec<BasicDecompressor>,
419                     types: &mut Vec<&PrimitiveType>,
420                     struct_field: &Field| {
421                        init.push(InitNested::Struct(field.is_nullable));
422                        let n = n_columns(&struct_field.dtype);
423                        let columns = columns.split_off(columns.len() - n);
424                        let types = types.split_off(types.len() - n);
425
426                        columns_to_iter_recursive(
427                            columns,
428                            types,
429                            struct_field.clone(),
430                            init,
431                            filter.clone(),
432                        )
433                    };
434
435                let (mut nested, last_array) =
436                    field_to_nested_array(init.clone(), &mut columns, &mut types, last_field)?;
437                debug_assert!(matches!(nested.last().unwrap(), NestedContent::Struct));
438                let (length, _, struct_validity) = nested.pop().unwrap();
439
440                let mut field_arrays = Vec::<Box<dyn Array>>::with_capacity(fields.len());
441                field_arrays.push(last_array);
442
443                for field in fields.iter().rev().skip(1) {
444                    let (mut _nested, array) =
445                        field_to_nested_array(init.clone(), &mut columns, &mut types, field)?;
446
447                    #[cfg(debug_assertions)]
448                    {
449                        debug_assert!(matches!(_nested.last().unwrap(), NestedContent::Struct));
450                        debug_assert_eq!(
451                            _nested.pop().unwrap().2.and_then(freeze_validity),
452                            struct_validity.clone().and_then(freeze_validity),
453                        );
454                    }
455
456                    field_arrays.push(array);
457                }
458
459                field_arrays.reverse();
460                let struct_validity = struct_validity.and_then(freeze_validity);
461
462                (
463                    nested,
464                    Box::new(StructArray::new(
465                        ArrowDataType::Struct(fields.clone()),
466                        length,
467                        field_arrays,
468                        struct_validity,
469                    )),
470                )
471            },
472            ArrowDataType::Map(inner, _) => {
473                init.push(InitNested::List(field.is_nullable));
474                let (mut nested, array) = columns_to_iter_recursive(
475                    columns,
476                    types,
477                    inner.as_ref().clone(),
478                    init,
479                    filter,
480                )?;
481                let array = create_map(field.dtype().clone(), &mut nested, array);
482                (nested, array)
483            },
484            other => {
485                polars_bail!(ComputeError:
486                    "Deserializing type {other:?} from parquet"
487                )
488            },
489        },
490    })

20pub(crate) enum StateTranslation<'a> {
21    Plain(BitMask<'a>),
22    Rle(HybridRleDecoder<'a>),
23}

25impl<'a> utils::StateTranslation<'a, BooleanDecoder> for StateTranslation<'a> {

28    fn new(
29        _decoder: &BooleanDecoder,
30        page: &'a DataPage,
31        _dict: Option<&'a <BooleanDecoder as Decoder>::Dict>,
32        page_validity: Option<&Bitmap>,
33    ) -> ParquetResult<Self> {
34        let values = split_buffer(page)?.values;
35
36        match page.encoding() {
37            Encoding::Plain => {
38                let max_num_values = values.len() * u8::BITS as usize;
39                let num_values = if page_validity.is_some() {
40                    // @NOTE: We overestimate the amount of values here, but in the V1
41                    // specification we don't really have a way to know the number of valid items.
42                    // Without traversing the list.
43                    max_num_values
44                } else {
45                    // @NOTE: We cannot really trust the value from this as it might relate to the
46                    // number of top-level nested values. Therefore, we do a `min` with the maximum
47                    // number of possible values.
48                    usize::min(page.num_values(), max_num_values)
49                };
50
51                Ok(Self::Plain(BitMask::new(values, 0, num_values)))
52            },
53            Encoding::Rle => {
54                // @NOTE: For a nullable list, we might very well overestimate the amount of
55                // values, but we never collect those items. We don't really have a way to know the
56                // number of valid items in the V1 specification.
57
58                // For RLE boolean values the length in bytes is pre-pended.
59                // https://github.com/apache/parquet-format/blob/e517ac4dbe08d518eb5c2e58576d4c711973db94/Encodings.md#run-length-encoding--bit-packing-hybrid-rle--3
60                let (_len_in_bytes, values) = values.split_at(4);
61                Ok(Self::Rle(HybridRleDecoder::new(
62                    values,
63                    1,
64                    page.num_values(),
65                )))
66            },
67            _ => Err(utils::not_implemented(page)),
68        }
69    }
70}

81fn decode_optional_rle(
82    values: HybridRleDecoder<'_>,
83    target: &mut MutableBitmap,
84    page_validity: &Bitmap,
85) -> ParquetResult<()> {
86    debug_assert!(page_validity.set_bits() <= values.len());
87
88    if page_validity.unset_bits() == 0 {
89        return decode_required_rle(values, Some(page_validity.len()), target);
90    }
91
92    target.reserve(page_validity.len());
93
94    let mut validity_mask = BitMask::from_bitmap(page_validity);
95
96    for chunk in values.into_chunk_iter() {
97        let chunk = chunk?;
98
99        match chunk {
100            HybridRleChunk::Rle(value, size) => {
101                let offset = validity_mask
102                    .nth_set_bit_idx(size, 0)
103                    .unwrap_or(validity_mask.len());
104
105                let t;
106                (t, validity_mask) = validity_mask.split_at(offset);
107
108                target.extend_constant(t.len(), value != 0);
109            },
110            HybridRleChunk::Bitpacked(decoder) => {
111                let decoder_slice = decoder.as_slice();
112                let offset = validity_mask
113                    .nth_set_bit_idx(decoder.len(), 0)
114                    .unwrap_or(validity_mask.len());
115
116                let decoder_validity;
117                (decoder_validity, validity_mask) = validity_mask.split_at(offset);
118
119                let mut offset = 0;
120                let mut validity_iter = decoder_validity.iter();
121                while validity_iter.num_remaining() > 0 {
122                    let num_valid = validity_iter.take_leading_ones();
123                    target.extend_from_slice(decoder_slice, offset, num_valid);
124                    offset += num_valid;
125
126                    let num_invalid = validity_iter.take_leading_zeros();
127                    target.extend_constant(num_invalid, false);
128                }
129            },
130        }
131    }
132
133    if cfg!(debug_assertions) {
134        assert_eq!(validity_mask.set_bits(), 0);
135    }
136    target.extend_constant(validity_mask.len(), false);
137
138    Ok(())
139}

141fn decode_masked_required_rle(
142    values: HybridRleDecoder<'_>,
143    target: &mut MutableBitmap,
144    mask: &Bitmap,
145) -> ParquetResult<()> {
146    debug_assert!(mask.len() <= values.len());
147
148    if mask.unset_bits() == 0 {
149        return decode_required_rle(values, Some(mask.len()), target);
150    }
151
152    let mut im_target = MutableBitmap::new();
153    decode_required_rle(values, Some(mask.len()), &mut im_target)?;
154
155    target.extend_from_bitmap(&filter_boolean_kernel(&im_target.freeze(), mask));
156
157    Ok(())
158}

160fn decode_masked_optional_rle(
161    values: HybridRleDecoder<'_>,
162    target: &mut MutableBitmap,
163    page_validity: &Bitmap,
164    mask: &Bitmap,
165) -> ParquetResult<()> {
166    debug_assert_eq!(page_validity.len(), mask.len());
167    debug_assert!(mask.len() <= values.len());
168
169    if mask.unset_bits() == 0 {
170        return decode_optional_rle(values, target, page_validity);
171    }
172
173    if page_validity.unset_bits() == 0 {
174        return decode_masked_required_rle(values, target, mask);
175    }
176
177    let mut im_target = MutableBitmap::new();
178    decode_optional_rle(values, &mut im_target, page_validity)?;
179
180    target.extend_from_bitmap(&filter_boolean_kernel(&im_target.freeze(), mask));
181
182    Ok(())
183}

185fn decode_required_plain(values: BitMask<'_>, target: &mut MutableBitmap) -> ParquetResult<()> {
186    target.extend_from_bitmask(values);
187    Ok(())

190fn decode_optional_plain(
191    mut values: BitMask<'_>,
192    target: &mut MutableBitmap,
193    mut page_validity: Bitmap,
194) -> ParquetResult<()> {
195    debug_assert!(page_validity.set_bits() <= values.len());
196
197    if page_validity.unset_bits() == 0 {
198        return decode_required_plain(values.sliced(0, page_validity.len()), target);
199    }
200
201    target.reserve(page_validity.len());
202
203    while !page_validity.is_empty() {
204        let num_valid = page_validity.take_leading_ones();
205        let iv;
206        (iv, values) = values.split_at(num_valid);
207        target.extend_from_bitmask(iv);
208
209        let num_invalid = page_validity.take_leading_zeros();
210        target.extend_constant(num_invalid, false);
211    }
212
213    Ok(())
214}

216fn decode_masked_required_plain(
217    mut values: BitMask,
218    target: &mut MutableBitmap,
219    mut mask: Bitmap,
220) -> ParquetResult<()> {
221    debug_assert!(mask.len() <= values.len());
222
223    let leading_zeros = mask.take_leading_zeros();
224    mask.take_trailing_zeros();
225
226    values = values.sliced(leading_zeros, mask.len());
227
228    if mask.unset_bits() == 0 {
229        return decode_required_plain(values, target);
230    }
231
232    let mut im_target = MutableBitmap::new();
233    decode_required_plain(values, &mut im_target)?;
234
235    target.extend_from_bitmap(&filter_boolean_kernel(&im_target.freeze(), &mask));
236
237    Ok(())
238}

240fn decode_masked_optional_plain(
241    mut values: BitMask<'_>,
242    target: &mut MutableBitmap,
243    mut page_validity: Bitmap,
244    mut mask: Bitmap,
245) -> ParquetResult<()> {
246    debug_assert_eq!(page_validity.len(), mask.len());
247    debug_assert!(page_validity.set_bits() <= values.len());
248
249    let leading_zeros = mask.take_leading_zeros();
250    mask.take_trailing_zeros();
251
252    let (skipped, truncated);
253    (skipped, page_validity) = page_validity.split_at(leading_zeros);
254    (page_validity, truncated) = page_validity.split_at(mask.len());
255
256    let skipped_values = skipped.set_bits();
257    let truncated_values = truncated.set_bits();
258    values = values.sliced(
259        skipped_values,
260        values.len() - skipped_values - truncated_values,
261    );
262
263    if mask.unset_bits() == 0 {
264        return decode_optional_plain(values, target, page_validity);
265    }
266
267    if page_validity.unset_bits() == 0 {
268        return decode_masked_required_plain(values, target, mask);
269    }
270
271    let mut im_target = MutableBitmap::new();
272    decode_optional_plain(values, &mut im_target, page_validity)?;
273
274    target.extend_from_bitmap(&filter_boolean_kernel(&im_target.freeze(), &mask));
275
276    Ok(())
277}

293impl Decoder for BooleanDecoder {
294    type Translation<'a> = StateTranslation<'a>;
295    type Dict = ();
296    type DecodedState = (MutableBitmap, MutableBitmap);
297    type Output = BooleanArray;
298
299    fn with_capacity(&self, capacity: usize) -> Self::DecodedState {
300        (
301            MutableBitmap::with_capacity(capacity),
302            MutableBitmap::with_capacity(capacity),
303        )
304    }
305
306    fn deserialize_dict(&mut self, _: DictPage) -> ParquetResult<Self::Dict> {
307        Ok(())
308    }
309
310    fn finalize(
311        &self,
312        dtype: ArrowDataType,
313        _dict: Option<Self::Dict>,
314        (values, validity): Self::DecodedState,
315    ) -> ParquetResult<Self::Output> {
316        let validity = freeze_validity(validity);
317        Ok(BooleanArray::new(dtype, values.freeze(), validity))
318    }
319
320    fn extend_filtered_with_state(
321        &mut self,
322        state: utils::State<'_, Self>,
323        (target, validity): &mut Self::DecodedState,
324        filter: Option<super::Filter>,
325    ) -> ParquetResult<()> {
326        match state.translation {
327            StateTranslation::Plain(values) => {
328                if state.is_optional {
329                    append_validity(
330                        state.page_validity.as_ref(),
331                        filter.as_ref(),
332                        validity,
333                        values.len(),
334                    );
335                }
336
337                let page_validity = constrain_page_validity(
338                    values.len(),
339                    state.page_validity.as_ref(),
340                    filter.as_ref(),
341                );
342
343                match (filter, page_validity) {
344                    (None, None) => decode_required_plain(values, target),
345                    (Some(Filter::Range(rng)), None) => {
346                        decode_required_plain(values.sliced(rng.start, rng.len()), target)
347                    },
348                    (None, Some(page_validity)) => {
349                        decode_optional_plain(values, target, page_validity)
350                    },
351                    (Some(Filter::Range(rng)), Some(mut page_validity)) => {
352                        let (skipped, truncated);
353                        (skipped, page_validity) = page_validity.split_at(rng.start);
354                        (page_validity, truncated) = page_validity.split_at(rng.len());
355
356                        let skipped_values = skipped.set_bits();
357                        let truncated_values = truncated.set_bits();
358                        let values = values.sliced(
359                            skipped_values,
360                            values.len() - skipped_values - truncated_values,
361                        );
362
363                        decode_optional_plain(values, target, page_validity)
364                    },
365                    (Some(Filter::Mask(mask)), None) => {
366                        decode_masked_required_plain(values, target, mask)
367                    },
368                    (Some(Filter::Mask(mask)), Some(page_validity)) => {
369                        decode_masked_optional_plain(values, target, page_validity, mask)
370                    },
371                }?;
372
373                Ok(())
374            },
375            StateTranslation::Rle(values) => {
376                if state.is_optional {
377                    append_validity(
378                        state.page_validity.as_ref(),
379                        filter.as_ref(),
380                        validity,
381                        values.len(),
382                    );
383                }
384
385                let page_validity = constrain_page_validity(
386                    values.len(),
387                    state.page_validity.as_ref(),
388                    filter.as_ref(),
389                );
390
391                match (filter, page_validity) {
392                    (None, None) => decode_required_rle(values, None, target),
393                    (Some(Filter::Range(rng)), None) if rng.start == 0 => {
394                        decode_required_rle(values, Some(rng.end), target)
395                    },
396                    (None, Some(page_validity)) => {
397                        decode_optional_rle(values, target, &page_validity)
398                    },
399                    (Some(Filter::Range(rng)), Some(page_validity)) if rng.start == 0 => {
400                        decode_optional_rle(values, target, &page_validity)
401                    },
402                    (Some(Filter::Mask(filter)), None) => {
403                        decode_masked_required_rle(values, target, &filter)
404                    },
405                    (Some(Filter::Mask(filter)), Some(page_validity)) => {
406                        decode_masked_optional_rle(values, target, &page_validity, &filter)
407                    },
408                    (Some(Filter::Range(rng)), None) => {
409                        decode_masked_required_rle(values, target, &filter_from_range(rng.clone()))
410                    },
411                    (Some(Filter::Range(rng)), Some(page_validity)) => decode_masked_optional_rle(
412                        values,
413                        target,
414                        &page_validity,
415                        &filter_from_range(rng.clone()),
416                    ),
417                }?;
418
419                Ok(())
420            },
421        }
422    }

1use crate::error::PyPolarsErr;
2use polars::prelude::*;
3use polars_arrow::ffi;
4use pyo3::ffi::Py_uintptr_t;
5use pyo3::prelude::*;
6
7pub fn array_to_rust(obj: &Bound<PyAny>) -> PyResult<ArrayRef> {
8    // prepare a pointer to receive the Array struct
9    let array = Box::new(ffi::ArrowArray::empty());
10    let schema = Box::new(ffi::ArrowSchema::empty());
11
12    let array_ptr = &*array as *const ffi::ArrowArray;
13    let schema_ptr = &*schema as *const ffi::ArrowSchema;
14
15    // make the conversion through PyArrow's private API
16    // this changes the pointer's memory and is thus unsafe. In particular, `_export_to_c` can go out of bounds
17    obj.call_method1(
18        "_export_to_c",
19        (array_ptr as Py_uintptr_t, schema_ptr as Py_uintptr_t),
20    )?;
21
22    unsafe {
23        let field = ffi::import_field_from_c(schema.as_ref()).map_err(PyPolarsErr::from)?;
24        let array = ffi::import_array_from_c(*array, field.dtype).map_err(PyPolarsErr::from)?;
25        Ok(array)
26    }
27}

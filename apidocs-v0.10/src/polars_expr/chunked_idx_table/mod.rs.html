11pub trait ChunkedIdxTable: Any + Send + Sync {
12    /// Creates a new empty ChunkedIdxTable similar to this one.
13    fn new_empty(&self) -> Box<dyn ChunkedIdxTable>;
14
15    /// Reserves space for the given number additional keys.
16    fn reserve(&mut self, additional: usize);
17
18    /// Returns the number of unique keys in this ChunkedIdxTable.
19    fn num_keys(&self) -> IdxSize;
20
21    /// Inserts the given key chunk into this ChunkedIdxTable.
22    fn insert_key_chunk(&mut self, keys: HashKeys, track_unmatchable: bool);
23
24    /// Probe the table, updating table_match and probe_match with
25    /// (ChunkId, IdxSize) pairs for each match. Will stop processing new keys
26    /// once limit matches have been generated, returning the number of keys
27    /// processed.
28    ///
29    /// If mark_matches is true, matches are marked in the table as such.
30    ///
31    /// If emit_unmatched is true, for keys that do not have a match we emit a
32    /// match with ChunkId::null() on the table match.
33    fn probe(
34        &self,
35        hash_keys: &HashKeys,
36        table_match: &mut Vec<ChunkId<32>>,
37        probe_match: &mut Vec<IdxSize>,
38        mark_matches: bool,
39        emit_unmatched: bool,
40        limit: IdxSize,
41    ) -> IdxSize;
42
43    /// The same as probe, except it will only apply to the specified subset of keys.
44    /// # Safety
45    /// The provided subset indices must be in-bounds.
46    #[allow(clippy::too_many_arguments)]
47    unsafe fn probe_subset(
48        &self,
49        hash_keys: &HashKeys,
50        subset: &[IdxSize],
51        table_match: &mut Vec<ChunkId<32>>,
52        probe_match: &mut Vec<IdxSize>,
53        mark_matches: bool,
54        emit_unmatched: bool,
55        limit: IdxSize,
56    ) -> IdxSize;
57
58    /// Get the ChunkIds for each key which was never marked during probing.
59    fn unmarked_keys(&self, out: &mut Vec<ChunkId<32>>, offset: IdxSize, limit: IdxSize)
60        -> IdxSize;
61}
62

22pub fn new_sum_reduction(dtype: DataType) -> Box<dyn GroupedReduction> {
23    use DataType::*;
24    match dtype {
25        Boolean => Box::new(SumReduce::<IdxType>::new(dtype)),
26        Int8 | UInt8 | Int16 | UInt16 => Box::new(SumReduce::<Int64Type>::new(dtype)),
27        UInt32 => Box::new(SumReduce::<UInt32Type>::new(dtype)),
28        UInt64 => Box::new(SumReduce::<UInt64Type>::new(dtype)),
29        Int32 => Box::new(SumReduce::<Int32Type>::new(dtype)),
30        Int64 => Box::new(SumReduce::<Int64Type>::new(dtype)),
31        Float32 => Box::new(SumReduce::<Float32Type>::new(dtype)),
32        Float64 => Box::new(SumReduce::<Float64Type>::new(dtype)),
33        #[cfg(feature = "dtype-decimal")]
34        Decimal(_, _) => Box::new(SumReduce::<Int128Type>::new(dtype)),
35        Duration(_) => Box::new(SumReduce::<Int64Type>::new(dtype)),
36        // For compatibility with the current engine, should probably be an error.
37        String | Binary => Box::new(super::NullGroupedReduction::new(dtype)),
38        _ => unimplemented!("{dtype:?} is not supported by sum reduction"),
39    }
40}

42fn cast_sum_input<'a>(s: &'a Series, dt: &DataType) -> PolarsResult<Cow<'a, Series>> {
43    use DataType::*;
44    match dt {
45        Boolean => Ok(Cow::Owned(s.cast(&IDX_DTYPE)?)),
46        Int8 | UInt8 | Int16 | UInt16 => Ok(Cow::Owned(s.cast(&Int64)?)),
47        #[cfg(feature = "dtype-decimal")]
48        Decimal(_, _) => Ok(Cow::Owned(
49            s.decimal().unwrap().physical().clone().into_series(),
50        )),
51        #[cfg(feature = "dtype-duration")]
52        Duration(_) => Ok(Cow::Owned(
53            s.duration().unwrap().physical().clone().into_series(),
54        )),
55        _ => Ok(Cow::Borrowed(s)),
56    }
57}

59fn out_dtype(in_dtype: &DataType) -> DataType {
60    use DataType::*;
61    match in_dtype {
62        Boolean => IDX_DTYPE,
63        Int8 | UInt8 | Int16 | UInt16 => Int64,
64        dt => dt.clone(),
65    }
66}

68impl<T> GroupedReduction for SumReduce<T>
69where
70    T: PolarsNumericType,
71    ChunkedArray<T>: ChunkAgg<T::Native> + IntoSeries,
72{
73    fn new_empty(&self) -> Box<dyn GroupedReduction> {
74        Box::new(Self {
75            sums: Vec::new(),
76            in_dtype: self.in_dtype.clone(),
77        })
78    }
79
80    fn reserve(&mut self, additional: usize) {
81        self.sums.reserve(additional);
82    }
83
84    fn resize(&mut self, num_groups: IdxSize) {
85        self.sums.resize(num_groups as usize, T::Native::zero());
86    }
87
88    fn update_group(
89        &mut self,
90        values: &Series,
91        group_idx: IdxSize,
92        _seq_id: u64,
93    ) -> PolarsResult<()> {
94        // TODO: we should really implement a sum-as-other-type operation instead
95        // of doing this materialized cast.
96        assert!(values.dtype() == &self.in_dtype);
97        let values = cast_sum_input(values, &self.in_dtype)?;
98        let ca: &ChunkedArray<T> = values.as_ref().as_ref().as_ref();
99        self.sums[group_idx as usize] += ChunkAgg::sum(ca).unwrap_or(T::Native::zero());
100        Ok(())
101    }
102
103    unsafe fn update_groups(
104        &mut self,
105        values: &Series,
106        group_idxs: &[IdxSize],
107        _seq_id: u64,
108    ) -> PolarsResult<()> {
109        // TODO: we should really implement a sum-as-other-type operation instead
110        // of doing this materialized cast.
111        assert!(values.dtype() == &self.in_dtype);
112        let values = cast_sum_input(values, &self.in_dtype)?;
113        assert!(values.len() == group_idxs.len());
114        let ca: &ChunkedArray<T> = values.as_ref().as_ref().as_ref();
115        unsafe {
116            // SAFETY: indices are in-bounds guaranteed by trait.
117            for (g, v) in group_idxs.iter().zip(ca.iter()) {
118                *self.sums.get_unchecked_mut(*g as usize) += v.unwrap_or(T::Native::zero());
119            }
120        }
121        Ok(())
122    }
123
124    unsafe fn combine(
125        &mut self,
126        other: &dyn GroupedReduction,
127        group_idxs: &[IdxSize],
128    ) -> PolarsResult<()> {
129        let other = other.as_any().downcast_ref::<Self>().unwrap();
130        assert!(self.in_dtype == other.in_dtype);
131        assert!(other.sums.len() == group_idxs.len());
132        unsafe {
133            // SAFETY: indices are in-bounds guaranteed by trait.
134            for (g, v) in group_idxs.iter().zip(other.sums.iter()) {
135                *self.sums.get_unchecked_mut(*g as usize) += *v;
136            }
137        }
138        Ok(())
139    }
140
141    unsafe fn gather_combine(
142        &mut self,
143        other: &dyn GroupedReduction,
144        subset: &[IdxSize],
145        group_idxs: &[IdxSize],
146    ) -> PolarsResult<()> {
147        let other = other.as_any().downcast_ref::<Self>().unwrap();
148        assert!(self.in_dtype == other.in_dtype);
149        assert!(subset.len() == group_idxs.len());
150        unsafe {
151            // SAFETY: indices are in-bounds guaranteed by trait.
152            for (i, g) in subset.iter().zip(group_idxs) {
153                *self.sums.get_unchecked_mut(*g as usize) += *other.sums.get_unchecked(*i as usize);
154            }
155        }
156        Ok(())
157    }
158
159    unsafe fn partition(
160        self: Box<Self>,
161        partition_sizes: &[IdxSize],
162        partition_idxs: &[IdxSize],
163    ) -> Vec<Box<dyn GroupedReduction>> {
164        partition::partition_vec(self.sums, partition_sizes, partition_idxs)
165            .into_iter()
166            .map(|sums| {
167                Box::new(Self {
168                    sums,
169                    in_dtype: self.in_dtype.clone(),
170                }) as _
171            })
172            .collect()
173    }
174
175    fn finalize(&mut self) -> PolarsResult<Series> {
176        let v = core::mem::take(&mut self.sums);
177        let arr = Box::new(PrimitiveArray::<T::Native>::from_vec(v));
178        Ok(unsafe {
179            Series::from_chunks_and_dtype_unchecked(
180                PlSmallStr::EMPTY,
181                vec![arr],
182                &out_dtype(&self.in_dtype),
183            )
184        })
185    }
186
187    fn as_any(&self) -> &dyn Any {
188        self
189    }

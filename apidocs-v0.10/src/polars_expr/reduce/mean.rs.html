8pub fn new_mean_reduction(dtype: DataType) -> Box<dyn GroupedReduction> {
9    use DataType::*;
10    use VecGroupedReduction as VGR;
11    match dtype {
12        Boolean => Box::new(VGR::new(dtype, BoolMeanReducer)),
13        _ if dtype.is_primitive_numeric() || dtype.is_temporal() => {
14            with_match_physical_numeric_polars_type!(dtype.to_physical(), |$T| {
15                Box::new(VGR::new(dtype, NumMeanReducer::<$T>(PhantomData)))
16            })
17        },
18        #[cfg(feature = "dtype-decimal")]
19        Decimal(_, _) => Box::new(VGR::new(dtype, NumMeanReducer::<Int128Type>(PhantomData))),
20
21        // For compatibility with the current engine, should probably be an error.
22        String | Binary => Box::new(super::NullGroupedReduction::new(dtype)),
23
24        _ => unimplemented!("{dtype:?} is not supported by mean reduction"),
25    }
26}

28fn finish_output(values: Vec<(f64, usize)>, dtype: &DataType) -> Series {
29    match dtype {
30        DataType::Float32 => {
31            let ca: Float32Chunked = values
32                .into_iter()
33                .map(|(s, c)| (c != 0).then(|| (s / c as f64) as f32))
34                .collect_ca(PlSmallStr::EMPTY);
35            ca.into_series()
36        },
37        dt if dt.is_primitive_numeric() => {
38            let ca: Float64Chunked = values
39                .into_iter()
40                .map(|(s, c)| (c != 0).then(|| s / c as f64))
41                .collect_ca(PlSmallStr::EMPTY);
42            ca.into_series()
43        },
44        #[cfg(feature = "dtype-decimal")]
45        DataType::Decimal(_prec, scale) => {
46            let inv_scale_factor = 1.0 / 10u128.pow(scale.unwrap() as u32) as f64;
47            let ca: Float64Chunked = values
48                .into_iter()
49                .map(|(s, c)| (c != 0).then(|| s / c as f64 * inv_scale_factor))
50                .collect_ca(PlSmallStr::EMPTY);
51            ca.into_series()
52        },
53        #[cfg(feature = "dtype-datetime")]
54        DataType::Date => {
55            const MS_IN_DAY: i64 = 86_400_000;
56            let ca: Int64Chunked = values
57                .into_iter()
58                .map(|(s, c)| (c != 0).then(|| (s / c as f64 * MS_IN_DAY as f64) as i64))
59                .collect_ca(PlSmallStr::EMPTY);
60            ca.into_datetime(TimeUnit::Milliseconds, None).into_series()
61        },
62        DataType::Datetime(_, _) | DataType::Duration(_) | DataType::Time => {
63            let ca: Int64Chunked = values
64                .into_iter()
65                .map(|(s, c)| (c != 0).then(|| (s / c as f64) as i64))
66                .collect_ca(PlSmallStr::EMPTY);
67            ca.into_series().cast(dtype).unwrap()
68        },
69        _ => unimplemented!(),
70    }
71}

73struct NumMeanReducer<T>(PhantomData<T>);
74impl<T> Clone for NumMeanReducer<T> {
75    fn clone(&self) -> Self {
76        Self(PhantomData)
77    }
78}
79
80impl<T> Reducer for NumMeanReducer<T>
81where
82    T: PolarsNumericType,
83    ChunkedArray<T>: ChunkAgg<T::Native> + IntoSeries,
84{
85    type Dtype = T;
86    type Value = (f64, usize);
87
88    #[inline(always)]
89    fn init(&self) -> Self::Value {
90        (0.0, 0)
91    }
92
93    fn cast_series<'a>(&self, s: &'a Series) -> Cow<'a, Series> {
94        s.to_physical_repr()
95    }
96
97    #[inline(always)]
98    fn combine(&self, a: &mut Self::Value, b: &Self::Value) {
99        a.0 += b.0;
100        a.1 += b.1;
101    }
102
103    #[inline(always)]
104    fn reduce_one(&self, a: &mut Self::Value, b: Option<T::Native>, _seq_id: u64) {
105        a.0 += b.unwrap_or(T::Native::zero()).as_();
106        a.1 += b.is_some() as usize;
107    }
108
109    fn reduce_ca(&self, v: &mut Self::Value, ca: &ChunkedArray<Self::Dtype>, _seq_id: u64) {
110        v.0 += ChunkAgg::_sum_as_f64(ca);
111        v.1 += ca.len() - ca.null_count();
112    }
113
114    fn finish(
115        &self,
116        v: Vec<Self::Value>,
117        m: Option<Bitmap>,
118        dtype: &DataType,
119    ) -> PolarsResult<Series> {
120        assert!(m.is_none());
121        Ok(finish_output(v, dtype))
122    }
123}

126struct BoolMeanReducer;

128impl Reducer for BoolMeanReducer {
129    type Dtype = BooleanType;
130    type Value = (usize, usize);
131
132    #[inline(always)]
133    fn init(&self) -> Self::Value {
134        (0, 0)
135    }
136
137    #[inline(always)]
138    fn combine(&self, a: &mut Self::Value, b: &Self::Value) {
139        a.0 += b.0;
140        a.1 += b.1;
141    }
142
143    #[inline(always)]
144    fn reduce_one(&self, a: &mut Self::Value, b: Option<bool>, _seq_id: u64) {
145        a.0 += b.unwrap_or(false) as usize;
146        a.1 += b.is_some() as usize;
147    }
148
149    fn reduce_ca(&self, v: &mut Self::Value, ca: &ChunkedArray<Self::Dtype>, _seq_id: u64) {
150        v.0 += ca.sum().unwrap_or(0) as usize;
151        v.1 += ca.len() - ca.null_count();
152    }
153
154    fn finish(
155        &self,
156        v: Vec<Self::Value>,
157        m: Option<Bitmap>,
158        dtype: &DataType,
159    ) -> PolarsResult<Series> {
160        assert!(m.is_none());
161        assert!(dtype == &DataType::Boolean);
162        let ca: Float64Chunked = v
163            .into_iter()
164            .map(|(s, c)| (c != 0).then(|| s as f64 / c as f64))
165            .collect_ca(PlSmallStr::EMPTY);
166        Ok(ca.into_series())
167    }
168}

3pub(super) fn convert_st_union(
4    inputs: &mut [Node],
5    lp_arena: &mut Arena<IR>,
6    expr_arena: &mut Arena<AExpr>,
7) -> PolarsResult<()> {
8    let mut schema = (**lp_arena.get(inputs[0]).schema(lp_arena)).clone();
9
10    let mut changed = false;
11    for input in inputs[1..].iter() {
12        let schema_other = lp_arena.get(*input).schema(lp_arena);
13        changed |= schema.to_supertype(schema_other.as_ref())?;
14    }
15
16    if changed {
17        for input in inputs {
18            let mut exprs = vec![];
19            let input_schema = lp_arena.get(*input).schema(lp_arena);
20
21            let to_cast = input_schema.iter().zip(schema.iter_values()).flat_map(
22                |((left_name, left_type), st)| {
23                    if left_type != st {
24                        Some(col(left_name.clone()).cast(st.clone()))
25                    } else {
26                        None
27                    }
28                },
29            );
30            exprs.extend(to_cast);
31
32            if !exprs.is_empty() {
33                let expr = to_expr_irs(exprs, expr_arena)?;
34                let lp = IRBuilder::new(*input, expr_arena, lp_arena)
35                    .with_columns(expr, Default::default())
36                    .build();
37
38                let node = lp_arena.add(lp);
39                *input = node
40            }
41        }
42    }
43    Ok(())
44}

46fn nodes_to_schemas(inputs: &[Node], lp_arena: &mut Arena<IR>) -> Vec<SchemaRef> {
47    inputs
48        .iter()
49        .map(|n| lp_arena.get(*n).schema(lp_arena).into_owned())
50        .collect()

53pub(super) fn convert_diagonal_concat(
54    mut inputs: Vec<Node>,
55    lp_arena: &mut Arena<IR>,
56    expr_arena: &mut Arena<AExpr>,
57) -> PolarsResult<Vec<Node>> {
58    let schemas = nodes_to_schemas(&inputs, lp_arena);
59
60    let upper_bound_width = schemas.iter().map(|sch| sch.len()).sum();
61
62    let mut total_schema = Schema::with_capacity(upper_bound_width);
63
64    for sch in schemas.iter() {
65        sch.iter().for_each(|(name, dtype)| {
66            if !total_schema.contains(name) {
67                total_schema.with_column(name.as_str().into(), dtype.clone());
68            }
69        });
70    }
71    if total_schema.is_empty() {
72        return Ok(inputs);
73    }
74
75    let mut has_empty = false;
76
77    for (node, lf_schema) in inputs.iter_mut().zip(schemas.iter()) {
78        // Discard, this works physically
79        if lf_schema.is_empty() {
80            has_empty = true;
81        }
82        let mut columns_to_add = vec![];
83
84        for (name, dtype) in total_schema.iter() {
85            // If a name from Total Schema is not present - append
86            if lf_schema.get_field(name).is_none() {
87                columns_to_add.push(NULL.lit().cast(dtype.clone()).alias(name.clone()))
88            }
89        }
90        let expr = to_expr_irs(columns_to_add, expr_arena)?;
91        *node = IRBuilder::new(*node, expr_arena, lp_arena)
92            // Add the missing columns
93            .with_columns(expr, Default::default())
94            // Now, reorder to match schema.
95            .project_simple(total_schema.iter_names().map(|v| v.as_str()))
96            .unwrap()
97            .node();
98    }
99
100    if has_empty {
101        Ok(inputs
102            .into_iter()
103            .zip(schemas)
104            .filter_map(|(input, schema)| if schema.is_empty() { None } else { Some(input) })
105            .collect())
106    } else {
107        Ok(inputs)
108    }
109}

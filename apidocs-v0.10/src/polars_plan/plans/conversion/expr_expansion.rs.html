5pub(crate) fn prepare_projection(
6    exprs: Vec<Expr>,
7    schema: &Schema,
8    opt_flags: &mut OptFlags,
9) -> PolarsResult<(Vec<Expr>, Schema)> {
10    let exprs = rewrite_projections(exprs, schema, &[], opt_flags)?;
11    let schema = expressions_to_schema(&exprs, schema, Context::Default)?;
12    Ok((exprs, schema))

15/// This replaces the wildcard Expr with a Column Expr. It also removes the Exclude Expr from the
16/// expression chain.
17pub(super) fn replace_wildcard_with_column(expr: Expr, column_name: &PlSmallStr) -> Expr {
18    expr.map_expr(|e| match e {
19        Expr::Wildcard => Expr::Column(column_name.clone()),
20        Expr::Exclude(input, _) => Arc::unwrap_or_clone(input),
21        e => e,
22    })
23}

28        Expr::Exclude(input, _) => Arc::unwrap_or_clone(input),
29        e => e,
30    })

33fn rewrite_special_aliases(expr: Expr) -> PolarsResult<Expr> {
34    // the blocks are added by cargo fmt
35    if has_expr(&expr, |e| {
36        matches!(e, Expr::KeepName(_) | Expr::RenameAlias { .. })
37    }) {
38        match expr {
39            Expr::KeepName(expr) => {
40                let roots = expr_to_leaf_column_names(&expr);
41                let name = roots
42                    .first()
43                    .expect("expected root column to keep expression name");
44                Ok(Expr::Alias(expr, name.clone()))
45            },
46            Expr::RenameAlias { expr, function } => {
47                let name = get_single_leaf(&expr)?;
48                let name = function.call(&name)?;
49                Ok(Expr::Alias(expr, name))
50            },
51            _ => {
52                polars_bail!(InvalidOperation: "`keep`, `suffix`, `prefix` should be last expression")
53            },
54        }
55    } else {
56        Ok(expr)
57    }
58}

60/// Take an expression with a root: col("*") and copies that expression for all columns in the schema,
61/// with the exclusion of the `names` in the exclude expression.
62/// The resulting expressions are written to result.
63fn replace_wildcard(
64    expr: &Expr,
65    result: &mut Vec<Expr>,
66    exclude: &PlHashSet<PlSmallStr>,
67    schema: &Schema,
68) -> PolarsResult<()> {
69    for name in schema.iter_names() {
70        if !exclude.contains(name.as_str()) {
71            let new_expr = replace_wildcard_with_column(expr.clone(), name);
72            let new_expr = rewrite_special_aliases(new_expr)?;
73            result.push(new_expr)
74        }
75    }
76    Ok(())
77}

79fn replace_nth(expr: Expr, schema: &Schema) -> Expr {
80    expr.map_expr(|e| {
81        if let Expr::Nth(i) = e {
82            match i.negative_to_usize(schema.len()) {
83                None => {
84                    let name = match i {
85                        0 => "first",
86                        -1 => "last",
87                        _ => "nth",
88                    };
89                    Expr::Column(PlSmallStr::from_static(name))
90                },
91                Some(idx) => {
92                    let (name, _dtype) = schema.get_at_index(idx).unwrap();
93                    Expr::Column(name.clone())
94                },
95            }
96        } else {
97            e
98        }
99    })
100}

102#[cfg(feature = "regex")]
103/// This function takes an expression containing a regex in `col("..")` and expands the columns
104/// that are selected by that regex in `result`.
105fn expand_regex(
106    expr: &Expr,
107    result: &mut Vec<Expr>,
108    schema: &Schema,
109    pattern: &str,
110    exclude: &PlHashSet<PlSmallStr>,
111) -> PolarsResult<()> {
112    let re =
113        regex::Regex::new(pattern).map_err(|e| polars_err!(ComputeError: "invalid regex {}", e))?;
114    for name in schema.iter_names() {
115        if re.is_match(name) && !exclude.contains(name.as_str()) {
116            let mut new_expr = remove_exclude(expr.clone());
117
118            new_expr = new_expr.map_expr(|e| match e {
119                Expr::Column(pat) if pat.as_str() == pattern => Expr::Column(name.clone()),
120                e => e,
121            });
122
123            let new_expr = rewrite_special_aliases(new_expr)?;
124            result.push(new_expr);
125        }
126    }
127    Ok(())
128}

130pub(crate) fn is_regex_projection(name: &str) -> bool {
131    name.starts_with('^') && name.ends_with('$')
132}

134#[cfg(feature = "regex")]
135/// This function searches for a regex expression in `col("..")` and expands the columns
136/// that are selected by that regex in `result`. The regex should start with `^` and end with `$`.
137fn replace_regex(
138    expr: &Expr,
139    result: &mut Vec<Expr>,
140    schema: &Schema,
141    exclude: &PlHashSet<PlSmallStr>,
142) -> PolarsResult<()> {
143    let roots = expr_to_leaf_column_names(expr);
144    let mut regex = None;
145    for name in &roots {
146        if is_regex_projection(name) {
147            match regex {
148                None => {
149                    regex = Some(name);
150                    expand_regex(expr, result, schema, name, exclude)?;
151                },
152                Some(r) => {
153                    polars_ensure!(
154                        r == name,
155                        ComputeError:
156                        "an expression is not allowed to have different regexes"
157                    )
158                },
159            }
160        }
161    }
162    if regex.is_none() {
163        let expr = rewrite_special_aliases(expr.clone())?;
164        result.push(expr)
165    }
166    Ok(())
167}

169/// replace `columns(["A", "B"])..` with `col("A")..`, `col("B")..`
170#[allow(unused_variables)]
171fn expand_columns(
172    expr: &Expr,
173    result: &mut Vec<Expr>,
174    names: &[PlSmallStr],
175    schema: &Schema,
176    exclude: &PlHashSet<PlSmallStr>,
177) -> PolarsResult<()> {
178    if !expr.into_iter().all(|e| match e {
179        // check for invalid expansions such as `col([a, b]) + col([c, d])`
180        Expr::Columns(ref members) => members.as_ref() == names,
181        _ => true,
182    }) {
183        polars_bail!(ComputeError: "expanding more than one `col` is not allowed");
184    }
185    for name in names {
186        if !exclude.contains(name) {
187            let new_expr = expr.clone().map_expr(|e| match e {
188                Expr::Columns(_) => Expr::Column((*name).clone()),
189                Expr::Exclude(input, _) => Arc::unwrap_or_clone(input),
190                e => e,
191            });
192
193            #[cfg(feature = "regex")]
194            replace_regex(&new_expr, result, schema, exclude)?;
195
196            #[cfg(not(feature = "regex"))]
197            result.push(rewrite_special_aliases(new_expr)?);
198        }
199    }
200    Ok(())
201}

203#[cfg(feature = "dtype-struct")]
204fn struct_index_to_field(expr: Expr, schema: &Schema) -> PolarsResult<Expr> {
205    expr.try_map_expr(|e| match e {
206        Expr::Function {
207            input,
208            function: FunctionExpr::StructExpr(sf),
209            options,
210        } => {
211            if let StructFunction::FieldByIndex(index) = sf {
212                let dtype = input[0].to_field(schema, Context::Default)?.dtype;
213                let DataType::Struct(fields) = dtype else {
214                    polars_bail!(InvalidOperation: "expected 'struct' dtype, got {:?}", dtype)
215                };
216                let index = index.try_negative_to_usize(fields.len())?;
217                let name = fields[index].name.clone();
218                Ok(Expr::Function {
219                    input,
220                    function: FunctionExpr::StructExpr(StructFunction::FieldByName(name)),
221                    options,
222                })
223            } else {
224                Ok(Expr::Function {
225                    input,
226                    function: FunctionExpr::StructExpr(sf),
227                    options,
228                })
229            }
230        },
231        e => Ok(e),
232    })
233}

235/// This replaces the dtype or index expanded Expr with a Column Expr.
236/// ()It also removes the Exclude Expr from the expression chain).
237fn replace_dtype_or_index_with_column(
238    expr: Expr,
239    column_name: &PlSmallStr,
240    replace_dtype: bool,
241) -> Expr {
242    expr.map_expr(|e| match e {
243        Expr::DtypeColumn(_) if replace_dtype => Expr::Column(column_name.clone()),
244        Expr::IndexColumn(_) if !replace_dtype => Expr::Column(column_name.clone()),
245        Expr::Exclude(input, _) => Arc::unwrap_or_clone(input),
246        e => e,
247    })
248}

250fn dtypes_match(d1: &DataType, d2: &DataType) -> bool {
251    match (d1, d2) {
252        // note: allow Datetime "*" wildcard for timezones...
253        (DataType::Datetime(tu_l, tz_l), DataType::Datetime(tu_r, tz_r)) => {
254            tu_l == tu_r
255                && (tz_l == tz_r
256                    || tz_r.is_some() && (tz_l.as_deref().unwrap_or("") == "*")
257                    || tz_l.is_some() && (tz_r.as_deref().unwrap_or("") == "*"))
258        },
259        // ...but otherwise require exact match
260        _ => d1 == d2,
261    }
262}

264/// replace `DtypeColumn` with `col("foo")..col("bar")`
265fn expand_dtypes(
266    expr: &Expr,
267    result: &mut Vec<Expr>,
268    schema: &Schema,
269    dtypes: &[DataType],
270    exclude: &PlHashSet<PlSmallStr>,
271) -> PolarsResult<()> {
272    // note: we loop over the schema to guarantee that we return a stable
273    // field-order, irrespective of which dtypes are filtered against
274    for field in schema.iter_fields().filter(|f| {
275        dtypes.iter().any(|dtype| dtypes_match(dtype, &f.dtype))
276            && !exclude.contains(f.name().as_str())
277    }) {
278        let name = field.name();
279        let new_expr = expr.clone();
280        let new_expr = replace_dtype_or_index_with_column(new_expr, name, true);
281        let new_expr = rewrite_special_aliases(new_expr)?;
282        result.push(new_expr)
283    }
284    Ok(())
285}

287#[cfg(feature = "dtype-struct")]
288fn replace_struct_multiple_fields_with_field(
289    expr: Expr,
290    column_name: &PlSmallStr,
291) -> PolarsResult<Expr> {
292    let mut count = 0;
293    let out = expr.map_expr(|e| match e {
294        Expr::Function {
295            function,
296            input,
297            options,
298        } => {
299            if matches!(
300                function,
301                FunctionExpr::StructExpr(StructFunction::MultipleFields(_))
302            ) {
303                count += 1;
304                Expr::Function {
305                    input,
306                    function: FunctionExpr::StructExpr(StructFunction::FieldByName(
307                        column_name.clone(),
308                    )),
309                    options,
310                }
311            } else {
312                Expr::Function {
313                    input,
314                    function,
315                    options,
316                }
317            }
318        },
319        e => e,
320    });
321    polars_ensure!(count == 1, InvalidOperation: "multiple expanding fields in a single struct not yet supported");
322    Ok(out)
323}

325#[cfg(feature = "dtype-struct")]
326fn expand_struct_fields(
327    struct_expr: &Expr,
328    full_expr: &Expr,
329    result: &mut Vec<Expr>,
330    schema: &Schema,
331    names: &[PlSmallStr],
332    exclude: &PlHashSet<PlSmallStr>,
333) -> PolarsResult<()> {
334    let first_name = names[0].as_ref();
335    if names.len() == 1 && first_name == "*" || is_regex_projection(first_name) {
336        let Expr::Function { input, .. } = struct_expr else {
337            unreachable!()
338        };
339        let field = input[0].to_field(schema, Context::Default)?;
340        let dtype = field.dtype();
341        let DataType::Struct(fields) = dtype else {
342            if !dtype.is_known() {
343                let mut msg = String::from(
344                    "expected 'struct' got an unknown data type
345
346This means there was an operation of which the output data type could not be determined statically.
347Try setting the output data type for that operation.",
348                );
349                for e in input[0].into_iter() {
350                    #[allow(clippy::single_match)]
351                    match e {
352                        #[cfg(feature = "list_to_struct")]
353                        Expr::Function {
354                            input: _,
355                            function,
356                            options: _,
357                        } => {
358                            if matches!(
359                                function,
360                                FunctionExpr::ListExpr(ListFunction::ToStruct(..))
361                            ) {
362                                msg.push_str(
363                                    "
364
365Hint: set 'upper_bound' for 'list.to_struct'.",
366                                );
367                            }
368                        },
369                        _ => {},
370                    }
371                }
372
373                polars_bail!(InvalidOperation: msg)
374            } else {
375                polars_bail!(InvalidOperation: "expected 'struct' got {}", field.dtype())
376            }
377        };
378
379        // Wildcard.
380        let names = if first_name == "*" {
381            fields
382                .iter()
383                .flat_map(|field| {
384                    let name = field.name();
385
386                    if exclude.contains(name.as_str()) {
387                        None
388                    } else {
389                        Some(name.clone())
390                    }
391                })
392                .collect::<Vec<_>>()
393        }
394        // Regex
395        else {
396            #[cfg(feature = "regex")]
397            {
398                let re = regex::Regex::new(first_name)
399                    .map_err(|e| polars_err!(ComputeError: "invalid regex {}", e))?;
400
401                fields
402                    .iter()
403                    .flat_map(|field| {
404                        let name = field.name();
405                        if exclude.contains(name.as_str()) || !re.is_match(name.as_str()) {
406                            None
407                        } else {
408                            Some(name.clone())
409                        }
410                    })
411                    .collect::<Vec<_>>()
412            }
413            #[cfg(not(feature = "regex"))]
414            {
415                panic!("activate 'regex' feature")
416            }
417        };
418
419        return expand_struct_fields(
420            struct_expr,
421            full_expr,
422            result,
423            schema,
424            names.as_slice(),
425            exclude,
426        );
427    }
428
429    for name in names {
430        polars_ensure!(name.as_str() != "*", InvalidOperation: "cannot combine wildcards and column names");
431
432        if !exclude.contains(name) {
433            let mut new_expr = replace_struct_multiple_fields_with_field(full_expr.clone(), name)?;
434            match new_expr {
435                Expr::KeepName(expr) => {
436                    new_expr = Expr::Alias(expr, name.clone());
437                },
438                Expr::RenameAlias { expr, function } => {
439                    let name = function.call(name)?;
440                    new_expr = Expr::Alias(expr, name);
441                },
442                _ => {},
443            }
444
445            result.push(new_expr)
446        }
447    }
448    Ok(())
449}

451/// replace `IndexColumn` with `col("foo")..col("bar")`
452fn expand_indices(
453    expr: &Expr,
454    result: &mut Vec<Expr>,
455    schema: &Schema,
456    indices: &[i64],
457    exclude: &PlHashSet<PlSmallStr>,
458) -> PolarsResult<()> {
459    let n_fields = schema.len() as i64;
460    for idx in indices {
461        let mut idx = *idx;
462        if idx < 0 {
463            idx += n_fields;
464            if idx < 0 {
465                polars_bail!(ComputeError: "invalid column index {}", idx)
466            }
467        }
468        if let Some((name, _)) = schema.get_at_index(idx as usize) {
469            if !exclude.contains(name.as_str()) {
470                let new_expr = expr.clone();
471                let new_expr = replace_dtype_or_index_with_column(new_expr, name, false);
472                let new_expr = rewrite_special_aliases(new_expr)?;
473                result.push(new_expr);
474            }
475        }
476    }
477    Ok(())
478}

480// schema is not used if regex not activated
481#[allow(unused_variables)]
482fn prepare_excluded(
483    expr: &Expr,
484    schema: &Schema,
485    keys: &[Expr],
486    has_exclude: bool,
487) -> PolarsResult<PlHashSet<PlSmallStr>> {
488    let mut exclude = PlHashSet::new();
489
490    // explicit exclude branch
491    if has_exclude {
492        for e in expr {
493            if let Expr::Exclude(_, to_exclude) = e {
494                #[cfg(feature = "regex")]
495                {
496                    // instead of matching the names for regex patterns and
497                    // expanding the matches in the schema we reuse the
498                    // `replace_regex` func; this is a bit slower but DRY.
499                    let mut buf = vec![];
500                    for to_exclude_single in to_exclude {
501                        match to_exclude_single {
502                            Excluded::Name(name) => {
503                                let e = Expr::Column(name.clone());
504                                replace_regex(&e, &mut buf, schema, &Default::default())?;
505                                // we cannot loop because of bchck
506                                while let Some(col) = buf.pop() {
507                                    if let Expr::Column(name) = col {
508                                        exclude.insert(name);
509                                    }
510                                }
511                            },
512                            Excluded::Dtype(dt) => {
513                                for fld in schema.iter_fields() {
514                                    if dtypes_match(fld.dtype(), dt) {
515                                        exclude.insert(fld.name.clone());
516                                    }
517                                }
518                            },
519                        }
520                    }
521                }
522
523                #[cfg(not(feature = "regex"))]
524                {
525                    for to_exclude_single in to_exclude {
526                        match to_exclude_single {
527                            Excluded::Name(name) => {
528                                exclude.insert(name.clone());
529                            },
530                            Excluded::Dtype(dt) => {
531                                for (name, dtype) in schema.iter() {
532                                    if matches!(dtype, dt) {
533                                        exclude.insert(name.clone());
534                                    }
535                                }
536                            },
537                        }
538                    }
539                }
540            }
541        }
542    }
543
544    // exclude group_by keys
545    for expr in keys.iter() {
546        if let Ok(name) = expr_output_name(expr) {
547            exclude.insert(name.clone());
548        }
549    }
550    Ok(exclude)
551}

553// functions can have col(["a", "b"]) or col(String) as inputs
554fn expand_function_inputs(
555    expr: Expr,
556    schema: &Schema,
557    opt_flags: &mut OptFlags,
558) -> PolarsResult<Expr> {
559    expr.try_map_expr(|mut e| match &mut e {
560        Expr::AnonymousFunction { input, options, .. } | Expr::Function { input, options, .. }
561            if options
562                .flags
563                .contains(FunctionFlags::INPUT_WILDCARD_EXPANSION) =>
564        {
565            *input = rewrite_projections(core::mem::take(input), schema, &[], opt_flags)?;
566            if input.is_empty() && !options.flags.contains(FunctionFlags::ALLOW_EMPTY_INPUTS) {
567                // Needed to visualize the error
568                *input = vec![Expr::Literal(LiteralValue::Null)];
569                polars_bail!(InvalidOperation: "expected at least 1 input in {}", e)
570            }
571            Ok(e)
572        },
573        _ => Ok(e),
574    })
575}

577#[derive(Copy, Clone, Debug)]
578struct ExpansionFlags {
579    multiple_columns: bool,
580    has_nth: bool,
581    has_wildcard: bool,
582    has_selector: bool,
583    has_exclude: bool,
584    #[cfg(feature = "dtype-struct")]
585    expands_fields: bool,
586    #[cfg(feature = "dtype-struct")]
587    has_struct_field_by_index: bool,
588}
589
590impl ExpansionFlags {
591    fn expands(&self) -> bool {
592        #[cfg(feature = "dtype-struct")]
593        let expands_fields = self.expands_fields;
594        #[cfg(not(feature = "dtype-struct"))]
595        let expands_fields = false;
596
597        self.multiple_columns || expands_fields
598    }
599}
600
601fn find_flags(expr: &Expr) -> PolarsResult<ExpansionFlags> {
602    let mut multiple_columns = false;
603    let mut has_nth = false;
604    let mut has_wildcard = false;
605    let mut has_selector = false;
606    let mut has_exclude = false;
607    #[cfg(feature = "dtype-struct")]
608    let mut has_struct_field_by_index = false;
609    #[cfg(feature = "dtype-struct")]
610    let mut expands_fields = false;
611
612    // Do a single pass and collect all flags at once.
613    // Supertypes/modification that can be done in place are also done in that pass
614    for expr in expr {
615        match expr {
616            Expr::Columns(_) | Expr::DtypeColumn(_) => multiple_columns = true,
617            Expr::IndexColumn(idx) => multiple_columns = idx.len() > 1,
618            Expr::Nth(_) => has_nth = true,
619            Expr::Wildcard => has_wildcard = true,
620            Expr::Selector(_) => has_selector = true,
621            #[cfg(feature = "dtype-struct")]
622            Expr::Function {
623                function: FunctionExpr::StructExpr(StructFunction::FieldByIndex(_)),
624                ..
625            } => {
626                has_struct_field_by_index = true;
627            },
628            #[cfg(feature = "dtype-struct")]
629            Expr::Function {
630                function: FunctionExpr::StructExpr(StructFunction::MultipleFields(_)),
631                ..
632            } => {
633                expands_fields = true;
634            },
635            Expr::Exclude(_, _) => has_exclude = true,
636            #[cfg(feature = "dtype-struct")]
637            Expr::Field(_) => {
638                polars_bail!(InvalidOperation: "field expression not allowed at location/context")
639            },
640            _ => {},
641        }
642    }
643    Ok(ExpansionFlags {
644        multiple_columns,
645        has_nth,
646        has_wildcard,
647        has_selector,
648        has_exclude,
649        #[cfg(feature = "dtype-struct")]
650        has_struct_field_by_index,
651        #[cfg(feature = "dtype-struct")]
652        expands_fields,
653    })
654}

656#[cfg(feature = "dtype-struct")]
657fn toggle_cse(opt_flags: &mut OptFlags) {
658    if opt_flags.contains(OptFlags::EAGER) && !opt_flags.contains(OptFlags::NEW_STREAMING) {
659        #[cfg(debug_assertions)]
660        {
661            use polars_core::config::verbose;
662            if verbose() {
663                eprintln!("CSE turned on because of struct expansion")
664            }
665        }
666        *opt_flags |= OptFlags::COMM_SUBEXPR_ELIM;
667    }
668}

670/// In case of single col(*) -> do nothing, no selection is the same as select all
671/// In other cases replace the wildcard with an expression with all columns
672pub(crate) fn rewrite_projections(
673    exprs: Vec<Expr>,
674    schema: &Schema,
675    keys: &[Expr],
676    opt_flags: &mut OptFlags,
677) -> PolarsResult<Vec<Expr>> {
678    let mut result = Vec::with_capacity(exprs.len() + schema.len());
679
680    for mut expr in exprs {
681        #[cfg(feature = "dtype-struct")]
682        let result_offset = result.len();
683
684        // Functions can have col(["a", "b"]) or col(String) as inputs.
685        expr = expand_function_inputs(expr, schema, opt_flags)?;
686
687        let mut flags = find_flags(&expr)?;
688        if flags.has_selector {
689            expr = replace_selector(expr, schema, keys)?;
690            // the selector is replaced with Expr::Columns
691            flags.multiple_columns = true;
692        }
693
694        replace_and_add_to_results(expr, flags, &mut result, schema, keys, opt_flags)?;
695
696        #[cfg(feature = "dtype-struct")]
697        if flags.has_struct_field_by_index {
698            toggle_cse(opt_flags);
699            for e in &mut result[result_offset..] {
700                *e = struct_index_to_field(std::mem::take(e), schema)?;
701            }
702        }
703    }
704    Ok(result)
705}

707fn replace_and_add_to_results(
708    mut expr: Expr,
709    flags: ExpansionFlags,
710    result: &mut Vec<Expr>,
711    schema: &Schema,
712    keys: &[Expr],
713    opt_flags: &mut OptFlags,
714) -> PolarsResult<()> {
715    if flags.has_nth {
716        expr = replace_nth(expr, schema);
717    }
718
719    // has multiple column names
720    // the expanded columns are added to the result
721    if flags.expands() {
722        if let Some(e) = expr.into_iter().find(|e| match e {
723            Expr::Columns(_) | Expr::DtypeColumn(_) | Expr::IndexColumn(_) => true,
724            #[cfg(feature = "dtype-struct")]
725            Expr::Function {
726                function: FunctionExpr::StructExpr(StructFunction::MultipleFields(_)),
727                ..
728            } => flags.expands_fields,
729            _ => false,
730        }) {
731            match &e {
732                Expr::Columns(names) => {
733                    // Don't exclude grouping keys if columns are explicitly specified.
734                    let exclude = prepare_excluded(&expr, schema, &[], flags.has_exclude)?;
735                    expand_columns(&expr, result, names, schema, &exclude)?;
736                },
737                Expr::DtypeColumn(dtypes) => {
738                    let exclude = prepare_excluded(&expr, schema, keys, flags.has_exclude)?;
739                    expand_dtypes(&expr, result, schema, dtypes, &exclude)?
740                },
741                Expr::IndexColumn(indices) => {
742                    let exclude = prepare_excluded(&expr, schema, keys, flags.has_exclude)?;
743                    expand_indices(&expr, result, schema, indices, &exclude)?
744                },
745                #[cfg(feature = "dtype-struct")]
746                Expr::Function { function, .. } => {
747                    let FunctionExpr::StructExpr(StructFunction::MultipleFields(names)) = function
748                    else {
749                        unreachable!()
750                    };
751                    let exclude = prepare_excluded(&expr, schema, keys, flags.has_exclude)?;
752
753                    // has both column and field expansion
754                    // col('a', 'b').struct.field('*')
755                    if flags.multiple_columns | flags.has_wildcard {
756                        // First expand col('a', 'b') into an intermediate result.
757                        let mut intermediate = vec![];
758                        let mut flags = flags;
759                        flags.expands_fields = false;
760                        replace_and_add_to_results(
761                            expr.clone(),
762                            flags,
763                            &mut intermediate,
764                            schema,
765                            keys,
766                            opt_flags,
767                        )?;
768
769                        // Then expand the fields and add to the final result vec.
770                        flags.expands_fields = true;
771                        flags.multiple_columns = false;
772                        flags.has_wildcard = false;
773                        for e in intermediate {
774                            replace_and_add_to_results(e, flags, result, schema, keys, opt_flags)?;
775                        }
776                    }
777                    // has only field expansion
778                    // col('a').struct.field('*')
779                    else {
780                        toggle_cse(opt_flags);
781                        expand_struct_fields(e, &expr, result, schema, names, &exclude)?
782                    }
783                },
784                _ => {},
785            }
786        }
787    }
788    // has multiple column names due to wildcards
789    else if flags.has_wildcard {
790        // keep track of column excluded from the wildcard
791        let exclude = prepare_excluded(&expr, schema, keys, flags.has_exclude)?;
792        // this path prepares the wildcard as input for the Function Expr
793        replace_wildcard(&expr, result, &exclude, schema)?;
794    }
795    // can have multiple column names due to a regex
796    else {
797        #[allow(clippy::collapsible_else_if)]
798        #[cfg(feature = "regex")]
799        {
800            // keep track of column excluded from the dtypes
801            let exclude = prepare_excluded(&expr, schema, keys, flags.has_exclude)?;
802            replace_regex(&expr, result, schema, &exclude)?;
803        }
804        #[cfg(not(feature = "regex"))]
805        {
806            let expr = rewrite_special_aliases(expr)?;
807            result.push(expr)
808        }
809    }
810    Ok(())
811}
812
813fn replace_selector_inner(
814    s: Selector,
815    members: &mut PlIndexSet<Expr>,
816    scratch: &mut Vec<Expr>,
817    schema: &Schema,
818    keys: &[Expr],
819) -> PolarsResult<()> {
820    match s {
821        Selector::Root(expr) => {
822            let local_flags = find_flags(&expr)?;
823            replace_and_add_to_results(
824                *expr,
825                local_flags,
826                scratch,
827                schema,
828                keys,
829                &mut Default::default(),
830            )?;
831            members.extend(scratch.drain(..))
832        },
833        Selector::Add(lhs, rhs) => {
834            let mut tmp_members: PlIndexSet<Expr> = Default::default();
835            replace_selector_inner(*lhs, members, scratch, schema, keys)?;
836            replace_selector_inner(*rhs, &mut tmp_members, scratch, schema, keys)?;
837            members.extend(tmp_members)
838        },
839        Selector::ExclusiveOr(lhs, rhs) => {
840            let mut tmp_members = Default::default();
841            replace_selector_inner(*lhs, &mut tmp_members, scratch, schema, keys)?;
842            replace_selector_inner(*rhs, members, scratch, schema, keys)?;
843
844            *members = tmp_members.symmetric_difference(members).cloned().collect();
845        },
846        Selector::Intersect(lhs, rhs) => {
847            let mut tmp_members = Default::default();
848            replace_selector_inner(*lhs, &mut tmp_members, scratch, schema, keys)?;
849            replace_selector_inner(*rhs, members, scratch, schema, keys)?;
850
851            *members = tmp_members.intersection(members).cloned().collect();
852        },
853        Selector::Sub(lhs, rhs) => {
854            let mut tmp_members = Default::default();
855            replace_selector_inner(*lhs, &mut tmp_members, scratch, schema, keys)?;
856            replace_selector_inner(*rhs, members, scratch, schema, keys)?;
857
858            *members = tmp_members.difference(members).cloned().collect();
859        },
860    }
861    Ok(())
862}
863
864fn replace_selector(expr: Expr, schema: &Schema, keys: &[Expr]) -> PolarsResult<Expr> {
865    // First pass we replace the selectors with Expr::Columns, we expand the `to_add` columns
866    // and then subtract the `to_subtract` columns.
867    expr.try_map_expr(|e| match e {
868        Expr::Selector(mut s) => {
869            let mut swapped = Selector::Root(Box::new(Expr::Wildcard));
870            std::mem::swap(&mut s, &mut swapped);
871
872            let cols = expand_selector(swapped, schema, keys)?;
873            Ok(Expr::Columns(cols))
874        },
875        e => Ok(e),
876    })
877}
878
879pub(crate) fn expand_selectors(
880    s: Vec<Selector>,
881    schema: &Schema,
882    keys: &[Expr],
883) -> PolarsResult<Arc<[PlSmallStr]>> {
884    let mut columns = vec![];
885
886    // Skip the column fast paths.
887    fn skip(name: &str) -> bool {
888        is_regex_projection(name) || name == "*"
889    }
890
891    for s in s {
892        match s {
893            Selector::Root(e) => match *e {
894                Expr::Column(name) if !skip(name.as_ref()) => columns.push(name),
895                Expr::Columns(names) if names.iter().all(|n| !skip(n.as_ref())) => {
896                    columns.extend_from_slice(names.as_ref())
897                },
898                Expr::Selector(s) => {
899                    let names = expand_selector(s, schema, keys)?;
900                    columns.extend_from_slice(names.as_ref());
901                },
902                e => {
903                    let names = expand_selector(Selector::new(e), schema, keys)?;
904                    columns.extend_from_slice(names.as_ref());
905                },
906            },
907            other => {
908                let names = expand_selector(other, schema, keys)?;
909                columns.extend_from_slice(names.as_ref());
910            },
911        }
912    }
913
914    Ok(Arc::from(columns))
915}
916
917pub(super) fn expand_selector(
918    s: Selector,
919    schema: &Schema,
920    keys: &[Expr],
921) -> PolarsResult<Arc<[PlSmallStr]>> {
922    let mut members = PlIndexSet::new();
923    replace_selector_inner(s, &mut members, &mut vec![], schema, keys)?;
924
925    if members.len() <= 1 {
926        members
927            .into_iter()
928            .map(|e| {
929                let Expr::Column(name) = e else {
930                    polars_bail!(InvalidOperation: "invalid selector expression: {}", e)
931                };
932                Ok(name)
933            })
934            .collect()
935    } else {
936        // Ensure that multiple columns returned from combined/nested selectors remain in schema order
937        let selected = schema
938            .iter_fields()
939            .map(|field| field.name().clone())
940            .filter(|field_name| members.contains(&Expr::Column(field_name.clone())))
941            .collect();
942
943        Ok(selected)
944    }
945}

15impl OptimizationRule for CountStar {
16    // Replace select count(*) from datasource with specialized map function.
17    fn optimize_plan(
18        &mut self,
19        lp_arena: &mut Arena<IR>,
20        expr_arena: &mut Arena<AExpr>,
21        node: Node,
22    ) -> PolarsResult<Option<IR>> {
23        Ok(
24            visit_logical_plan_for_scan_paths(node, lp_arena, expr_arena, false).map(
25                |count_star_expr| {
26                    // MapFunction needs a leaf node, hence we create a dummy placeholder node
27                    let placeholder = IR::DataFrameScan {
28                        df: Arc::new(Default::default()),
29                        schema: Arc::new(Default::default()),
30                        output_schema: None,
31                    };
32                    let placeholder_node = lp_arena.add(placeholder);
33
34                    let alp = IR::MapFunction {
35                        input: placeholder_node,
36                        function: FunctionIR::FastCount {
37                            sources: count_star_expr.sources,
38                            scan_type: count_star_expr.scan_type,
39                            alias: count_star_expr.alias,
40                        },
41                    };
42
43                    lp_arena.replace(count_star_expr.node, alp.clone());
44                    alp
45                },
46            ),
47        )
48    }
49}

51struct CountStarExpr {
52    // Top node of the projection to replace
53    node: Node,
54    // Paths to the input files
55    sources: ScanSources,
56    // File Type
57    scan_type: FileScan,
58    // Column Alias
59    alias: Option<PlSmallStr>,
60}

62// Visit the logical plan and return CountStarExpr with the expr information gathered
63// Return None if query is not a simple COUNT(*) FROM SOURCE
64fn visit_logical_plan_for_scan_paths(
65    node: Node,
66    lp_arena: &Arena<IR>,
67    expr_arena: &Arena<AExpr>,
68    inside_union: bool, // Inside union's we do not check for COUNT(*) expression
69) -> Option<CountStarExpr> {
70    match lp_arena.get(node) {
71        IR::Union { inputs, .. } => {
72            enum MutableSources {
73                Paths(Vec<PathBuf>),
74                Buffers(Vec<MemSlice>),
75            }
76
77            let mut scan_type: Option<FileScan> = None;
78            let mut sources = None;
79            for input in inputs {
80                match visit_logical_plan_for_scan_paths(*input, lp_arena, expr_arena, true) {
81                    Some(expr) => {
82                        match (expr.sources, &mut sources) {
83                            (
84                                ScanSources::Paths(paths),
85                                Some(MutableSources::Paths(ref mut mutable_paths)),
86                            ) => mutable_paths.extend_from_slice(&paths[..]),
87                            (ScanSources::Paths(paths), None) => {
88                                sources = Some(MutableSources::Paths(paths.to_vec()))
89                            },
90                            (
91                                ScanSources::Buffers(buffers),
92                                Some(MutableSources::Buffers(ref mut mutable_buffers)),
93                            ) => mutable_buffers.extend_from_slice(&buffers[..]),
94                            (ScanSources::Buffers(buffers), None) => {
95                                sources = Some(MutableSources::Buffers(buffers.to_vec()))
96                            },
97                            _ => return None,
98                        }
99
100                        match &scan_type {
101                            None => scan_type = Some(expr.scan_type),
102                            Some(scan_type) => {
103                                // All scans must be of the same type (e.g. csv / parquet)
104                                if std::mem::discriminant(scan_type)
105                                    != std::mem::discriminant(&expr.scan_type)
106                                {
107                                    return None;
108                                }
109                            },
110                        };
111                    },
112                    None => return None,
113                }
114            }
115            Some(CountStarExpr {
116                sources: match sources {
117                    Some(MutableSources::Paths(paths)) => ScanSources::Paths(paths.into()),
118                    Some(MutableSources::Buffers(buffers)) => ScanSources::Buffers(buffers.into()),
119                    None => ScanSources::default(),
120                },
121                scan_type: scan_type.unwrap(),
122                node,
123                alias: None,
124            })
125        },
126        IR::Scan {
127            scan_type, sources, ..
128        } if !matches!(scan_type, FileScan::Anonymous { .. }) => Some(CountStarExpr {
129            sources: sources.clone(),
130            scan_type: scan_type.clone(),
131            node,
132            alias: None,
133        }),
134        // A union can insert a simple projection to ensure all projections align.
135        // We can ignore that if we are inside a count star.
136        IR::SimpleProjection { input, .. } if inside_union => {
137            visit_logical_plan_for_scan_paths(*input, lp_arena, expr_arena, false)
138        },
139        IR::Select { input, expr, .. } => {
140            if expr.len() == 1 {
141                let (valid, alias) = is_valid_count_expr(&expr[0], expr_arena);
142                if valid || inside_union {
143                    return visit_logical_plan_for_scan_paths(*input, lp_arena, expr_arena, false)
144                        .map(|mut expr| {
145                            expr.alias = alias;
146                            expr.node = node;
147                            expr
148                        });
149                }
150            }
151            None
152        },
153        _ => None,
154    }
155}

157fn is_valid_count_expr(e: &ExprIR, expr_arena: &Arena<AExpr>) -> (bool, Option<PlSmallStr>) {
158    match expr_arena.get(e.node()) {
159        AExpr::Len => (true, e.get_alias().cloned()),
160        _ => (false, None),
161    }

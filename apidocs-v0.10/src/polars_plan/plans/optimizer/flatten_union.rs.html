1use polars_core::error::PolarsResult;
2use polars_utils::arena::{Arena, Node};
3use IR::*;
4
5use super::OptimizationRule;
6use crate::prelude::IR;
7
8pub struct FlattenUnionRule {}
9
10fn get_union_inputs(node: Node, lp_arena: &Arena<IR>) -> Option<&[Node]> {
11    match lp_arena.get(node) {
12        IR::Union { inputs, .. } => Some(inputs),
13        _ => None,
14    }
15}
16
17impl OptimizationRule for FlattenUnionRule {
18    fn optimize_plan(
19        &mut self,
20        lp_arena: &mut polars_utils::arena::Arena<IR>,
21        _expr_arena: &mut polars_utils::arena::Arena<crate::prelude::AExpr>,
22        node: polars_utils::arena::Node,
23    ) -> PolarsResult<Option<IR>> {
24        let lp = lp_arena.get(node);
25
26        match lp {
27            Union {
28                inputs,
29                mut options,
30            } if inputs.iter().any(|node| match lp_arena.get(*node) {
31                Union { options, .. } => !options.flattened_by_opt,
32                _ => false,
33            }) =>
34            {
35                let mut new_inputs = Vec::with_capacity(inputs.len() * 2);
36
37                for node in inputs {
38                    match get_union_inputs(*node, lp_arena) {
39                        Some(inp) => new_inputs.extend_from_slice(inp),
40                        None => new_inputs.push(*node),
41                    }
42                }
43                options.flattened_by_opt = true;
44
45                Ok(Some(Union {
46                    inputs: new_inputs,
47                    options,
48                }))
49            },
50            _ => Ok(None),
51        }
52    }
53}

1use arrow::legacy::error::{polars_bail, PolarsResult};
2use polars_core::prelude::Field;
3use polars_core::schema::Schema;
4use polars_utils::pl_str::PlSmallStr;
5
6use super::{ColumnsUdf, Expr, GetOutput, OpaqueColumnUdf};
7use crate::prelude::{new_column_udf, Context, FunctionOptions};

9/// Represents a user-defined function
10#[derive(Clone)]
11pub struct UserDefinedFunction {
12    /// name
13    pub name: PlSmallStr,
14    /// The function signature.
15    pub input_fields: Vec<Field>,
16    /// The function output type.
17    pub return_type: GetOutput,
18    /// The function implementation.
19    pub fun: OpaqueColumnUdf,
20    /// Options for the function.
21    pub options: FunctionOptions,
22}
23
24impl std::fmt::Debug for UserDefinedFunction {
25    fn fmt(&self, f: &mut std::fmt::Formatter) -> std::fmt::Result {
26        f.debug_struct("UserDefinedFunction")
27            .field("name", &self.name)
28            .field("signature", &self.input_fields)
29            .field("fun", &"<FUNC>")
30            .field("options", &self.options)
31            .finish()
32    }
33}
34
35impl UserDefinedFunction {
36    /// Create a new UserDefinedFunction
37    pub fn new(
38        name: PlSmallStr,
39        input_fields: Vec<Field>,
40        return_type: GetOutput,
41        fun: impl ColumnsUdf + 'static,
42    ) -> Self {
43        Self {
44            name,
45            input_fields,
46            return_type,
47            fun: new_column_udf(fun),
48            options: FunctionOptions::default(),
49        }
50    }
51
52    /// creates a logical expression with a call of the UDF
53    /// This utility allows using the UDF without requiring access to the registry.
54    /// The schema is validated and the query will fail if the schema is invalid.
55    pub fn call(self, args: Vec<Expr>) -> PolarsResult<Expr> {
56        if args.len() != self.input_fields.len() {
57            polars_bail!(InvalidOperation: "expected {} arguments, got {}", self.input_fields.len(), args.len())
58        }
59        let schema = Schema::from_iter(self.input_fields);
60
61        if args
62            .iter()
63            .map(|e| e.to_field(&schema, Context::Default))
64            .collect::<PolarsResult<Vec<_>>>()
65            .is_err()
66        {
67            polars_bail!(InvalidOperation: "unexpected field in UDF \nexpected: {:?}\n received {:?}", schema, args)
68        };
69
70        Ok(Expr::AnonymousFunction {
71            input: args,
72            function: self.fun,
73            output_type: self.return_type,
74            options: self.options,
75        })
76    }
77
78    /// creates a logical expression with a call of the UDF
79    /// This does not do any schema validation and is therefore faster.
80    ///
81    /// Only use this if you are certain that the schema is correct.
82    /// If the schema is invalid, the query will fail at runtime.
83    pub fn call_unchecked(self, args: Vec<Expr>) -> Expr {
84        Expr::AnonymousFunction {
85            input: args,
86            function: self.fun,
87            output_type: self.return_type.clone(),
88            options: self.options,
89        }
90    }
91}

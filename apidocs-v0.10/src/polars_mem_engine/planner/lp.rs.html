11fn partitionable_gb(
12    keys: &[ExprIR],
13    aggs: &[ExprIR],
14    input_schema: &Schema,
15    expr_arena: &Arena<AExpr>,
16    apply: &Option<Arc<dyn DataFrameUdf>>,
17) -> bool {
18    // checks:
19    //      1. complex expressions in the group_by itself are also not partitionable
20    //          in this case anything more than col("foo")
21    //      2. a custom function cannot be partitioned
22    //      3. we don't bother with more than 2 keys, as the cardinality likely explodes
23    //         by the combinations
24    if !keys.is_empty() && keys.len() < 3 && apply.is_none() {
25        // complex expressions in the group_by itself are also not partitionable
26        // in this case anything more than col("foo")
27        for key in keys {
28            if (expr_arena).iter(key.node()).count() > 1
29                || has_aexpr(key.node(), expr_arena, |ae| match ae {
30                    AExpr::Literal(lv) => !lv.is_scalar(),
31                    _ => false,
32                })
33            {
34                return false;
35            }
36        }
37
38        can_pre_agg_exprs(aggs, expr_arena, input_schema)
39    } else {
40        false
41    }
42}

56pub fn create_physical_plan(
57    root: Node,
58    lp_arena: &mut Arena<IR>,
59    expr_arena: &Arena<AExpr>,
60) -> PolarsResult<Box<dyn Executor>> {
61    let state = ConversionState::new()?;
62    create_physical_plan_impl(root, lp_arena, expr_arena, &state)
63}

65fn create_physical_plan_impl(
66    root: Node,
67    lp_arena: &mut Arena<IR>,
68    expr_arena: &Arena<AExpr>,
69    state: &ConversionState,
70) -> PolarsResult<Box<dyn Executor>> {
71    use IR::*;
72
73    let logical_plan = lp_arena.take(root);
74    match logical_plan {
75        #[cfg(feature = "python")]
76        PythonScan { mut options } => {
77            let mut predicate_serialized = None;
78
79            let predicate = if let PythonPredicate::Polars(e) = &options.predicate {
80                let phys_expr = || {
81                    let mut state = ExpressionConversionState::new(true, state.expr_depth);
82                    create_physical_expr(
83                        e,
84                        Context::Default,
85                        expr_arena,
86                        &options.schema,
87                        &mut state,
88                    )
89                };
90
91                // Convert to a pyarrow eval string.
92                if matches!(options.python_source, PythonScanSource::Pyarrow) {
93                    if let Some(eval_str) = polars_plan::plans::python::pyarrow::predicate_to_pa(
94                        e.node(),
95                        expr_arena,
96                        Default::default(),
97                    ) {
98                        options.predicate = PythonPredicate::PyArrow(eval_str);
99                        // We don't have to use a physical expression as pyarrow deals with the filter.
100                        None
101                    } else {
102                        Some(phys_expr()?)
103                    }
104                }
105                // Convert to physical expression for the case the reader cannot consume the predicate.
106                else {
107                    let dsl_expr = e.to_expr(expr_arena);
108                    predicate_serialized =
109                        polars_plan::plans::python::predicate::serialize(&dsl_expr)?;
110
111                    Some(phys_expr()?)
112                }
113            } else {
114                None
115            };
116            Ok(Box::new(executors::PythonScanExec {
117                options,
118                predicate,
119                predicate_serialized,
120            }))
121        },
122        Sink { payload, .. } => match payload {
123            SinkType::Memory => {
124                polars_bail!(InvalidOperation: "memory sink not supported in the standard engine")
125            },
126            SinkType::File { file_type, .. } => {
127                polars_bail!(InvalidOperation:
128                    "sink_{file_type:?} not yet supported in standard engine. Use 'collect().write_{file_type:?}()'"
129                )
130            },
131        },
132        Union { inputs, options } => {
133            let inputs = inputs
134                .into_iter()
135                .map(|node| create_physical_plan_impl(node, lp_arena, expr_arena, state))
136                .collect::<PolarsResult<Vec<_>>>()?;
137            Ok(Box::new(executors::UnionExec { inputs, options }))
138        },
139        HConcat {
140            inputs, options, ..
141        } => {
142            let inputs = inputs
143                .into_iter()
144                .map(|node| create_physical_plan_impl(node, lp_arena, expr_arena, state))
145                .collect::<PolarsResult<Vec<_>>>()?;
146            Ok(Box::new(executors::HConcatExec { inputs, options }))
147        },
148        Slice { input, offset, len } => {
149            let input = create_physical_plan_impl(input, lp_arena, expr_arena, state)?;
150            Ok(Box::new(executors::SliceExec { input, offset, len }))
151        },
152        Filter { input, predicate } => {
153            let mut streamable =
154                is_elementwise_rec_no_cat_cast(expr_arena.get(predicate.node()), expr_arena);
155            let input_schema = lp_arena.get(input).schema(lp_arena).into_owned();
156            if streamable {
157                // This can cause problems with string caches
158                streamable = !input_schema
159                    .iter_values()
160                    .any(|dt| dt.contains_categoricals())
161                    || {
162                        #[cfg(feature = "dtype-categorical")]
163                        {
164                            polars_core::using_string_cache()
165                        }
166
167                        #[cfg(not(feature = "dtype-categorical"))]
168                        {
169                            false
170                        }
171                    }
172            }
173            let input = create_physical_plan_impl(input, lp_arena, expr_arena, state)?;
174            let mut state = ExpressionConversionState::new(true, state.expr_depth);
175            let predicate = create_physical_expr(
176                &predicate,
177                Context::Default,
178                expr_arena,
179                &input_schema,
180                &mut state,
181            )?;
182            Ok(Box::new(executors::FilterExec::new(
183                predicate,
184                input,
185                state.has_windows,
186                streamable,
187            )))
188        },
189        #[allow(unused_variables)]
190        Scan {
191            sources,
192            file_info,
193            hive_parts,
194            output_schema,
195            scan_type,
196            predicate,
197            mut file_options,
198        } => {
199            file_options.slice = if let Some((offset, len)) = file_options.slice {
200                Some((offset, _set_n_rows_for_scan(Some(len)).unwrap()))
201            } else {
202                _set_n_rows_for_scan(None).map(|x| (0, x))
203            };
204
205            let mut state = ExpressionConversionState::new(true, state.expr_depth);
206            let predicate = predicate
207                .map(|pred| {
208                    create_physical_expr(
209                        &pred,
210                        Context::Default,
211                        expr_arena,
212                        output_schema.as_ref().unwrap_or(&file_info.schema),
213                        &mut state,
214                    )
215                })
216                .map_or(Ok(None), |v| v.map(Some))?;
217
218            if sources.len() > 1
219                && std::env::var("POLARS_NEW_MULTIFILE").as_deref() == Ok("1")
220                && !matches!(scan_type, FileScan::Anonymous { .. })
221            {
222                return Ok(Box::new(executors::MultiScanExec::new(
223                    sources,
224                    file_info,
225                    hive_parts,
226                    predicate,
227                    file_options,
228                    scan_type,
229                )));
230            }
231
232            match scan_type.clone() {
233                #[cfg(feature = "csv")]
234                FileScan::Csv { options, .. } => Ok(Box::new(executors::CsvExec {
235                    sources,
236                    file_info,
237                    options,
238                    predicate,
239                    file_options,
240                })),
241                #[cfg(feature = "ipc")]
242                FileScan::Ipc {
243                    options,
244                    cloud_options,
245                    metadata,
246                } => Ok(Box::new(executors::IpcExec {
247                    sources,
248                    file_info,
249                    predicate,
250                    options,
251                    file_options,
252                    hive_parts,
253                    cloud_options,
254                    metadata,
255                })),
256                #[cfg(feature = "parquet")]
257                FileScan::Parquet {
258                    options,
259                    cloud_options,
260                    metadata,
261                } => Ok(Box::new(executors::ParquetExec::new(
262                    sources,
263                    file_info,
264                    hive_parts,
265                    predicate,
266                    options,
267                    cloud_options,
268                    file_options,
269                    metadata,
270                ))),
271                #[cfg(feature = "json")]
272                FileScan::NDJson { options, .. } => Ok(Box::new(executors::JsonExec::new(
273                    sources,
274                    options,
275                    file_options,
276                    file_info,
277                    predicate,
278                ))),
279                FileScan::Anonymous { function, .. } => {
280                    Ok(Box::new(executors::AnonymousScanExec {
281                        function,
282                        predicate,
283                        file_options,
284                        file_info,
285                        output_schema,
286                        predicate_has_windows: state.has_windows,
287                    }))
288                },
289            }
290        },
291        Select {
292            expr,
293            input,
294            schema: _schema,
295            options,
296            ..
297        } => {
298            let input_schema = lp_arena.get(input).schema(lp_arena).into_owned();
299            let input = create_physical_plan_impl(input, lp_arena, expr_arena, state)?;
300            let mut state = ExpressionConversionState::new(
301                POOL.current_num_threads() > expr.len(),
302                state.expr_depth,
303            );
304            let phys_expr = create_physical_expressions_from_irs(
305                &expr,
306                Context::Default,
307                expr_arena,
308                &input_schema,
309                &mut state,
310            )?;
311
312            let allow_vertical_parallelism = options.should_broadcast && expr.iter().all(|e| is_elementwise_rec_no_cat_cast(expr_arena.get(e.node()), expr_arena))
313                // If all columns are literal we would get a 1 row per thread.
314                && !phys_expr.iter().all(|p| {
315                    p.is_literal()
316                });
317
318            Ok(Box::new(executors::ProjectionExec {
319                input,
320                expr: phys_expr,
321                has_windows: state.has_windows,
322                input_schema,
323                #[cfg(test)]
324                schema: _schema,
325                options,
326                allow_vertical_parallelism,
327            }))
328        },
329        DataFrameScan {
330            df, output_schema, ..
331        } => Ok(Box::new(executors::DataFrameExec {
332            df,
333            projection: output_schema.map(|s| s.iter_names_cloned().collect()),
334        })),
335        Sort {
336            input,
337            by_column,
338            slice,
339            sort_options,
340        } => {
341            let input_schema = lp_arena.get(input).schema(lp_arena);
342            let by_column = create_physical_expressions_from_irs(
343                &by_column,
344                Context::Default,
345                expr_arena,
346                input_schema.as_ref(),
347                &mut ExpressionConversionState::new(true, state.expr_depth),
348            )?;
349            let input = create_physical_plan_impl(input, lp_arena, expr_arena, state)?;
350            Ok(Box::new(executors::SortExec {
351                input,
352                by_column,
353                slice,
354                sort_options,
355            }))
356        },
357        Cache {
358            input,
359            id,
360            cache_hits,
361        } => {
362            let input = create_physical_plan_impl(input, lp_arena, expr_arena, state)?;
363            Ok(Box::new(executors::CacheExec {
364                id,
365                input,
366                count: cache_hits,
367            }))
368        },
369        Distinct { input, options } => {
370            let input = create_physical_plan_impl(input, lp_arena, expr_arena, state)?;
371            Ok(Box::new(executors::UniqueExec { input, options }))
372        },
373        GroupBy {
374            input,
375            keys,
376            aggs,
377            apply,
378            schema,
379            maintain_order,
380            options,
381        } => {
382            let input_schema = lp_arena.get(input).schema(lp_arena).into_owned();
383            let options = Arc::try_unwrap(options).unwrap_or_else(|options| (*options).clone());
384            let phys_keys = create_physical_expressions_from_irs(
385                &keys,
386                Context::Default,
387                expr_arena,
388                &input_schema,
389                &mut ExpressionConversionState::new(true, state.expr_depth),
390            )?;
391            let phys_aggs = create_physical_expressions_from_irs(
392                &aggs,
393                Context::Aggregation,
394                expr_arena,
395                &input_schema,
396                &mut ExpressionConversionState::new(true, state.expr_depth),
397            )?;
398
399            let _slice = options.slice;
400            #[cfg(feature = "dynamic_group_by")]
401            if let Some(options) = options.dynamic {
402                let input = create_physical_plan_impl(input, lp_arena, expr_arena, state)?;
403                return Ok(Box::new(executors::GroupByDynamicExec {
404                    input,
405                    keys: phys_keys,
406                    aggs: phys_aggs,
407                    options,
408                    input_schema,
409                    slice: _slice,
410                    apply,
411                }));
412            }
413
414            #[cfg(feature = "dynamic_group_by")]
415            if let Some(options) = options.rolling {
416                let input = create_physical_plan_impl(input, lp_arena, expr_arena, state)?;
417                return Ok(Box::new(executors::GroupByRollingExec {
418                    input,
419                    keys: phys_keys,
420                    aggs: phys_aggs,
421                    options,
422                    input_schema,
423                    slice: _slice,
424                    apply,
425                }));
426            }
427
428            // We first check if we can partition the group_by on the latest moment.
429            let partitionable = partitionable_gb(&keys, &aggs, &input_schema, expr_arena, &apply);
430            if partitionable {
431                let from_partitioned_ds = (&*lp_arena).iter(input).any(|(_, lp)| {
432                    if let Union { options, .. } = lp {
433                        options.from_partitioned_ds
434                    } else {
435                        false
436                    }
437                });
438                let input = create_physical_plan_impl(input, lp_arena, expr_arena, state)?;
439                let keys = keys
440                    .iter()
441                    .map(|e| e.to_expr(expr_arena))
442                    .collect::<Vec<_>>();
443                let aggs = aggs
444                    .iter()
445                    .map(|e| e.to_expr(expr_arena))
446                    .collect::<Vec<_>>();
447                Ok(Box::new(executors::PartitionGroupByExec::new(
448                    input,
449                    phys_keys,
450                    phys_aggs,
451                    maintain_order,
452                    options.slice,
453                    input_schema,
454                    schema,
455                    from_partitioned_ds,
456                    keys,
457                    aggs,
458                )))
459            } else {
460                let input = create_physical_plan_impl(input, lp_arena, expr_arena, state)?;
461                Ok(Box::new(executors::GroupByExec::new(
462                    input,
463                    phys_keys,
464                    phys_aggs,
465                    apply,
466                    maintain_order,
467                    input_schema,
468                    options.slice,
469                )))
470            }
471        },
472        Join {
473            input_left,
474            input_right,
475            left_on,
476            right_on,
477            options,
478            schema,
479            ..
480        } => {
481            let parallel = if options.force_parallel {
482                true
483            } else if options.allow_parallel {
484                // check if two DataFrames come from a separate source.
485                // If they don't we can parallelize,
486                // we may deadlock if we don't check this
487                let mut sources_left = PlHashSet::new();
488                agg_source_paths(input_left, &mut sources_left, lp_arena);
489                let mut sources_right = PlHashSet::new();
490                agg_source_paths(input_right, &mut sources_right, lp_arena);
491                sources_left.intersection(&sources_right).next().is_none()
492            } else {
493                false
494            };
495            let schema_left = lp_arena.get(input_left).schema(lp_arena).into_owned();
496            let schema_right = lp_arena.get(input_right).schema(lp_arena).into_owned();
497
498            let input_left = create_physical_plan_impl(input_left, lp_arena, expr_arena, state)?;
499            let input_right = create_physical_plan_impl(input_right, lp_arena, expr_arena, state)?;
500
501            let left_on = create_physical_expressions_from_irs(
502                &left_on,
503                Context::Default,
504                expr_arena,
505                &schema_left,
506                &mut ExpressionConversionState::new(true, state.expr_depth),
507            )?;
508            let right_on = create_physical_expressions_from_irs(
509                &right_on,
510                Context::Default,
511                expr_arena,
512                &schema_right,
513                &mut ExpressionConversionState::new(true, state.expr_depth),
514            )?;
515            let options = Arc::try_unwrap(options).unwrap_or_else(|options| (*options).clone());
516
517            // Convert the join options, to the physical join options. This requires the physical
518            // planner, so we do this last minute.
519            let join_type_options = options
520                .options
521                .map(|o| {
522                    o.compile(|e| {
523                        let phys_expr = create_physical_expr(
524                            e,
525                            Context::Default,
526                            expr_arena,
527                            &schema,
528                            &mut ExpressionConversionState::new(false, state.expr_depth),
529                        )?;
530
531                        let execution_state = ExecutionState::default();
532
533                        Ok(Arc::new(move |df: DataFrame| {
534                            let mask = phys_expr.evaluate(&df, &execution_state)?;
535                            let mask = mask.as_materialized_series();
536                            let mask = mask.bool()?;
537                            df._filter_seq(mask)
538                        }))
539                    })
540                })
541                .transpose()?;
542
543            Ok(Box::new(executors::JoinExec::new(
544                input_left,
545                input_right,
546                left_on,
547                right_on,
548                parallel,
549                options.args,
550                join_type_options,
551            )))
552        },
553        HStack {
554            input,
555            exprs,
556            schema: output_schema,
557            options,
558        } => {
559            let input_schema = lp_arena.get(input).schema(lp_arena).into_owned();
560            let input = create_physical_plan_impl(input, lp_arena, expr_arena, state)?;
561
562            let allow_vertical_parallelism = options.should_broadcast
563                && exprs
564                    .iter()
565                    .all(|e| is_elementwise_rec_no_cat_cast(expr_arena.get(e.node()), expr_arena));
566
567            let mut state = ExpressionConversionState::new(
568                POOL.current_num_threads() > exprs.len(),
569                state.expr_depth,
570            );
571
572            let phys_exprs = create_physical_expressions_from_irs(
573                &exprs,
574                Context::Default,
575                expr_arena,
576                &input_schema,
577                &mut state,
578            )?;
579            Ok(Box::new(executors::StackExec {
580                input,
581                has_windows: state.has_windows,
582                exprs: phys_exprs,
583                input_schema,
584                output_schema,
585                options,
586                allow_vertical_parallelism,
587            }))
588        },
589        MapFunction {
590            input, function, ..
591        } => {
592            let input = create_physical_plan_impl(input, lp_arena, expr_arena, state)?;
593            Ok(Box::new(executors::UdfExec { input, function }))
594        },
595        ExtContext {
596            input, contexts, ..
597        } => {
598            let input = create_physical_plan_impl(input, lp_arena, expr_arena, state)?;
599            let contexts = contexts
600                .into_iter()
601                .map(|node| create_physical_plan_impl(node, lp_arena, expr_arena, state))
602                .collect::<PolarsResult<_>>()?;
603            Ok(Box::new(executors::ExternalContext { input, contexts }))
604        },
605        SimpleProjection { input, columns } => {
606            let input = create_physical_plan_impl(input, lp_arena, expr_arena, state)?;
607            let exec = executors::ProjectionSimple { input, columns };
608            Ok(Box::new(exec))
609        },
610        Invalid => unreachable!(),
611    }

19impl CsvExec {
20    fn read_impl(&self) -> PolarsResult<DataFrame> {
21        let with_columns = self
22            .file_options
23            .with_columns
24            .clone()
25            // Interpret selecting no columns as selecting all columns.
26            .filter(|columns| !columns.is_empty());
27
28        let n_rows = _set_n_rows_for_scan(self.file_options.slice.map(|x| {
29            assert_eq!(x.0, 0);
30            x.1
31        }));
32        let predicate = self.predicate.clone().map(phys_expr_to_io_expr);
33        let options_base = self
34            .options
35            .clone()
36            .with_schema(Some(
37                self.file_info.reader_schema.clone().unwrap().unwrap_right(),
38            ))
39            .with_columns(with_columns)
40            .with_rechunk(
41                // We rechunk at the end to avoid rechunking multiple times in the
42                // case of reading multiple files.
43                false,
44            )
45            .with_row_index(None)
46            .with_path::<&str>(None);
47
48        if self.sources.is_empty() {
49            let out = if let Some(schema) = options_base.schema {
50                DataFrame::from_rows_and_schema(&[], schema.as_ref())?
51            } else {
52                Default::default()
53            };
54            return Ok(out);
55        }
56
57        let verbose = config::verbose();
58        let force_async = config::force_async();
59        let run_async = (self.sources.is_paths() && force_async) || self.sources.is_cloud_url();
60
61        if self.sources.is_paths() && force_async && verbose {
62            eprintln!("ASYNC READING FORCED");
63        }
64
65        let finish_read =
66            |i: usize, options: CsvReadOptions, predicate: Option<Arc<dyn PhysicalIoExpr>>| {
67                let source = self.sources.at(i);
68                let owned = &mut vec![];
69
70                let memslice = source.to_memslice_async_assume_latest(run_async)?;
71
72                let reader = std::io::Cursor::new(maybe_decompress_bytes(&memslice, owned)?);
73                let mut df = options
74                    .into_reader_with_file_handle(reader)
75                    ._with_predicate(predicate.clone())
76                    .finish()?;
77
78                if let Some(col) = &self.file_options.include_file_paths {
79                    let name = source.to_include_path_name();
80
81                    unsafe {
82                        df.with_column_unchecked(Column::new_scalar(
83                            col.clone(),
84                            Scalar::new(DataType::String, AnyValue::StringOwned(name.into())),
85                            df.height(),
86                        ))
87                    };
88                }
89
90                Ok(df)
91            };
92
93        let mut df = if n_rows.is_some()
94            || (predicate.is_some() && self.file_options.row_index.is_some())
95        {
96            // Basic sequential read
97            // predicate must be done after n_rows and row_index, so we must read sequentially
98            if verbose {
99                eprintln!("read per-file to apply n_rows or (predicate + row_index)");
100            }
101
102            let mut n_rows_read = 0usize;
103            let mut out = Vec::with_capacity(self.sources.len());
104            // If we have n_rows or row_index then we need to count how many rows we read, so we need
105            // to delay applying the predicate.
106            let predicate_during_read = predicate
107                .clone()
108                .filter(|_| n_rows.is_none() && self.file_options.row_index.is_none());
109
110            for i in 0..self.sources.len() {
111                let opts = options_base
112                    .clone()
113                    .with_row_index(self.file_options.row_index.clone().map(|mut ri| {
114                        ri.offset += n_rows_read as IdxSize;
115                        ri
116                    }))
117                    .with_n_rows(n_rows.map(|n| n - n_rows_read));
118
119                let mut df = finish_read(i, opts, predicate_during_read.clone())?;
120
121                n_rows_read = n_rows_read.saturating_add(df.height());
122
123                let df = if predicate.is_some() && predicate_during_read.is_none() {
124                    let predicate = predicate.clone().unwrap();
125
126                    // We should have a chunked df since we read with rechunk false,
127                    // so we parallelize over row-wise batches.
128                    // Safety: We can accumulate unchecked here as these DataFrames
129                    // all come from the same file.
130                    accumulate_dataframes_vertical_unchecked(
131                        POOL.install(|| {
132                            df.split_chunks()
133                                .collect::<Vec<_>>()
134                                .into_par_iter()
135                                .map(|df| {
136                                    let s = predicate.evaluate_io(&df)?;
137                                    let mask = s
138                                        .bool()
139                                        .expect("filter predicates was not of type boolean");
140                                    df.filter(mask)
141                                })
142                                .collect::<PolarsResult<Vec<_>>>()
143                        })?
144                        .into_iter(),
145                    )
146                } else {
147                    df
148                };
149
150                out.push(df);
151
152                if n_rows.is_some() && n_rows_read == n_rows.unwrap() {
153                    if verbose {
154                        eprintln!(
155                            "reached n_rows = {} at source {} / {}",
156                            n_rows.unwrap(),
157                            1 + i,
158                            self.sources.len()
159                        )
160                    }
161                    break;
162                }
163            }
164
165            accumulate_dataframes_vertical(out.into_iter())?
166        } else {
167            // Basic parallel read
168            assert!(
169                n_rows.is_none()
170                    && !(
171                        // We can do either but not both because we are doing them
172                        // out-of-order for parallel.
173                        predicate.is_some() && self.file_options.row_index.is_some()
174                    )
175            );
176            if verbose {
177                eprintln!("read files in parallel")
178            }
179
180            let dfs = POOL.install(|| {
181                let step = std::cmp::min(POOL.current_num_threads(), 128);
182
183                (0..self.sources.len())
184                    .step_by(step)
185                    .map(|start| {
186                        (start..std::cmp::min(start.saturating_add(step), self.sources.len()))
187                            .into_par_iter()
188                            .map(|i| finish_read(i, options_base.clone(), predicate.clone()))
189                            .collect::<PolarsResult<Vec<_>>>()
190                    })
191                    .collect::<PolarsResult<Vec<_>>>()
192            })?;
193
194            let mut df =
195                accumulate_dataframes_vertical(dfs.into_iter().flat_map(|dfs| dfs.into_iter()))?;
196
197            if let Some(row_index) = self.file_options.row_index.clone() {
198                df.with_row_index_mut(row_index.name.clone(), Some(row_index.offset));
199            }
200
201            df
202        };
203
204        if self.file_options.rechunk {
205            df.as_single_chunk_par();
206        };
207
208        Ok(df)
209    }
210}

212impl ScanExec for CsvExec {
213    fn read(
214        &mut self,
215        with_columns: Option<Arc<[PlSmallStr]>>,
216        slice: Option<(usize, usize)>,
217        predicate: Option<Arc<dyn PhysicalExpr>>,
218        row_index: Option<polars_io::RowIndex>,
219    ) -> PolarsResult<DataFrame> {
220        self.file_options.with_columns = with_columns;
221        self.file_options.slice = slice.map(|(o, l)| (o as i64, l));
222        self.predicate = predicate;
223        self.file_options.row_index = row_index;
224
225        if self.file_info.reader_schema.is_none() {
226            self.schema()?;
227        }
228        self.read_impl()
229    }
230
231    fn schema(&mut self) -> PolarsResult<&SchemaRef> {
232        let mut schema = self.file_info.reader_schema.take();
233        if schema.is_none() {
234            let force_async = config::force_async();
235            let run_async = (self.sources.is_paths() && force_async) || self.sources.is_cloud_url();
236
237            let source = self.sources.at(0);
238            let owned = &mut vec![];
239
240            let memslice = source.to_memslice_async_assume_latest(run_async)?;
241
242            // @TODO!: Cache the decompression
243            let bytes = maybe_decompress_bytes(&memslice, owned)?;
244
245            schema = Some(arrow::Either::Right(Arc::new(
246                infer_file_schema(
247                    &get_reader_bytes(&mut std::io::Cursor::new(bytes))?,
248                    self.options.parse_options.as_ref(),
249                    self.options.infer_schema_length,
250                    self.options.has_header,
251                    self.options.schema_overwrite.as_deref(),
252                    self.options.skip_rows,
253                    self.options.skip_lines,
254                    self.options.skip_rows_after_header,
255                    self.options.raise_if_empty,
256                    &mut self.options.n_threads,
257                )?
258                .0,
259            )));
260        }
261
262        Ok(self
263            .file_info
264            .reader_schema
265            .insert(schema.unwrap())
266            .as_ref()
267            .unwrap_right())
268    }
269
270    fn num_unfiltered_rows(&mut self) -> PolarsResult<IdxSize> {
271        let (lb, ub) = self.file_info.row_estimation;
272        if lb.is_none_or(|lb| lb != ub) {
273            return Ok(ub as IdxSize);
274        }
275
276        let force_async = config::force_async();
277        let run_async = (self.sources.is_paths() && force_async) || self.sources.is_cloud_url();
278
279        let source = self.sources.at(0);
280        let owned = &mut vec![];
281
282        // @TODO!: Cache the decompression
283        let memslice = source.to_memslice_async_assume_latest(run_async)?;
284
285        let popt = self.options.parse_options.as_ref();
286
287        let bytes = maybe_decompress_bytes(&memslice, owned)?;
288
289        let num_rows = count_rows_from_slice(
290            bytes,
291            popt.separator,
292            popt.quote_char,
293            popt.comment_prefix.as_ref(),
294            popt.eol_char,
295            self.options.has_header,
296        )?;
297
298        self.file_info.row_estimation = (Some(num_rows), num_rows);
299        Ok(num_rows as IdxSize)
300    }
301}

303impl Executor for CsvExec {
304    fn execute(&mut self, state: &mut ExecutionState) -> PolarsResult<DataFrame> {
305        let profile_name = if state.has_node_timer() {
306            let mut ids = vec![self.sources.id()];
307            if self.predicate.is_some() {
308                ids.push("predicate".into())
309            }
310            let name = comma_delimited("csv".to_string(), &ids);
311            Cow::Owned(name)
312        } else {
313            Cow::Borrowed("")
314        };
315
316        state.record(|| self.read_impl(), profile_name)
317    }

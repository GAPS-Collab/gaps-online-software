38impl Executor for JoinExec {
39    fn execute<'a>(&'a mut self, state: &'a mut ExecutionState) -> PolarsResult<DataFrame> {
40        state.should_stop()?;
41        #[cfg(debug_assertions)]
42        {
43            if state.verbose() {
44                eprintln!("run JoinExec")
45            }
46        }
47        if state.verbose() {
48            eprintln!("join parallel: {}", self.parallel);
49        };
50        let mut input_left = self.input_left.take().unwrap();
51        let mut input_right = self.input_right.take().unwrap();
52
53        let (df_left, df_right) = if self.parallel {
54            let mut state_right = state.split();
55            let mut state_left = state.split();
56            state_right.branch_idx += 1;
57            // propagate the fetch_rows static value to the spawning threads.
58            let fetch_rows = FETCH_ROWS.with(|fetch_rows| fetch_rows.get());
59
60            POOL.join(
61                move || {
62                    FETCH_ROWS.with(|fr| fr.set(fetch_rows));
63                    input_left.execute(&mut state_left)
64                },
65                move || {
66                    FETCH_ROWS.with(|fr| fr.set(fetch_rows));
67                    input_right.execute(&mut state_right)
68                },
69            )
70        } else {
71            (input_left.execute(state), input_right.execute(state))
72        };
73
74        let df_left = df_left?;
75        let df_right = df_right?;
76
77        let profile_name = if state.has_node_timer() {
78            let by = self
79                .left_on
80                .iter()
81                .map(|s| Ok(s.to_field(df_left.schema())?.name))
82                .collect::<PolarsResult<Vec<_>>>()?;
83            let name = comma_delimited("join".to_string(), &by);
84            Cow::Owned(name)
85        } else {
86            Cow::Borrowed("")
87        };
88
89        state.record(|| {
90
91            let left_on_series = self
92                .left_on
93                .iter()
94                .map(|e| e.evaluate(&df_left, state))
95                .collect::<PolarsResult<Vec<_>>>()?;
96
97            let right_on_series = self
98                .right_on
99                .iter()
100                .map(|e| e.evaluate(&df_right, state))
101                .collect::<PolarsResult<Vec<_>>>()?;
102
103            // prepare the tolerance
104            // we must ensure that we use the right units
105            #[cfg(feature = "asof_join")]
106            {
107                if let JoinType::AsOf(options) = &mut self.args.how {
108                    use polars_core::utils::arrow::temporal_conversions::MILLISECONDS_IN_DAY;
109                    if let Some(tol) = &options.tolerance_str {
110                        let duration = polars_time::Duration::try_parse(tol)?;
111                        polars_ensure!(
112                            duration.months() == 0,
113                            ComputeError: "cannot use month offset in timedelta of an asof join; \
114                            consider using 4 weeks"
115                        );
116                        let left_asof = df_left.column(left_on_series[0].name())?;
117                        use DataType::*;
118                        match left_asof.dtype() {
119                            Datetime(tu, _) | Duration(tu) => {
120                                let tolerance = match tu {
121                                    TimeUnit::Nanoseconds => duration.duration_ns(),
122                                    TimeUnit::Microseconds => duration.duration_us(),
123                                    TimeUnit::Milliseconds => duration.duration_ms(),
124                                };
125                                options.tolerance = Some(AnyValue::from(tolerance))
126                            }
127                            Date => {
128                                let days = (duration.duration_ms() / MILLISECONDS_IN_DAY) as i32;
129                                options.tolerance = Some(AnyValue::from(days))
130                            }
131                            Time => {
132                                let tolerance = duration.duration_ns();
133                                options.tolerance = Some(AnyValue::from(tolerance))
134                            }
135                            _ => {
136                                panic!("can only use timedelta string language with Date/Datetime/Duration/Time dtypes")
137                            }
138                        }
139                    }
140                }
141            }
142
143            let df = df_left._join_impl(
144                &df_right,
145                left_on_series.into_iter().map(|c| c.take_materialized_series()).collect(),
146                right_on_series.into_iter().map(|c| c.take_materialized_series()).collect(),
147                self.args.clone(),
148                self.options.clone(),
149                true,
150                state.verbose(),
151            );
152
153            if state.verbose() {
154                eprintln!("{:?} join dataframes finished", self.args.how);
155            };
156            df
157
158        }, profile_name)
159    }

64fn run_partitions(
65    df: &mut DataFrame,
66    exec: &PartitionGroupByExec,
67    state: &ExecutionState,
68    n_threads: usize,
69    maintain_order: bool,
70) -> PolarsResult<(Vec<DataFrame>, Vec<Vec<Column>>)> {
71    // We do a partitioned group_by.
72    // Meaning that we first do the group_by operation arbitrarily
73    // split on several threads. Than the final result we apply the same group_by again.
74    let dfs = split_df(df, n_threads, true);
75
76    let phys_aggs = &exec.phys_aggs;
77    let keys = &exec.phys_keys;
78
79    let mut keys = DataFrame::from_iter(compute_keys(keys, df, state)?);
80    let splitted_keys = split_df(&mut keys, n_threads, true);
81
82    POOL.install(|| {
83        dfs.into_par_iter()
84            .zip(splitted_keys)
85            .map(|(df, keys)| {
86                let gb = df.group_by_with_series(keys.into(), false, maintain_order)?;
87                let groups = gb.get_groups();
88
89                let mut columns = gb.keys();
90                // don't naively call par_iter here, it will segfault in rayon
91                // if you do, throw it on the POOL threadpool.
92                let agg_columns = phys_aggs
93                    .iter()
94                    .map(|expr| {
95                        let agg_expr = expr.as_partitioned_aggregator().unwrap();
96                        let agg = agg_expr.evaluate_partitioned(&df, groups, state)?;
97                        Ok(if agg.len() != groups.len() {
98                            polars_ensure!(agg.len() == 1, agg_len = agg.len(), groups.len());
99                            match groups.len() {
100                                0 => agg.clear(),
101                                len => agg.new_from_index(0, len),
102                            }
103                        } else {
104                            agg
105                        }
106                        .into_column())
107                    })
108                    .collect::<PolarsResult<Vec<_>>>()?;
109
110                columns.extend_from_slice(&agg_columns);
111
112                let df = DataFrame::new(columns)?;
113                Ok((df, gb.keys()))
114            })
115            .collect()
116    })
117}

119fn estimate_unique_count(keys: &[Column], mut sample_size: usize) -> PolarsResult<usize> {
120    // https://stats.stackexchange.com/a/19090/147321
121    // estimated unique size
122    // u + ui / m (s - m)
123    // s: set_size
124    // m: sample_size
125    // u: total unique groups counted in sample
126    // ui: groups with single unique value counted in sample
127    let set_size = keys[0].len();
128    if set_size < sample_size {
129        sample_size = set_size;
130    }
131
132    let finish = |groups: &GroupsType| {
133        let u = groups.len() as f64;
134        let ui = if groups.len() == sample_size {
135            u
136        } else {
137            groups.iter().filter(|g| g.len() == 1).count() as f64
138        };
139
140        (u + (ui / sample_size as f64) * (set_size - sample_size) as f64) as usize
141    };
142
143    if keys.len() == 1 {
144        // we sample as that will work also with sorted data.
145        // not that sampling without replacement is *very* expensive. don't do that.
146        let s = keys[0].sample_n(sample_size, true, false, None).unwrap();
147        // fast multi-threaded way to get unique.
148        let groups = s.as_materialized_series().group_tuples(true, false)?;
149        Ok(finish(&groups))
150    } else {
151        let offset = (keys[0].len() / 2) as i64;
152        let keys = keys
153            .iter()
154            .map(|s| s.slice(offset, sample_size))
155            .collect::<Vec<_>>();
156        let df = unsafe { DataFrame::new_no_checks_height_from_first(keys) };
157        let names = df.get_column_names().into_iter().cloned();
158        let gb = df.group_by(names).unwrap();
159        Ok(finish(gb.get_groups()))
160    }
161}

166#[cfg(not(debug_assertions))]
167const PARTITION_LIMIT: usize = 1000;

169// Checks if we should run normal or default aggregation
170// by sampling data.
171fn can_run_partitioned(
172    keys: &[Column],
173    original_df: &DataFrame,
174    state: &ExecutionState,
175    from_partitioned_ds: bool,
176) -> PolarsResult<bool> {
177    if !keys
178        .iter()
179        .take(1)
180        .all(|s| matches!(s.is_sorted_flag(), IsSorted::Not))
181    {
182        if state.verbose() {
183            eprintln!("FOUND SORTED KEY: running default HASH AGGREGATION")
184        }
185        Ok(false)
186    } else if std::env::var("POLARS_NO_PARTITION").is_ok() {
187        if state.verbose() {
188            eprintln!("POLARS_NO_PARTITION set: running default HASH AGGREGATION")
189        }
190        Ok(false)
191    } else if std::env::var("POLARS_FORCE_PARTITION").is_ok() {
192        if state.verbose() {
193            eprintln!("POLARS_FORCE_PARTITION set: running partitioned HASH AGGREGATION")
194        }
195        Ok(true)
196    } else if original_df.height() < PARTITION_LIMIT && !cfg!(test) {
197        if state.verbose() {
198            eprintln!("DATAFRAME < {PARTITION_LIMIT} rows: running default HASH AGGREGATION")
199        }
200        Ok(false)
201    } else {
202        // below this boundary we assume the partitioned group_by will be faster
203        let unique_count_boundary = std::env::var("POLARS_PARTITION_UNIQUE_COUNT")
204            .map(|s| s.parse::<usize>().unwrap())
205            .unwrap_or(1000);
206
207        let (unique_estimate, sampled_method) = match (keys.len(), keys[0].dtype()) {
208            #[cfg(feature = "dtype-categorical")]
209            (1, DataType::Categorical(Some(rev_map), _) | DataType::Enum(Some(rev_map), _)) => {
210                (rev_map.len(), "known")
211            },
212            _ => {
213                // sqrt(N) is a good sample size as it remains low on large numbers
214                // it is better than taking a fraction as it saturates
215                let sample_size = (original_df.height() as f64).powf(0.5) as usize;
216
217                // we never sample less than 100 data points.
218                let sample_size = std::cmp::max(100, sample_size);
219                (estimate_unique_count(keys, sample_size)?, "estimated")
220            },
221        };
222        if state.verbose() {
223            eprintln!("{sampled_method} unique values: {unique_estimate}");
224        }
225
226        if from_partitioned_ds {
227            let estimated_cardinality = unique_estimate as f32 / original_df.height() as f32;
228            if estimated_cardinality < 0.4 {
229                if state.verbose() {
230                    eprintln!("PARTITIONED DS");
231                }
232                Ok(true)
233            } else {
234                if state.verbose() {
235                    eprintln!("PARTITIONED DS: estimated cardinality: {estimated_cardinality} exceeded the boundary: 0.4, running default HASH AGGREGATION");
236                }
237                Ok(false)
238            }
239        } else if unique_estimate > unique_count_boundary {
240            if state.verbose() {
241                eprintln!("estimated unique count: {unique_estimate} exceeded the boundary: {unique_count_boundary}, running default HASH AGGREGATION")
242            }
243            Ok(false)
244        } else {
245            Ok(true)
246        }
247    }
248}

250impl PartitionGroupByExec {
251    fn execute_impl(
252        &mut self,
253        state: &mut ExecutionState,
254        mut original_df: DataFrame,
255    ) -> PolarsResult<DataFrame> {
256        let (splitted_dfs, splitted_keys) = {
257            // already get the keys. This is the very last minute decision which group_by method we choose.
258            // If the column is a categorical, we know the number of groups we have and can decide to continue
259            // partitioned or go for the standard group_by. The partitioned is likely to be faster on a small number
260            // of groups.
261            let keys = self.keys(&original_df, state)?;
262
263            if !can_run_partitioned(&keys, &original_df, state, self.from_partitioned_ds)? {
264                return group_by_helper(
265                    original_df,
266                    keys,
267                    &self.phys_aggs,
268                    None,
269                    state,
270                    self.maintain_order,
271                    self.slice,
272                );
273            }
274
275            if state.verbose() {
276                eprintln!("run PARTITIONED HASH AGGREGATION")
277            }
278
279            // Run the partitioned aggregations
280            let n_threads = POOL.current_num_threads();
281
282            run_partitions(
283                &mut original_df,
284                self,
285                state,
286                n_threads,
287                self.maintain_order,
288            )?
289        };
290
291        // MERGE phase
292
293        let df = accumulate_dataframes_vertical(splitted_dfs)?;
294        let keys = splitted_keys
295            .into_iter()
296            .reduce(|mut acc, e| {
297                acc.iter_mut().zip(e).for_each(|(acc, e)| {
298                    let _ = acc.append(&e);
299                });
300                acc
301            })
302            .unwrap();
303
304        // the partitioned group_by has added columns so we must update the schema.
305        state.set_schema(self.output_schema.clone());
306
307        // merge and hash aggregate again
308
309        // first get mutable access and optionally sort
310        let gb = df.group_by_with_series(keys, true, self.maintain_order)?;
311        let mut groups = gb.get_groups();
312
313        #[allow(unused_assignments)]
314        // it is unused because we only use it to keep the lifetime of sliced_group valid
315        let mut sliced_groups = None;
316
317        if let Some((offset, len)) = self.slice {
318            sliced_groups = Some(groups.slice(offset, len));
319            groups = sliced_groups.as_ref().unwrap();
320        }
321
322        let get_columns = || gb.keys_sliced(self.slice);
323        let get_agg = || {
324            let out: PolarsResult<Vec<_>> = self
325                .phys_aggs
326                .par_iter()
327                // we slice the keys off and finalize every aggregation
328                .zip(&df.get_columns()[self.phys_keys.len()..])
329                .map(|(expr, partitioned_s)| {
330                    let agg_expr = expr.as_partitioned_aggregator().unwrap();
331                    agg_expr.finalize(partitioned_s.clone(), groups, state)
332                })
333                .collect();
334
335            out
336        };
337        let (mut columns, agg_columns): (Vec<_>, _) = POOL.join(get_columns, get_agg);
338
339        columns.extend(agg_columns?);
340        state.clear_schema_cache();
341
342        Ok(DataFrame::new(columns).unwrap())
343    }
344}

346impl Executor for PartitionGroupByExec {
347    fn execute(&mut self, state: &mut ExecutionState) -> PolarsResult<DataFrame> {
348        state.should_stop()?;
349        #[cfg(debug_assertions)]
350        {
351            if state.verbose() {
352                eprintln!("run PartitionGroupbyExec")
353            }
354        }
355        let original_df = self.input.execute(state)?;
356
357        let profile_name = if state.has_node_timer() {
358            let by = self
359                .phys_keys
360                .iter()
361                .map(|s| Ok(s.to_field(&self.input_schema)?.name))
362                .collect::<PolarsResult<Vec<_>>>()?;
363            let name = comma_delimited("group_by_partitioned".to_string(), &by);
364            Cow::Owned(name)
365        } else {
366            Cow::Borrowed("")
367        };
368        if state.has_node_timer() {
369            let new_state = state.clone();
370            new_state.record(|| self.execute_impl(state, original_df), profile_name)
371        } else {
372            self.execute_impl(state, original_df)
373        }
374    }

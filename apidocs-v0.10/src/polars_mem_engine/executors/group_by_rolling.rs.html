3#[cfg_attr(not(feature = "dynamic_group_by"), allow(dead_code))]
4pub(crate) struct GroupByRollingExec {
5    pub(crate) input: Box<dyn Executor>,
6    pub(crate) keys: Vec<Arc<dyn PhysicalExpr>>,
7    pub(crate) aggs: Vec<Arc<dyn PhysicalExpr>>,
8    #[cfg(feature = "dynamic_group_by")]
9    pub(crate) options: RollingGroupOptions,
10    pub(crate) input_schema: SchemaRef,
11    pub(crate) slice: Option<(i64, usize)>,
12    pub(crate) apply: Option<Arc<dyn DataFrameUdf>>,
13}

15#[cfg(feature = "dynamic_group_by")]
16unsafe fn update_keys(keys: &mut [Column], groups: &GroupsType) {
17    match groups {
18        GroupsType::Idx(groups) => {
19            let first = groups.first();
20            // we don't use agg_first here, because the group
21            // can be empty, but we still want to know the first value
22            // of that group
23            for key in keys.iter_mut() {
24                *key = key.take_slice_unchecked(first);
25            }
26        },
27        GroupsType::Slice { groups, .. } => {
28            for key in keys.iter_mut() {
29                let indices = groups
30                    .iter()
31                    .map(|[first, _len]| *first)
32                    .collect_ca(PlSmallStr::EMPTY);
33                *key = key.take_unchecked(&indices);
34            }
35        },
36    }
37}

39impl GroupByRollingExec {
40    #[cfg(feature = "dynamic_group_by")]
41    fn execute_impl(
42        &mut self,
43        state: &ExecutionState,
44        mut df: DataFrame,
45    ) -> PolarsResult<DataFrame> {
46        df.as_single_chunk_par();
47
48        let keys = self
49            .keys
50            .iter()
51            .map(|e| e.evaluate(&df, state))
52            .collect::<PolarsResult<Vec<_>>>()?;
53
54        let (mut time_key, mut keys, groups) = df.rolling(keys, &self.options)?;
55
56        if let Some(f) = &self.apply {
57            let gb = GroupBy::new(&df, vec![], groups, None);
58            let out = gb.apply(move |df| f.call_udf(df))?;
59            return Ok(if let Some((offset, len)) = self.slice {
60                out.slice(offset, len)
61            } else {
62                out
63            });
64        }
65
66        let mut groups = &groups;
67        #[allow(unused_assignments)]
68        // it is unused because we only use it to keep the lifetime of sliced_group valid
69        let mut sliced_groups = None;
70
71        if let Some((offset, len)) = self.slice {
72            sliced_groups = Some(groups.slice(offset, len));
73            groups = sliced_groups.as_ref().unwrap();
74
75            time_key = time_key.slice(offset, len);
76        }
77
78        // the ordering has changed due to the group_by
79        if !keys.is_empty() {
80            unsafe { update_keys(&mut keys, groups) }
81        };
82
83        let agg_columns = evaluate_aggs(&df, &self.aggs, groups, state)?;
84
85        let mut columns = Vec::with_capacity(agg_columns.len() + 1 + keys.len());
86        columns.extend_from_slice(&keys);
87        columns.push(time_key);
88        columns.extend(agg_columns);
89
90        DataFrame::new(columns)
91    }
92}

94impl Executor for GroupByRollingExec {
95    #[cfg(not(feature = "dynamic_group_by"))]
96    fn execute(&mut self, _state: &mut ExecutionState) -> PolarsResult<DataFrame> {
97        panic!("activate feature dynamic_group_by")
98    }
99
100    #[cfg(feature = "dynamic_group_by")]
101    fn execute(&mut self, state: &mut ExecutionState) -> PolarsResult<DataFrame> {
102        state.should_stop()?;
103        #[cfg(debug_assertions)]
104        {
105            if state.verbose() {
106                eprintln!("run GroupbyRollingExec")
107            }
108        }
109        let df = self.input.execute(state)?;
110        let profile_name = if state.has_node_timer() {
111            let by = self
112                .keys
113                .iter()
114                .map(|s| Ok(s.to_field(&self.input_schema)?.name))
115                .collect::<PolarsResult<Vec<_>>>()?;
116            let name = comma_delimited("group_by_rolling".to_string(), &by);
117            Cow::Owned(name)
118        } else {
119            Cow::Borrowed("")
120        };
121
122        if state.has_node_timer() {
123            let new_state = state.clone();
124            new_state.record(|| self.execute_impl(state, df), profile_name)
125        } else {
126            self.execute_impl(state, df)
127        }
128    }

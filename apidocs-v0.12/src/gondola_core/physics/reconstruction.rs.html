69/// Settings for the hit cleaning of LineFit::fit_cleaned
70///
71/// A hit is accepted if its distance from the line is at
72/// most the radius of a cylinder around the line.
73///
74/// The radius of the cylinder starts at cylinder_radius. If
75/// final_radius is smaller, it shrinks after each fit by
76/// shrink_factor, down to final_radius, so that a wide
77/// cylinder collects the hits around a poor first line and a
78/// narrow one removes the hits which do not belong to the
79/// track once the line is better known.
80#[derive(Debug, Copy, Clone, PartialEq)]
81#[cfg_attr(feature="pybindings", pyclass)]
82pub struct LineFitCleaning {
83  /// Only hits within a cylinder of this radius (in mm)
84  /// around the line are used in the fit
85  pub cylinder_radius : f32,
86  /// Maximum number of fits. The iteration stops earlier
87  /// once the final radius is reached and the set of
88  /// accepted hits does not change anymore
89  pub max_iterations  : u32,
90  /// The cost function for each fit
91  pub cost            : LineFitCost,
92  /// The radius (in mm) the cylinder shrinks to. If it is
93  /// not smaller than cylinder_radius, the radius stays
94  /// the same (the default)
95  pub final_radius    : f32,
96  /// After each fit, the radius is multiplied by this
97  /// factor (but not below final_radius). A factor outside
98  /// of (0, 1) goes to final_radius right away
99  pub shrink_factor   : f32,
100}
101
102impl LineFitCleaning {
103  pub fn new(cylinder_radius : f32) -> Self {
104    Self {
105      cylinder_radius,
106      max_iterations : 10,
107      cost           : LineFitCost::Chi2,
108      final_radius   : cylinder_radius,
109      shrink_factor  : 0.5,
110    }
111  }
112
113  /// The radius of the cylinder after the next fit
114  fn next_radius(&self, radius : f32) -> f32 {
115    let target = self.final_radius.min(self.cylinder_radius);
116    if self.shrink_factor > 0.0 && self.shrink_factor < 1.0 {
117      (radius*self.shrink_factor).max(target)
118    } else {
119      target
120    }
121  }
122}

148/// A straight line through a pointcloud, parametrized
149/// in z, since that is our best constrained value
150///
151/// x(z) = x0 + tx * (z - z0)
152/// y(z) = y0 + ty * (z - z0)
153///
154/// The anchor z0 is fixed to the mean z of the hits, so
155/// the line has 4 free parameters (x0, y0, tx, ty). The
156/// 6 parameter model of line3d is degenerate (the anchor
157/// can slide along the line and the direction vector can
158/// be scaled), which only introduces flat directions
159/// for a minimizer.
160///
161/// For the standard chi2 (sum of squared, normalized
162/// residuals) the fit decouples into two independent
163/// weighted linear regressions x(z) and y(z), which
164/// can be solved analytically (LineFit::fit).
165///
166/// The python version minimizes the sum of the normalized
167/// distances instead (sum of sqrt(rx^2/sx^2 + ry^2/sy^2)),
168/// which is less sensitive to outliers, but has no closed
169/// form solution. This is available through
170/// LineFit::fit_sum_of_distances, which uses the analytic
171/// solution as a seed for a Nelder-Mead minimization.
172#[derive(Debug, Copy, Clone, PartialEq)]
173#[cfg_attr(feature="pybindings", pyclass)]
174pub struct LineFit {
175  /// anchor point
176  pub x0       : f32,
177  pub y0       : f32,
178  pub z0       : f32,
179  /// slopes dx/dz and dy/dz
180  pub tx       : f32,
181  pub ty       : f32,
182  /// parameter errors (from the analytic chi2 fit)
183  pub x0_err   : f32,
184  pub y0_err   : f32,
185  pub tx_err   : f32,
186  pub ty_err   : f32,
187  /// sum of the squared, normalized residuals
188  pub chi2     : f32,
189  /// value of the minimized cost function. For the
190  /// analytic fit this is the same as chi2
191  pub cost     : f32,
192  /// 2*npoints - 4
193  pub ndof     : u32,
194  pub npoints  : u32,
195  pub status   : FitStatus,
196}

218  /// Analytic, weighted least squares fit of a straight line
219  /// to the given points.
220  ///
221  /// Returns None if there are less than 2 points, or all
222  /// points have the same z (the line can not be described
223  /// as a function of z)
224  ///
225  /// # Arguments:
226  ///   * errs_x, errs_y : errors on x and y for each point.
227  ///                      If None, LINEFIT_DEFAULT_ERR_XY
228  ///                      will be used
229  pub fn fit(xs     : &[f32],
230             ys     : &[f32],
231             zs     : &[f32],
232             errs_x : Option<&[f32]>,
233             errs_y : Option<&[f32]>) -> Option<Self> {
234    let points = LineFitPoints::new(xs, ys, zs, errs_x, errs_y)?;
235    let (x0, tx, x0_var, tx_var) = points.regression(&points.xs, &points.wx)?;
236    let (y0, ty, y0_var, ty_var) = points.regression(&points.ys, &points.wy)?;
237    let mut lf  = Self::new();
238    lf.z0       = points.z0 as f32;
239    lf.x0       = x0 as f32;
240    lf.y0       = y0 as f32;
241    lf.tx       = tx as f32;
242    lf.ty       = ty as f32;
243    lf.x0_err   = x0_var.sqrt() as f32;
244    lf.y0_err   = y0_var.sqrt() as f32;
245    lf.tx_err   = tx_var.sqrt() as f32;
246    lf.ty_err   = ty_var.sqrt() as f32;
247    lf.chi2     = points.chi2(&[x0, y0, tx, ty]) as f32;
248    lf.cost     = lf.chi2;
249    lf.npoints  = xs.len() as u32;
250    lf.ndof     = 2*lf.npoints - 4;
251    lf.status   = FitStatus::Success;
252    Some(lf)
253  }

255  /// Fit a straight line to the given points by minimizing
256  /// the sum of the normalized distances, as done in the
257  /// python version of the linefit.
258  ///
259  /// The analytic chi2 fit is used as a seed for the
260  /// minimization with the Nelder-Mead algorithm. The parameter
261  /// errors are the ones of the seed.
262  pub fn fit_sum_of_distances(xs     : &[f32],
263                              ys     : &[f32],
264                              zs     : &[f32],
265                              errs_x : Option<&[f32]>,
266                              errs_y : Option<&[f32]>) -> Option<Self> {
267    let mut lf  = Self::fit(xs, ys, zs, errs_x, errs_y)?;
268    let points  = LineFitPoints::new(xs, ys, zs, errs_x, errs_y)?;
269    let seed    = vec![lf.x0 as f64, lf.y0 as f64, lf.tx as f64, lf.ty as f64];
270    // initial simplex - step sizes from the seed errors, so
271    // that they come with the right scale for each parameter
272    let steps   = [lf.x0_err as f64, lf.y0_err as f64, lf.tx_err as f64, lf.ty_err as f64];
273    let mut simplex = vec![seed.clone()];
274    for k in 0..seed.len() {
275      let mut vertex = seed.clone();
276      vertex[k] += if steps[k] > 0.0 { steps[k] } else { 1e-3 };
277      simplex.push(vertex);
278    }
279    let solver = match NelderMead::new(simplex).with_sd_tolerance(1e-8) {
280      Err(err) => {
281        error!("Unable to set up Nelder-Mead solver! {err}");
282        lf.status = FitStatus::DidNotConverge;
283        return Some(lf);
284      }
285      Ok(solver) => solver
286    };
287    let result = Executor::new(points.clone(), solver)
288      .configure(|state| state.max_iters(2000))
289      .run();
290    match result {
291      Err(err) => {
292        error!("Minimization failed! {err}");
293        lf.status = FitStatus::DidNotConverge;
294      }
295      Ok(res) => {
296        let state = res.state();
297        if let Some(best) = state.get_best_param() {
298          lf.x0   = best[0] as f32;
299          lf.y0   = best[1] as f32;
300          lf.tx   = best[2] as f32;
301          lf.ty   = best[3] as f32;
302          lf.chi2 = points.chi2(best) as f32;
303          lf.cost = state.get_best_cost() as f32;
304        }
305        lf.status = match state.get_termination_reason() {
306          Some(TerminationReason::SolverConverged) => FitStatus::Success,
307          _                                        => FitStatus::DidNotConverge,
308        };
309      }
310    }
311    Some(lf)
312  }

314  /// Fit a straight line, using only hits within a cylinder
315  /// around the line (LineFitCleaning::cylinder_radius).
316  ///
317  /// The hits have to be ordered in time, with the TOF hits
318  /// first. Tracker hits do not have a time (NaN).
319  ///
320  /// 1) The first line is constructed through the first two
321  ///    hits with a time. These are not necessarily one outer
322  ///    and one inner TOF hit, fit_cleaned_event makes sure
323  ///    of that, using the paddle ids.
324  /// 2) All hits within the cylinder around the line are
325  ///    accepted, the others rejected.
326  /// 3) The line is fitted to the accepted hits.
327  /// 4) The radius of the cylinder shrinks, if this is
328  ///    configured (LineFitCleaning::final_radius).
329  /// 5) Steps 2) to 4) are repeated with the fitted line
330  ///    until the final radius is reached and the accepted
331  ///    hits do not change anymore, or
332  ///    LineFitCleaning::max_iterations is reached.
333  ///
334  /// If a fit after the first one fails (e.g. less than 2 hits
335  /// are accepted with a smaller radius), the fit of the
336  /// iteration before is returned, with its accepted hits and
337  /// the status FitStatus::FallbackPreviousIteration.
338  ///
339  /// Returns None if there are less than 2 hits with a time,
340  /// the first two of them are at the same position, or the
341  /// first fit fails (less than 2 hits are accepted or they
342  /// can not be fit).
343  ///
344  /// # Arguments:
345  ///   * ts             : hit times, NaN for tracker hits
346  ///   * errs_x, errs_y : errors on x and y for each point.
347  ///                      If None, LINEFIT_DEFAULT_ERR_XY
348  ///                      will be used
349  pub fn fit_cleaned(xs       : &[f32],
350                     ys       : &[f32],
351                     zs       : &[f32],
352                     ts       : &[f32],
353                     errs_x   : Option<&[f32]>,
354                     errs_y   : Option<&[f32]>,
355                     cleaning : &LineFitCleaning) -> Option<CleanedLineFit> {
356    if ts.len() != xs.len() {
357      error!("Got {} t values for {} points!", ts.len(), xs.len());
358      return None;
359    }
360    let mut timed = (0..ts.len()).filter(|k| !ts[*k].is_nan());
361    let (Some(i), Some(j)) = (timed.next(), timed.next()) else {
362      error!("Need at least 2 hits with a time (TOF hits) to construct the first line!");
363      return None;
364    };
365    Self::fit_cleaned_with_seed(xs, ys, zs, errs_x, errs_y, (i, j), cleaning)
366  }

368  /// Fit a straight line to the hits of an event, using only
369  /// hits within a cylinder around the line, see fit_cleaned.
370  ///
371  /// The first line is constructed through the first outer
372  /// and the first inner TOF hit (in time), the inner TOF
373  /// being the paddles up to TOF_INNER_MAX_PADDLE_ID.
374  ///
375  /// The hits are the TOF hits (in the order of
376  /// event.tof_event.hits), followed by the tracker hits (in
377  /// the order of event.tracker_hits), so for the n TOF hits,
378  /// accepted[k] is the TOF hit k for k < n and the tracker
379  /// hit k - n otherwise. Tracker hits without coordinates
380  /// are never accepted.
381  ///
382  /// Returns None if the TOF paddle information has not been
383  /// set for the event (TofEvent::set_paddles), there is no
384  /// outer or no inner TOF hit with a time, or the fit fails
385  /// (see fit_cleaned).
386  pub fn fit_cleaned_event(event : &TelemetryEvent, cleaning : &LineFitCleaning) -> Option<CleanedLineFit> {
387    // without paddle information the TOF hits have no
388    // coordinates, get_pointcloud reports that
389    let tof   = event.tof_event.get_pointcloud()?;
390    let first = |inner : bool| -> Option<usize> {
391      event.tof_event.hits.iter().enumerate()
392        .filter(|(_, h)| (h.paddle_id <= TOF_INNER_MAX_PADDLE_ID) == inner && !h.event_t0.is_nan())
393        .min_by(|(_, a), (_, b)| a.event_t0.total_cmp(&b.event_t0))
394        .map(|(k, _)| k)
395    };
396    let (Some(outer), Some(inner)) = (first(false), first(true)) else {
397      error!("Need at least one outer and one inner TOF hit with a time to construct the first line!");
398      return None;
399    };
400    let points : Vec<(f32, f32, f32, f32, f32)> = tof.into_iter()
401      .chain(event.get_tracker_pointcloud())
402      .collect();
403    let usable : Vec<bool> = event.tof_event.hits.iter().map(|_| true)
404      .chain(event.tracker_hits.iter().map(|h| h.has_coordinates))
405      .collect();
406    let xs : Vec<f32> = points.iter().map(|p| p.0).collect();
407    let ys : Vec<f32> = points.iter().map(|p| p.1).collect();
408    let zs : Vec<f32> = points.iter().map(|p| p.2).collect();
409    Self::fit_cleaned_core(&xs, &ys, &zs, None, None, &usable, (outer, inner), cleaning)
410  }

426  /// The iteration of fit_cleaned. Hits which are not usable
427  /// are never accepted.
428  fn fit_cleaned_core(xs       : &[f32],
429                      ys       : &[f32],
430                      zs       : &[f32],
431                      errs_x   : Option<&[f32]>,
432                      errs_y   : Option<&[f32]>,
433                      usable   : &[bool],
434                      seed     : (usize, usize),
435                      cleaning : &LineFitCleaning) -> Option<CleanedLineFit> {
436    let n = xs.len();
437    if ys.len() != n || zs.len() != n || usable.len() != n {
438      error!("Got {} x, {} y and {} z values! They need to be of the same length!", n, ys.len(), zs.len());
439      return None;
440    }
441    for errs in [errs_x, errs_y].into_iter().flatten() {
442      if errs.len() != n {
443        error!("Got {} errors for {} points!", errs.len(), n);
444        return None;
445      }
446    }
447    let (i, j) = seed;
448    if i >= n || j >= n {
449      error!("Seed hits {:?} out of range for {} points!", seed, n);
450      return None;
451    }
452    let anchor = (xs[i], ys[i], zs[i]);
453    let dir    = (xs[j] - xs[i], ys[j] - ys[i], zs[j] - zs[i]);
454    let norm   = (dir.0*dir.0 + dir.1*dir.1 + dir.2*dir.2).sqrt();
455    if norm == 0.0 {
456      error!("The two seed hits are at the same position, can not construct a line!");
457      return None;
458    }
459    let final_radius = cleaning.final_radius.min(cleaning.cylinder_radius);
460    let mut line     = (anchor, (dir.0/norm, dir.1/norm, dir.2/norm));
461    let mut accepted : Vec<bool> = Vec::new();
462    let mut fit : Option<LineFit> = None;
463    let mut converged = false;
464    let mut iterations = 0;
465    // the radius for the next selection of hits, and the
466    // one the accepted hits of fit have been selected with
467    let mut radius   = cleaning.cylinder_radius;
468    let mut fit_radius = radius;
469    while iterations < cleaning.max_iterations {
470      let now : Vec<bool> = (0..n)
471        .map(|k| usable[k] && distance_to_line((xs[k], ys[k], zs[k]), line.0, line.1) <= radius)
472        .collect();
473      if fit.is_some() && now == accepted {
474        // the same hits give the same fit
475        fit_radius = radius;
476        if radius <= final_radius {
477          converged = true;
478          break;
479        }
480        radius = cleaning.next_radius(radius);
481        continue;
482      }
483      let select = |vs : &[f32]| -> Vec<f32> {
484        vs.iter().zip(&now).filter(|(_, a)| **a).map(|(v, _)| *v).collect()
485      };
486      let (ex, ey) = (errs_x.map(select), errs_y.map(select));
487      let (sx, sy, sz) = (select(xs), select(ys), select(zs));
488      // with a previous fit to fall back to, less than 2 hits
489      // are an expected outcome (e.g. of a shrinking cylinder)
490      // and not worth an error from the fit
491      let result = if fit.is_some() && sx.len() < 2 {
492        None
493      } else {
494        match cleaning.cost {
495          LineFitCost::Chi2           => Self::fit(&sx, &sy, &sz, ex.as_deref(), ey.as_deref()),
496          LineFitCost::SumOfDistances => Self::fit_sum_of_distances(&sx, &sy, &sz, ex.as_deref(), ey.as_deref()),
497        }
498      };
499      let Some(lf) = result else {
500        // keep the fit of the iteration before, if there is one
501        let mut previous = fit?;
502        previous.status  = FitStatus::FallbackPreviousIteration;
503        return Some(CleanedLineFit {
504          fit        : previous,
505          accepted,
506          seed,
507          iterations,
508          converged  : false,
509          radius     : fit_radius,
510        });
511      };
512      iterations += 1;
513      accepted    = now;
514      fit_radius  = radius;
515      radius      = cleaning.next_radius(radius);
516      line        = ((lf.x0, lf.y0, lf.z0), lf.direction());
517      fit         = Some(lf);
518    }
519    Some(CleanedLineFit {
520      fit  : fit?,
521      accepted,
522      seed,
523      iterations,
524      converged,
525      radius : fit_radius,
526    })
527  }

535  /// Evaluate the line at a given z
536  pub fn eval(&self, z : f32) -> (f32, f32, f32) {
537    (self.x0 + self.tx*(z - self.z0), self.y0 + self.ty*(z - self.z0), z)
538  }

540  /// Unit direction vector of the line, pointing
541  /// towards +z
542  pub fn direction(&self) -> (f32, f32, f32) {
543    let norm = (self.tx*self.tx + self.ty*self.ty + 1.0).sqrt();
544    (self.tx/norm, self.ty/norm, 1.0/norm)
545  }

957/// The points (and their weights) a LineFit is performed on.
958/// Also serves as the cost function for the minimization.
959#[derive(Debug, Clone)]
960struct LineFitPoints {
961  xs : Vec<f64>,
962  ys : Vec<f64>,
963  zs : Vec<f64>,
964  /// weights 1/err^2
965  wx : Vec<f64>,
966  wy : Vec<f64>,
967  /// common anchor z for x(z) and y(z)
968  z0 : f64,
969}
970
971impl LineFitPoints {
972  fn new(xs     : &[f32],
973         ys     : &[f32],
974         zs     : &[f32],
975         errs_x : Option<&[f32]>,
976         errs_y : Option<&[f32]>) -> Option<Self> {
977    if xs.len() != ys.len() || xs.len() != zs.len() {
978      error!("Got {} x, {} y and {} z values! They need to be of the same length!", xs.len(), ys.len(), zs.len());
979      return None;
980    }
981    if xs.len() < 2 {
982      error!("Not enough points! Need at least 2, got {}", xs.len());
983      return None;
984    }
985    let weights = |errs : Option<&[f32]>| -> Option<Vec<f64>> {
986      match errs {
987        None    => Some(vec![1.0/(LINEFIT_DEFAULT_ERR_XY as f64).powi(2); xs.len()]),
988        Some(e) => {
989          if e.len() != xs.len() {
990            error!("Got {} errors for {} points!", e.len(), xs.len());
991            return None;
992          }
993          Some(e.iter().map(|k| 1.0/(*k as f64).powi(2)).collect())
994        }
995      }
996    };
997    let zs : Vec<f64> = zs.iter().map(|k| *k as f64).collect();
998    let z0 = zs.iter().sum::<f64>()/zs.len() as f64;
999    Some(Self {
1000      xs : xs.iter().map(|k| *k as f64).collect(),
1001      ys : ys.iter().map(|k| *k as f64).collect(),
1002      zs,
1003      wx : weights(errs_x)?,
1004      wy : weights(errs_y)?,
1005      z0,
1006    })
1007  }
1008
1009  /// Weighted linear regression v(z) = a + b*(z - z0)
1010  ///
1011  /// Returns a, b, var(a), var(b) or None if all z
1012  /// are the same
1013  fn regression(&self, vs : &[f64], ws : &[f64]) -> Option<(f64, f64, f64, f64)> {
1014    let sw   : f64 = ws.iter().sum();
1015    let zbar : f64 = ws.iter().zip(&self.zs).map(|(w,z)| w*z).sum::<f64>()/sw;
1016    let vbar : f64 = ws.iter().zip(vs).map(|(w,v)| w*v).sum::<f64>()/sw;
1017    let mut szz = 0.0f64;
1018    let mut szv = 0.0f64;
1019    for k in 0..vs.len() {
1020      szz += ws[k]*(self.zs[k] - zbar).powi(2);
1021      szv += ws[k]*(self.zs[k] - zbar)*(vs[k] - vbar);
1022    }
1023    if szz <= f64::EPSILON*sw {
1024      // not an error of the program: the fit just returns None,
1025      // e.g. for candidate lines of searchlight with all hits on
1026      // one paddle plane
1027      debug!("All points have the same z, can not fit a line as a function of z!");
1028      return None;
1029    }
1030    let b     = szv/szz;
1031    // shift the anchor from the weighted mean zbar to z0
1032    let a     = vbar + b*(self.z0 - zbar);
1033    let var_b = 1.0/szz;
1034    let var_a = 1.0/sw + (self.z0 - zbar).powi(2)/szz;
1035    Some((a, b, var_a, var_b))
1036  }
1037
1038  /// Squared, normalized residuals in x and y for each
1039  /// point, for parameters x0, y0, tx, ty
1040  fn residuals2<'a>(&'a self, p : &'a [f64]) -> impl Iterator<Item = (f64, f64)> + 'a {
1041    (0..self.xs.len()).map(move |k| {
1042      let dz = self.zs[k] - self.z0;
1043      let rx = self.xs[k] - (p[0] + p[2]*dz);
1044      let ry = self.ys[k] - (p[1] + p[3]*dz);
1045      (rx*rx*self.wx[k], ry*ry*self.wy[k])
1046    })
1047  }
1048
1049  fn chi2(&self, p : &[f64]) -> f64 {
1050    self.residuals2(p).map(|(rx2, ry2)| rx2 + ry2).sum()
1051  }
1052
1053  fn sum_of_distances(&self, p : &[f64]) -> f64 {
1054    self.residuals2(p).map(|(rx2, ry2)| (rx2 + ry2).sqrt()).sum()
1055  }
1056}

1058impl CostFunction for LineFitPoints {
1059  type Param  = Vec<f64>;
1060  type Output = f64;
1061
1062  fn cost(&self, p : &Self::Param) -> Result<Self::Output, argmin::core::Error> {
1063    Ok(self.sum_of_distances(p))
1064  }

1083/// Neighbouring TOF paddles of a panel overlap a little across
1084/// their width (about 11 mm). If both paddles of such a pair are
1085/// hit at the same time, the particle went through the overlap,
1086/// which fixes its position across the paddles much better than
1087/// the width of a single paddle (160 mm).
1088///
1089/// The overlaps are found once from the paddle geometry and can
1090/// then be used for all events, see TofEvent::merge_overlapping_hits.
1091#[cfg(feature="database")]
1092#[derive(Debug, Clone, Default)]
1093#[cfg_attr(feature="pybindings", pyclass)]
1094pub struct TofPaddleOverlaps {
1095  /// (paddle id a, paddle id b) with a < b -> their overlap
1096  pub overlaps : BTreeMap<(u8, u8), TofPaddleOverlap>,
1097}
1098
1099#[cfg(feature="database")]
1100impl TofPaddleOverlaps {
1101  /// Paddles further apart than this along their normal do not
1102  /// belong to the same panel (mm)
1103  pub const MAX_NORMAL_DISTANCE : f32 = 60.0;
1104  /// Overlaps narrower than this are ignored (mm)
1105  pub const MIN_WIDTH : f32 = 0.5;
1106  /// Default for the largest time difference of two hits in
1107  /// overlapping paddles to be merged (ns)
1108  pub const DEFAULT_MAX_DT : f32 = 1.0;
1109
1110  /// Find the overlapping pairs: parallel paddles of the same
1111  /// panel whose widths overlap
1112  ///
1113  /// # Arguments:
1114  ///   * paddles : paddle id -> paddle, as from the database
1115  pub fn new(paddles : &HashMap<u8, TofPaddle>) -> Self {
1116    let dot   = |a : [f32;3], b : [f32;3]| a[0]*b[0] + a[1]*b[1] + a[2]*b[2];
1117    let cross = |a : [f32;3], b : [f32;3]| [a[1]*b[2] - a[2]*b[1],
1118                                            a[2]*b[0] - a[0]*b[2],
1119                                            a[0]*b[1] - a[1]*b[0]];
1120    let vec3  = |v : (f32, f32, f32)| [v.0, v.1, v.2];
1121    let mut overlaps = BTreeMap::<(u8, u8), TofPaddleOverlap>::new();
1122    let mut ids : Vec<u8> = paddles.keys().copied().collect();
1123    ids.sort();
1124    for (k, &a) in ids.iter().enumerate() {
1125      for &b in &ids[k+1..] {
1126        let (pa, pb) = (&paddles[&a], &paddles[&b]);
1127        let (na, nb) = (vec3(pa.normal()), vec3(pb.normal()));
1128        let (ra, rb) = (vec3(pa.principal()), vec3(pb.principal()));
1129        if (dot(na, nb).abs() - 1.0).abs() > 1e-3 || (dot(ra, rb).abs() - 1.0).abs() > 1e-3 {
1130          continue;
1131        }
1132        // the database has positions and sizes in cm
1133        let (ca, cb) = (vec3(pa.center_pos()), vec3(pb.center_pos()));
1134        let ca = [10.0*ca[0], 10.0*ca[1], 10.0*ca[2]];
1135        let d  = [10.0*cb[0] - ca[0], 10.0*cb[1] - ca[1], 10.0*cb[2] - ca[2]];
1136        if dot(d, na).abs() > Self::MAX_NORMAL_DISTANCE {
1137          continue;
1138        }
1139        let across = cross(na, ra);
1140        let norm   = dot(across, across).sqrt();
1141        if norm < 0.5 {
1142          // normal or principal axis not set
1143          continue;
1144        }
1145        let across = [across[0]/norm, across[1]/norm, across[2]/norm];
1146        let da     = dot(d, across);
1147        let width  = 5.0*(pa.width + pb.width) - da.abs();
1148        if width < Self::MIN_WIDTH || dot(d, ra).abs() >= 5.0*(pa.length + pb.length) {
1149          continue;
1150        }
1151        let position = dot(ca, across) + da.signum()*(5.0*pa.width - 0.5*width);
1152        overlaps.insert((a, b), TofPaddleOverlap { across, position, width });
1153      }
1154    }
1155    Self { overlaps }
1156  }

1158  /// The overlap of the paddles a and b (in any order), if any
1159  pub fn get(&self, a : u8, b : u8) -> Option<&TofPaddleOverlap> {
1160    self.overlaps.get(&(a.min(b), a.max(b)))
1161  }
1162
1163  pub fn len(&self) -> usize {
1164    self.overlaps.len()
1165  }
1166
1167  pub fn is_empty(&self) -> bool {
1168    self.overlaps.is_empty()
1169  }

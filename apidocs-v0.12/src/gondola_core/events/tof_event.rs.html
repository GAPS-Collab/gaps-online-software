11/// Main event class for the TOF. This will be sent over telemetry and be 
12/// written to disk
13///
14/// CHANGELOG: v0.11 (gondola-core): Merges TofEvent, TofEventSummary and 
15/// MasterTriggerEvent all into a single TofEvent, based on former TofEventSummary.
16/// The new TofEvent has the ability to cary RBEvents and thus mimick the "old" TofEvent
17/// We are using the version flag to indicate:
18/// * ProtocolVersion::Unknown - The "old" TofEventSummary. (now "TofEvent"). No extra 
19///   variables for the GCU, no RBEvents
20/// * ProtocolVersion::V1      - The version crafted for Antarctica '24/'25 containing 
21///   a bunch of summary variables for the GCU (e.g. nhits(umbrella)). This version will 
22///   add these variables to the bytestream and also if ProtocolVersion::V1 is read out 
23///   from the bytestream, the variables are expected to be in it. This is to keep compatibility
24///   with the gcu
25/// * ProtocolVersion::V2     - v0.11 (gondola-core) version of TofEvent(Summary).
26///   This version will not write out GCU variables and does not expect them to be in the 
27///   bytestream. If desired, this version can read/write RBEvents. 
28/// * ProtocolVersion::V3     - the "latest and greatest". This version has gcuvariables 
29///                             AND rbevents. RBEvents can be stripped off later on.
30#[derive(Debug, Clone, PartialEq)]
31#[cfg_attr(feature="pybindings", pyclass)]

130  /// Remove hits from the event as proposed by Elena. 
131  /// This is pretty similar to the non-causal hits, 
132  /// however, a bit stricter
133  pub fn elena_cuts(hit : &TofHit) -> bool {
134    let peak_cut   = 10.0f32;
135    let charge_cut = 5.0f32;
136    let time_sat   = 490.032;
137    
138    let hit_a = hit.time_a.to_f32() > 0.1
139             && hit.time_a.to_f32() < time_sat
140             && hit.peak_a.to_f32() > peak_cut
141             && hit.charge_a.to_f32() > charge_cut;
142    let hit_b = hit.time_b.to_f32() > 0.1
143             && hit.time_b.to_f32() < time_sat
144             && hit.peak_b.to_f32() > peak_cut
145             && hit.charge_b.to_f32() > charge_cut;
146    
147    return hit_a && hit_b;
148  }

150  /// Remove the hits from the event which are not passing 
151  /// Elena's cuts 
152  ///
153  /// # Arguments:
154  ///   * no_return : If true, don't return anything, which 
155  ///                 will make the execution time shorter
156  ///
157  pub fn remove_elena_hits(&mut self, no_return : bool) -> Option<Vec<TofHit>> {
158    if !no_return {
159      let mut removed = Vec::<TofHit>::new();
160      for h in &self.hits {
161        if Self::elena_cuts(h) {
162          removed.push(h.clone());
163        }
164      }
165      self.hits.retain(|h| Self::elena_cuts(h));
166      return Some(removed);
167    } else {
168      self.hits.retain(|h| Self::elena_cuts(h));
169      return None; 
170    }
171  }

174  /// The actual time-of-flight as calculated by the 
175  /// first inner hit time - first outer hit time. 
176  ///
177  /// This is a crude version and does not include 
178  /// any reconstruction, just the bare information 
179  /// as available to the TOF. 
180  ///
181  /// Be careful with using this number for anything else 
182  /// than a crude guess
183  ///
184  /// # Returns:
185  ///   * (tof,beta) for events with at least one inner and outer hit, 
186  ///     None for the others
187  pub fn get_tof(&mut self) -> Option<(f32,f32,f32,f32,f32)> {
188    // in case the hits are already normalized, this should do 
189    // nothing 
190    self.normalize_hit_times();
191    let mut outer_h = Vec::<TofHit>::new();
192    let mut inner_h = Vec::<TofHit>::new();
193    for h in &self.hits {
194      if h.paddle_id < 61 {
195        inner_h.push(*h);
196      } else {
197        outer_h.push(*h);
198      }
199    }
200    if inner_h.len() == 0 || outer_h.len() == 0 {
201      return None;
202    }
203    inner_h .sort_by(|a,b| (a.get_t0_with_local_corrections()).partial_cmp(&&b.get_t0_with_local_corrections()).unwrap_or(Ordering::Greater));
204    outer_h .sort_by(|a,b| (a.get_t0_with_local_corrections()).partial_cmp(&&b.get_t0_with_local_corrections()).unwrap_or(Ordering::Greater));
205    //let tof  = (inner_h[0].t0_fr() - outer_h[0].t0_fr()).abs();
206    let dist     = inner_h[0].distance(&outer_h[0])/1000.0; 
207    // fix wrong phase difference calculation 
208    let t0       = inner_h[0].get_t0_with_local_corrections() - inner_h[0].get_phase_ns() - inner_h[0].timing_offset;
209    let t1       = outer_h[0].get_t0_with_local_corrections() - outer_h[0].get_phase_ns() - outer_h[0].timing_offset;
210    let mut pd   = inner_h[0].get_phase_ns() - outer_h[0].get_phase_ns();
211    if pd > 25.0 {
212      pd -= 50.0;
213    }
214    if pd < -25.0 {
215      pd += 50.0; 
216    }
217    let tof = t0 - t1 + pd;  
218    //let mut tof  = inner_h[0].t0_fr() - outer_h[0].t0_fr();
219    //if tof > 50.0 {
220    //  tof -= 50.0; 
221    //}
222    let beta = dist.abs()/(tof*1e-9)/299792458.0;
223    //let pd   = inner_h[0].get_phase_ns() - outer_h[0].get_phase_ns();
224    let hart_diff = inner_h[0].hart_cable_time - outer_h[0].hart_cable_time;
225    Some((tof, beta, pd, dist, hart_diff))
226  }

234    let gps = self.mt_tiu_gps32 as u64;
235    let mut timestamp = self.mt_timestamp as u64;
236    if timestamp < self.mt_tiu_timestamp as u64 {
237      // it has wrapped
238      timestamp += u32::MAX as u64 + 1;
239    }
240    let gps_mult = match 100_000_000u64.checked_mul(gps) {

247      }
248    };
249    let ts = gps_mult + (timestamp - self.mt_tiu_timestamp as u64);
250    ts

304  /// Simple check if the event contains as much RBEvents 
305  /// as expected from the provided boards masks by the MTB
306  pub fn is_complete(&mut self, exclude_rbs : Option<(&Vec<u8>,&DsiJChRbMapping)>) -> bool {
307    if exclude_rbs.is_none() {
308      return self.get_rb_link_ids().len() == self.rb_events.len();
309    } else {
310      let dead_rbs = exclude_rbs.unwrap();
311      let mut n_known_dead = 0usize;
312      let t_hits = self.get_trigger_hits();
313      for h in t_hits {
314        match dead_rbs.1.get(&h.0) {
315          None => {
316            continue;
317          }
318          Some(dsi) => {
319            match dsi.get(&h.1) {
320              None => {
321                continue;
322              }
323              Some(j) => {
324                match j.get(&h.2.0) {
325                  None => {
326                    continue;
327                  }
328                  Some(rb) => {
329                    if dead_rbs.0.contains(&rb) {
330                      n_known_dead += 1
331                    }
332                  }
333                }
334              }
335            }
336          }
337        }
338      } // end loop over trigger hits
339        //  we use <= here, because sometimes, 
340        //  the link ids are wrong, so n_known_dead 
341        //  might be false positive
342      let n_rb_link_ids = self.get_rb_link_ids().len();
343      //if n_rb_link_ids <= self.rb_events.len() + n_known_dead {
344      //  self.status = EventStatus::KnownDeadRB; 
345      //}
346      return n_rb_link_ids <= self.rb_events.len() + n_known_dead;
347    }
348  }

386  /// Calculate the TOF part of the interesting events mechanism, whcih is
387  /// NHIT (CBE, COR, UMB) and EDEP (CBE, COR, UMB)
388  ///
389  /// The energies are the linear estimate without attenuation
390  /// (TofHit::get_edep_linear_noatt), not the default TofHit::get_edep:
391  /// this runs on the flight computer (liftof), where the hits do not
392  /// know their paddle, and the thresholds of the interesting events
393  /// are set for this estimate.
394  pub fn calc_gcu_variables(&mut self) {
395    if self.hits.len() == 0 {
396      for rbev in &self.rb_events {
397        for h in &rbev.hits {
398          if h.paddle_id <= 60 {
399            self.n_hits_cbe += 1;
400            self.tot_edep_cbe += h.get_edep_linear_noatt();
401          }
402          else if h.paddle_id <= 108 && h.paddle_id > 60 {
403            self.n_hits_umb += 1;
404            self.tot_edep_umb += h.get_edep_linear_noatt();
405          }
406          else {
407            self.n_hits_cor += 1;
408            self.tot_edep_cor += h.get_edep_linear_noatt();
409          }
410        }
411      }
412    } else { 
413      for h in &self.hits {
414        if h.paddle_id <= 60 {
415          self.n_hits_cbe += 1;
416          self.tot_edep_cbe += h.get_edep_linear_noatt();
417        }
418        else if h.paddle_id <= 108 && h.paddle_id > 60 {
419          self.n_hits_umb += 1;
420          self.tot_edep_umb += h.get_edep_linear_noatt();
421        }
422        else {
423          self.n_hits_cor += 1;
424          self.tot_edep_cor += h.get_edep_linear_noatt();
425        }
426      }
427    }

440  /// which absorb any systematic besides the phase & cable length.
441  ///
442  /// This will NOT yet subtract these numbers. To 
443  /// do so, the hit times have to be "normalized" by

446  /// Hits of paddles which are not in offsets get an offset

458      error!("The hit times of event {} have already been normalized, the timing offsets can not be applied anymore! Call set_timing_offsets before normalize_hit_times.", self.event_id);
459      return;
460    }
461    for h in self.hits.iter_mut() {
462      h.timing_offset = offsets.get(&h.paddle_id).copied().unwrap_or(0.0);
463    }
464  }

557  /// Revisit all hit times and apply all event-wide 
558  /// corrections (local, e.g. cable corrections) 
559  /// and global (phase difference, timing offsets) 
560  /// applied and then after, sort them by time.
561  ///
562  /// The order of functions calls needs to be 
563  /// 1) ev.set_timing_offsets 
564  /// 2) ev.normalize_hit_times
565  ///
566  /// This is only done once per event (see
567  /// hit_times_normalized), further calls do nothing.
568  #[cfg(feature="database")]
569  pub fn normalize_hit_times(&mut self) {
570    if self.hits.len() == 0 {
571      return;
572    }
573    if self.hit_times_normalized {
574      return;
575    }
576
577    let phase0 = self.hits[0].phase.to_f32();
578    // add the corrections here 
579    // 1) phase
580    // 2) second timing offsets 
581    for h in &mut self.hits {
582      let t0 = h.get_t0_uncorrected() + h.get_cable_delay();
583      // deal with the phase difference
584      let mut phase_diff = h.phase.to_f32() - phase0;
585      while phase_diff < - PI/2.0 {
586        phase_diff += 2.0*PI;
587      }
588      while phase_diff > PI/2.0 {
589        phase_diff -= 2.0*PI;
590      }
591      let t_shift = 50.0*phase_diff/(2.0*PI);
592      h.event_t0 = t0 + t_shift - h.timing_offset;
593    }
594    // start the first hit at 0
595    //self.hits.sort_by(|a,b| (a.event_t0).partial_cmp(&b.event_t0).unwrap_or(Ordering::Greater));
596    self.hits.sort_by(|a,b| (a.event_t0).total_cmp(&b.event_t0));
597    let t0_first_hit = self.hits[0].event_t0;
598    for h in self.hits.iter_mut() {
599      h.event_t0 -= t0_first_hit;
600    }
601    self.hit_times_normalized = true;
602  }

619  /// Hits in two overlapping paddles (see TofPaddleOverlaps) at
620  /// the same time come from a particle going through the
621  /// overlap: move both hits across the paddles to the middle of
622  /// the overlap. Their positions along the paddles and along the
623  /// normal are kept.
624  ///
625  /// Needs the hit positions (set_paddles) and normalized hit
626  /// times (normalize_hit_times), otherwise nothing is changed.
627  /// A hit whose paddle overlaps with two hit neighbours is moved
628  /// for each pair, the last pair (ordered by paddle ids) wins.
629  ///
630  /// # Arguments:
631  ///   * overlaps : the overlapping paddles
632  ///   * max_dt   : largest time difference of the two hits (ns)
633  ///
634  /// # Returns:
635  ///   * the number of merged pairs
636  #[cfg(feature="database")]
637  pub fn merge_overlapping_hits(&mut self, overlaps : &TofPaddleOverlaps, max_dt : f32) -> usize {
638    if !self.hit_times_normalized {
639      warn!("The hit times are not normalized, can not merge hits of overlapping paddles!");
640      return 0;
641    }
642    let mut index = HashMap::<u8, usize>::new();
643    for (k, h) in self.hits.iter().enumerate() {
644      index.entry(h.paddle_id).or_insert(k);
645    }
646    let mut nmerged = 0usize;
647    for (&(a, b), ov) in &overlaps.overlaps {
648      let (Some(&i), Some(&j)) = (index.get(&a), index.get(&b)) else {
649        continue;
650      };
651      let (hi, hj) = (&self.hits[i], &self.hits[j]);
652      if !(hi.event_t0.is_finite() && hj.event_t0.is_finite())
653         || (hi.event_t0 - hj.event_t0).abs() > max_dt
654         || !(hi.x.is_finite() && hj.x.is_finite()) {
655        continue;
656      }
657      for k in [i, j] {
658        let h     = &mut self.hits[k];
659        let shift = ov.position - (h.x*ov.across[0] + h.y*ov.across[1] + h.z*ov.across[2]);
660        h.x += shift*ov.across[0];
661        h.y += shift*ov.across[1];
662        h.z += shift*ov.across[2];
663      }
664      nmerged += 1;
665    }
666    nmerged
667  }

1008  /// Change the status version when the event is already 
1009  /// packed. The status version is encoded in byte 2 
1010  /// (starting from 0) in the payload of the TofPacket 
1011  pub fn set_packed_status_version(pack : &mut TofPacket, version : ProtocolVersion) 
1012    -> Result<(), SerializationError> {
1013    if pack.packet_type != TofPacketType::TofEvent {
1014      return Err(SerializationError::IncorrectPacketType);
1015    }
1016    let mut status_version = pack.payload[2];
1017    // null the bytes relevant for the version 
1018    status_version  = status_version & 0x3f;
1019    // now or the bytes for the new version 
1020    status_version  = status_version | version.to_u8();
1021    pack.payload[2] = status_version; 
1022    Ok(()) 
1023  }

1025  /// For events with ProtocolVersion == V3, 
1026  /// we have the rbevents at the end and the 
1027  /// packet contains the GCU variables.
1028  ///
1029  /// This can remove the rbevents from a packed bytestrem,
1030  /// and will reset the ProtocolVersion to V2.
1031  /// The result is an event which should be ready 
1032  /// to be sent to the GCU.
1033  pub fn strip_packed_rbevents_for_pv3(pack : &mut TofPacket)
1034    -> Result<(), SerializationError> {
1035    if pack.packet_type != TofPacketType::TofEvent {
1036      return Err(SerializationError::IncorrectPacketType);
1037    }
1038    let status_version = pack.payload[2];
1039    let mut version    = ProtocolVersion::from(status_version & 0xc0);
1040    if version != ProtocolVersion::V3 {
1041      error!("This operation can only be executed on {}, however, this is version {}!", ProtocolVersion::V3, version);
1042      return Err(SerializationError::WrongProtocolVersion);
1043    }
1044    // jump to the start of RBEvents 
1045    let mut pos = 0usize;
1046    pos += 10; // header 
1047    pos += 15; // gcu variables (protocolversion V1 & V3) 
1048    pos += 15;
1049    if pack.payload.len() >= pos {
1050      return Err(SerializationError::StreamTooShort);
1051    }
1052    let nmasks = parse_u8(&pack.payload, &mut pos);
1053    for _ in 0..nmasks {
1054      pos += 2;
1055    }
1056    pos += 8;
1057    let nhits  = parse_u16(&pack.payload,&mut pos);
1058    // set back the version
1059    for _ in 0..nhits {
1060      pos += TofHit::SIZE;
1061      // FIXME - if we don't be able to manage 
1062      //         to have a consistent size for 
1063      //         TofHit, we have to deserialize them 
1064      //         here (or write a minimal deserializer
1065    }
1066    // the next byte is finally the number of RBEvents. 
1067    // So we set that to 0, strip the rest of the paylod 
1068    // and re-attach the TAIL
1069    pack.payload.truncate(pack.payload.len() - pos);
1070    pack.payload.extend_from_slice(&Self::TAIL.to_le_bytes());
1071    version = ProtocolVersion::V2;
1072    Self::set_packed_status_version(pack, version)?;
1073    Ok(())
1074  }

1107    stream.extend_from_slice(&self.timestamp32.to_le_bytes());
1108    stream.extend_from_slice(&self.timestamp16.to_le_bytes());

1136  fn from_bytestream(stream    : &Vec<u8>, 
1137                     pos       : &mut usize) 
1138    -> Result<Self, SerializationError>{
1139    let mut event = Self::new();
1140    let head      = parse_u16(stream, pos);
1141    if head != Self::HEAD {
1142      error!("Decoding of HEAD failed! Got {} instead!", head);
1143      return Err(SerializationError::HeadInvalid);
1144    }
1145    
1146    let status_version_u8   = parse_u8(stream, pos);
1147    let status              = EventStatus::from(status_version_u8 & 0x3f);
1148    let version             = ProtocolVersion::from(status_version_u8 & 0xc0); 
1149    event.status            = status;
1150    event.version           = version;
1151    event.trigger_sources   = parse_u16(stream, pos);
1152    event.n_trigger_paddles = parse_u8(stream, pos);
1153    event.event_id          = parse_u32(stream, pos);
1154    if event.version == ProtocolVersion::V1
1155      || event.version == ProtocolVersion::V3 {
1156      event.n_hits_umb      = parse_u8(stream, pos); 
1157      event.n_hits_cbe      = parse_u8(stream, pos); 
1158      event.n_hits_cor      = parse_u8(stream, pos); 
1159      event.tot_edep_umb    = parse_f32(stream, pos); 
1160      event.tot_edep_cbe    = parse_f32(stream, pos); 
1161      event.tot_edep_cor    = parse_f32(stream, pos); 
1162    }
1163    event.quality            = EventQuality::from(parse_u8(stream, pos));
1164    event.timestamp32        = parse_u32(stream, pos);
1165    event.timestamp16        = parse_u16(stream, pos);
1166    event.run_id             = parse_u16(stream, pos);
1167    event.drs_dead_lost_hits = parse_u16(stream, pos);
1168    event.dsi_j_mask         = parse_u32(stream, pos);
1169    let n_channel_masks        = parse_u8(stream, pos);
1170    for _ in 0..n_channel_masks {
1171      event.channel_mask.push(parse_u16(stream, pos));
1172    }
1173    event.mtb_link_mask      = parse_u64(stream, pos);
1174    let nhits                = parse_u16(stream, pos);
1175    //println!("{}", event);
1176    if nhits > 160 {
1177      error!("There are an abnormous amount of hits in this event!");
1178      return Err(SerializationError::StreamTooLong);
1179    } 
1180    for _ in 0..nhits {
1181      event.hits.push(TofHit::from_bytestream(stream, pos)?);
1182    }
1183    if event.version == ProtocolVersion::V2 
1184      || event.version == ProtocolVersion::V3 {
1185      let n_rb_events = parse_u8(stream, pos);
1186      if n_rb_events > 0 {
1187        for _ in 0..n_rb_events {
1188          event.rb_events.push(RBEvent::from_bytestream(stream, pos)?);
1189        }
1190      }
1191    }
1192
1193    let tail = parse_u16(stream, pos);
1194    if tail != Self::TAIL {
1195      error!("Decoding of TAIL failed for version {}! Got {} instead!", version, tail);
1196      return Err(SerializationError::TailInvalid);
1197    }
1198    Ok(event)
1199  }

1201  /// Allows to get TofEvent from a packet 
1202  /// of the deprecate packet type TofEventDeprecated.
1203  /// This packet type was formerly known as TofEvent
1204  ///
1205  /// This will produce an event with rbevents & hits.
1206  fn from_bytestream_alt(stream    : &Vec<u8>, 
1207                         pos       : &mut usize) 
1208    -> Result<Self, SerializationError> {
1209    let head = parse_u16(stream, pos);
1210    if head != TofEvent::HEAD {
1211      return Err(SerializationError::HeadInvalid);
1212    }
1213    let mut te            = Self::new();
1214    // the compression level will always be 0 for old data
1215    let _compression_level = parse_u8(stream, pos);
1216
1217    te.quality            = EventQuality::from(parse_u8(stream, pos));
1218    // at this position is the serialized TofEventHeader. We don't have that anymore (>v0.11). 
1219    // However, the only information we need from it is the run id, the other fields are anyway
1220    // empty
1221    *pos += 2; // for TofEventHeader::HEAD
1222    // FIXME - potentially dangerous for u16 overflow!
1223    te.run_id             = parse_u32(stream, pos) as u16;
1224    *pos += 43 - 6;// rest of TofEventHeader 
1225    //let header         = TofEventHeader::from_bytestream(stream, &mut pos)?;
1226    // now parse the "old" MasterTriggerEvent
1227    *pos += 2; // MasterTriggerEvent::HEAD
1228    let event_status   = parse_u8 (stream, pos);
1229    te.status              = EventStatus::from(event_status);
1230    if te.has_any_mangling() {
1231      te.status = EventStatus::AnyDataMangling;
1232    }
1233    te.event_id        = parse_u32(stream, pos);
1234    let mtb_timestamp  = parse_u32(stream, pos);
1235    let tiu_timestamp  = parse_u32(stream, pos);
1236    let tiu_gps32      = parse_u32(stream, pos);
1237    let _tiu_gps16     = parse_u16(stream,pos);
1238    let _crc           = parse_u32(stream, pos);
1239    let mt_timestamp   = (mt_event_get_timestamp_abs48(mtb_timestamp, tiu_gps32, tiu_timestamp ) as f64/1000.0).floor()  as u64; 
1240    te.timestamp32      = (mt_timestamp  & 0x00000000ffffffff ) as u32;
1241    te.timestamp16      = ((mt_timestamp & 0x0000ffff00000000 ) >> 32) as u16;
1242    te.trigger_sources  = parse_u16(stream, pos);
1243    te.dsi_j_mask       = parse_u32(stream, pos);
1244    let n_channel_masks = parse_u8(stream, pos);
1245    for _ in 0..n_channel_masks {
1246      te.channel_mask.push(parse_u16(stream, pos));
1247    }
1248
1249    te.mtb_link_mask      = parse_u64(stream, pos);
1250    let mt_event_tail     = parse_u16(stream, pos);
1251    if mt_event_tail != Self::TAIL {
1252      // (tail for mt event was the same)
1253      error!("Parsed TAIL from MT event is incorrect! Got {} instead of {} at pos {}", mt_event_tail, Self::TAIL, pos);
1254    }
1255    //let mt_event      = MasterTriggerEvent::from_bytestream(stream, &mut pos)?;
1256    let v_sizes           = Self::decode_depr_tofevent_size_header(&parse_u32(stream, pos));
1257    //println!("TofEvent - rbevents,  {:?}", v_sizes);
1258    for _ in 0..v_sizes.0 {
1259      // we are getting all waveforms for now, but we can 
1260      // discard them later
1261      let next_rb_event = RBEvent::from_bytestream(stream, pos)?;
1262      //println!("{}", next_rb_event);
1263      te.rb_events.push(next_rb_event);
1264    }
1265    //println!("{}",te);
1266    
1267    // FIXME - this is slow, use Arc<> instead. However, then make 
1268    // sure to copy them in case we get rid of the rb evvnts 
1269    // (or does Arc take care of it) 
1270    for rbev in &te.rb_events {
1271      for h in &rbev.hits {
1272        te.hits.push(*h);
1273      }
1274    }
1275    let tail = parse_u16(stream, pos);
1276    if tail != Self::TAIL {
1277      error!("Decoding of TAIL failed! Got {} instead!", tail);
1278      return Err(SerializationError::TailInvalid);
1279    }
1280    return Ok(te);
1281  }
1282}

1973fn tofevent_timing_offsets_applied_once() {
1974  let mut event = event_for_timing_offsets();
1975  assert!(!event.hit_times_normalized);
1976  // a stale offset, paddle 3 is not in the map
1977  event.hits[2].timing_offset = 7.0;

1979  event.set_timing_offsets(&offsets);

1981  assert_eq!(set, vec![1.0, 4.0, 0.0]);

1985  let t0 : Vec<f32> = event.hits.iter().map(|h| h.event_t0).collect();
1986  assert_eq!(t0, vec![0.0, 7.0, 21.0]);
1987  // normalizing again does not change anything

1989  let again : Vec<f32> = event.hits.iter().map(|h| h.event_t0).collect();
1990  assert_eq!(again, t0);

2009#[cfg(feature="database")]
2010fn tofevent_normalize_with_offsets() {
2011  // the hits have NaN fields (no paddle information), so
2012  // compare only the times and offsets
2013  let times = |ev : &TofEvent| -> Vec<(f32, f32)> {
2014    ev.hits.iter().map(|h| (h.event_t0, h.timing_offset)).collect()
2015  };
2016  let offsets = HashMap::from([(1u8, 1.0f32), (2, 4.0)]);
2017  let mut separate = event_for_timing_offsets();
2018  separate.set_timing_offsets(&offsets);
2019  separate.normalize_hit_times();
2020  let mut together = event_for_timing_offsets();
2021  assert!(!together.timing_offsets_applied());
2022  together.normalize_hit_times_with_offsets(Some(&offsets));
2023  assert_eq!(times(&separate), times(&together));
2024  assert!(together.hit_times_normalized && together.timing_offsets_applied());
2025  // None uses the offsets set before

2027  before.set_timing_offsets(&offsets);

2029  assert_eq!(times(&before), times(&separate));
2030  // without any offsets
2031  let mut none = event_for_timing_offsets();
2032  none.normalize_hit_times_with_offsets(None);
2033  assert!(none.hit_times_normalized && !none.timing_offsets_applied());
2034  // too late, as for set_timing_offsets
2035  none.normalize_hit_times_with_offsets(Some(&offsets));
2036  assert!(!none.timing_offsets_applied());

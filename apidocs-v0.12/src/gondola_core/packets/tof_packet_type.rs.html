7/// Types of serializable data structures used
8/// throughout the TOF system. 
9#[derive(Debug, Hash, Eq, PartialEq, Clone, Copy, FromRepr, AsRefStr, EnumIter)]
10#[cfg_attr(feature = "pybindings", pyclass(eq, eq_int))]
11#[repr(u8)]
12pub enum TofPacketType {
13  Unknown               = 0u8, 
14  RBEvent               = 20u8,
15  // v0.11 TofEvent -> TofEventDeprecated
16  TofEventDeprecated    = 21u8,
17  RBWaveform            = 22u8,
18  // v0.11 TofEventSummary -> TofEvent
19  TofEvent              = 23u8,
20  DataSinkHB            = 40u8,    
21  MasterTrigger         = 60u8,    // needs to be renamed to either MasterTriggerEvent or MTEvent
22  TriggerConfig         = 61u8,
23  MasterTriggerHB       = 62u8, 
24  EventBuilderHB        = 63u8,
25  RBChannelMaskConfig   = 64u8,
26  TofRBConfig           = 68u8,
27  AnalysisEngineConfig  = 69u8,
28  RBEventHeader         = 70u8,    // needs to go away
29  TOFEventBuilderConfig = 71u8,
30  DataPublisherConfig   = 72u8,
31  TofRunConfig          = 73u8,
32  CPUMoniData           = 80u8,
33  MtbMoniData            = 90u8,
34  RBMoniData            = 100u8,
35  PBMoniData            = 101u8,
36  LTBMoniData           = 102u8,
37  PAMoniData            = 103u8,
38  RBEventMemoryView     = 120u8, // We'll keep it for now - indicates that the event
39                                 // still needs to be processed.
40  RBCalibration         = 130u8,
41  TofCommand            = 140u8,
42  TofCommandV2          = 141u8,
43  TofResponse           = 142u8,
44  // needs to go away
45  RBCommand             = 150u8,
46  // > 160 configuration packets
47  RBPing                = 160u8,
48  PreampBiasConfig      = 161u8,
49  RunConfig             = 162u8,
50  LTBThresholdConfig    = 163u8,
51  // avoid 170 since it is our 
52  // delimiter
53  // >= 171 detector status
54  TofDetectorStatus     = 171u8,
55  // use the > 200 values for transmitting
56  // various binary files
57  ConfigBinary          = 201u8,
58  LiftofRBBinary        = 202u8,
59  LiftofBinaryService   = 203u8,
60  LiftofCCBinary        = 204u8,
61  RBCalibrationFlightV  = 210u8,
62  RBCalibrationFlightT  = 211u8,
63  /// A klude which allows us to send bfsw ack packets
64  /// through the TOF system
65  BfswAckPacket         = 212u8,
66  /// a MultiPacket consists of other TofPackets
67  MultiPacket           = 255u8,
68}

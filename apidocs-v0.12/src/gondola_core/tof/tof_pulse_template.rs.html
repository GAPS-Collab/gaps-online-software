3//! TOF pulse template.
4//!
5//! A `TofPulseTemplate` contains the mean and RMS waveform for a
6//! particular TOF paddle, paddle side, position interval, and charge
7//! interval.
8//!
9//! Position is measured from paddle side A in mm.
10//!
11//! The template contains 108 regularly spaced waveform samples. The
12//! corresponding time axis is described by `time_start` and
13//! `time_step`.
14//!
15//! The type implements Gondola's `Serialization` and `TofPackable`
16//! traits and can therefore be stored in a `TofPacket`.
17

22// ============================================================================
23// Constants
24// ============================================================================

26/// Number of samples in a TOF pulse template.
27pub const N_PULSE_TEMPLATE_BINS: usize = 108;
28
29/// Side A identifier.
30pub const PULSE_TEMPLATE_SIDE_A: u8 = 0;
31
32/// Side B identifier.
33pub const PULSE_TEMPLATE_SIDE_B: u8 = 1;
34
35/// Average waveform template for a TOF paddle.
36///
37/// Each template corresponds to:
38///
39/// * one paddle,
40/// * one paddle side,
41/// * one position interval,
42/// * one charge interval.
43///
44/// Position is measured from paddle side A in mm.
45///
46/// `side` uses the convention:
47///
48/// * `0` = side A
49/// * `1` = side B
50///
51/// The waveform time axis is regular and is given by
52///
53/// ```text
54/// time[i] = time_start + i * time_step
55/// ```
56#[derive(Debug, Clone, PartialEq)]
57#[cfg_attr(feature = "pybindings", pyclass)]
58pub struct TofPulseTemplate {
59  /// Paddle ID.
60  pub paddle_id: u8,
61  /// Paddle side:
62  ///
63  /// * 0 = A
64  /// * 1 = B
65  pub side: u8,
66  /// Number of waveforms which went into this template 
67  pub n_waveforms : u32,
68  /// Lower edge of the position interval, measured from side A [mm].
69  pub position_low: f32,
70  /// Upper edge of the position interval, measured from side A [mm].
71  pub position_high: f32,
72  /// Lower edge of the charge interval [pC].
73  pub charge_low: f32,
74  /// Upper edge of the charge interval [pC].
75  pub charge_high: f32,
76  /// Time of the first waveform sample [ns].
77  pub time_start: f32,
78  /// Time spacing between waveform samples [ns].
79  pub time_step: f32,
80  /// Mean waveform [mV].
81  pub mean: [f32; N_PULSE_TEMPLATE_BINS],
82  /// RMS waveform [mV].
83  pub rms: [f32; N_PULSE_TEMPLATE_BINS],
84}
85
86
87impl TofPulseTemplate {
88  /// Create an empty TOF pulse template.
89  pub fn new() -> Self {
90    Self {
91      paddle_id    : 0,
92      side         : PULSE_TEMPLATE_SIDE_A,
93      n_waveforms  : 0,
94      position_low : 0.0,
95      position_high: 0.0,
96      charge_low   : 0.0,
97      charge_high  : 0.0,
98      time_start   : 0.0,
99      time_step    : 1.0,
100      mean         : [0.0; N_PULSE_TEMPLATE_BINS],
101      rms          : [0.0; N_PULSE_TEMPLATE_BINS],
102    }
103  }
104
105  /// Return the time coordinate for a waveform sample.
106  pub fn time_at(&self, index: usize) -> Option<f32> {
107      if index >= N_PULSE_TEMPLATE_BINS {
108        return None;
109      }
110      Some(
111        self.time_start  + index as f32 * self.time_step
112      )
113  }
114
115  /// Return the complete waveform time axis.
116  pub fn time_axis(&self) -> Vec<f32> {
117    (0..N_PULSE_TEMPLATE_BINS)
118      .map(|index| {
119        self.time_start + index as f32 * self.time_step
120      })
121      .collect()
122  }
123
124  /// Return `true` if this template describes paddle side A.
125  pub fn is_side_a(&self) -> bool {
126    self.side == PULSE_TEMPLATE_SIDE_A
127  }
128
129  /// Return `true` if this template describes paddle side B.
130  pub fn is_side_b(&self) -> bool {
131    self.side == PULSE_TEMPLATE_SIDE_B
132  }
133}

136impl Default for TofPulseTemplate {
137  fn default() -> Self {
138    Self::new()
139  }

147impl fmt::Display for TofPulseTemplate {
148  fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
149    let side = match self.side {
150      PULSE_TEMPLATE_SIDE_A => "A",
151      PULSE_TEMPLATE_SIDE_B => "B",
152      _ => "?",
153    };
154    let time_end = self.time_start
155      + (N_PULSE_TEMPLATE_BINS - 1) as f32
156          * self.time_step;
157    let mean_min = self
158      .mean
159      .iter()
160      .copied()
161      .filter(|value| value.is_finite())
162      .fold(f32::INFINITY, f32::min);
163    let mean_max = self
164      .mean
165      .iter()
166      .copied()
167      .filter(|value| value.is_finite())
168      .fold(f32::NEG_INFINITY, f32::max);
169    let rms_min = self
170      .rms
171      .iter()
172      .copied()
173      .filter(|value| value.is_finite())
174      .fold(f32::INFINITY, f32::min);
175    let rms_max = self
176      .rms
177      .iter()
178      .copied()
179      .filter(|value| value.is_finite())
180      .fold(f32::NEG_INFINITY, f32::max);
181    write!(
182      f,
183      concat!(
184        "<TofPulseTemplate {{ ",
185        "paddle: {}, ",
186        "side: {}, ",
187        "n wf: {}, ",
188        "position: [{:.1}, {:.1}) mm, ",
189        "charge: [{:.1}, {:.1}) pC, ",
190        "time: [{:.1}, {:.1}] ns (dt={:.2} ns), ",
191        "mean: [{:.2}, {:.2}] mV, ",
192        "rms: [{:.2}, {:.2}] mV ",
193        "}}>"
194      ),
195      self.paddle_id,
196      side,
197      self.n_waveforms,
198      self.position_low,
199      self.position_high,
200      self.charge_low,
201      self.charge_high,
202      self.time_start,
203      time_end,
204      self.time_step,
205      mean_min,
206      mean_max,
207      rms_min,
208      rms_max,
209    )
210  }
211}

218impl Serialization for TofPulseTemplate {
219  const HEAD: u16 = 0xAAAA;
220  const TAIL: u16 = 0x5555;
221
222  const SIZE: usize =
223    2                                      // HEAD
224    + 1                                    // paddle_id
225    + 1                                    // side
226    + 4                                    // n waveforms
227    + 6 * std::mem::size_of::<f32>()       // scalar f32 metadata
228    + N_PULSE_TEMPLATE_BINS
229        * std::mem::size_of::<f32>()       // mean
230    + N_PULSE_TEMPLATE_BINS
231        * std::mem::size_of::<f32>()       // rms
232    + 2;                                   // TAIL
233
234  fn to_bytestream(&self) -> Vec<u8> {
235    let mut stream = Vec::<u8>::with_capacity(Self::SIZE);
236    // HEAD
237    stream.extend_from_slice(&Self::HEAD.to_le_bytes());
238    // Template identification
239    stream.push(self.paddle_id);
240    stream.push(self.side);
241    stream.extend_from_slice(&self.n_waveforms.to_le_bytes());
242    // Position interval
243    stream.extend_from_slice(&self.position_low.to_le_bytes());
244    stream.extend_from_slice(&self.position_high.to_le_bytes());
245    // Charge interval
246    stream.extend_from_slice(&self.charge_low.to_le_bytes());
247    stream.extend_from_slice(&self.charge_high.to_le_bytes());
248    // Time axis
249    stream.extend_from_slice(&self.time_start.to_le_bytes());
250    stream.extend_from_slice(&self.time_step.to_le_bytes());
251    // Mean waveform
252    for value in &self.mean {
253      stream.extend_from_slice(&value.to_le_bytes() );
254    }
255    // RMS waveform
256    for value in &self.rms {
257      stream.extend_from_slice(&value.to_le_bytes());
258    }
259    stream.extend_from_slice(&Self::TAIL.to_le_bytes());
260    stream
261  }
262
263  fn from_bytestream(stream: &Vec<u8>,pos: &mut usize,) 
264    -> Result<Self, SerializationError> {
265
266    // verify_fixed() checks HEAD and advances pos past it.
267    Self::verify_fixed(stream, pos)?;
268    let mut temp       = Self::new();
269    temp.paddle_id     = parse_u8(stream, pos);
270    temp.side          = parse_u8(stream, pos);
271    temp.n_waveforms   = parse_u32(stream, pos);
272    temp.position_low  = parse_f32(stream, pos);
273    temp.position_high = parse_f32(stream, pos);
274    temp.charge_low    = parse_f32(stream, pos);
275    temp.charge_high   = parse_f32(stream, pos);
276    temp.time_start    = parse_f32(stream, pos);
277    temp.time_step     = parse_f32(stream, pos);
278    let mut mean       = [0.0_f32; N_PULSE_TEMPLATE_BINS];
279    let mut rms        = [0.0_f32; N_PULSE_TEMPLATE_BINS];
280    for value in &mut mean {
281      *value           = parse_f32(stream, pos);
282    }
283    for value in &mut rms {
284      *value           = parse_f32(stream, pos);
285    }
286    let tail =        parse_u16(stream, pos);
287    if tail != Self::TAIL {
288      error!("Got wrong tail bytes! Found {} instead of {}", tail, Self::TAIL);
289      return Err(SerializationError::TailInvalid);
290    }
291    temp.mean = mean;
292    temp.rms  = rms;
293    Ok(temp)
294  }
295}

302#[cfg(feature="random")]
303impl FromRandom for TofPulseTemplate {
304
305  fn from_random() -> Self {
306    let mut template            = Self::new();
307    let mut rng                 = rand::rng();
308    template.paddle_id          = rng.random::<u8>();
309    template.side               = rng.random::<u8>();
310    template.n_waveforms        = rng.random::<u32>();
311    template.position_low       = rng.random::<f32>();
312    template.position_high      = rng.random::<f32>();
313    template.charge_low         = rng.random::<f32>();
314    template.charge_high        = rng.random::<f32>();
315    template.time_start         = rng.random::<f32>();
316    template.time_step          = rng.random::<f32>();
317    for k in 0..N_PULSE_TEMPLATE_BINS {
318      template.mean[k] = rng.random::<f32>();
319      template.rms [k] = rng.random::<f32>();
320    }
321    template
322  }
323}

541#[cfg(test)]
542mod tests {
543  use super::*;

545  #[test]
546  fn packable_tofpulsetemplate() {
547    for _ in 0..500 {
548      let mut data = TofPulseTemplate::from_random();
549      let mut test : TofPulseTemplate = data.pack().unpack().unwrap();
550      assert_eq!(data, test);
551    }
552    let template = TofPulseTemplate::from_random();
553    let stream = template.to_bytestream();
554    assert_eq!(stream.len(), TofPulseTemplate::SIZE);
555
556    let mut pos = 0;
557    let restored = TofPulseTemplate::from_bytestream(&stream, &mut pos).unwrap();
558    assert_eq!(template, restored);
559    assert_eq!(pos, TofPulseTemplate::SIZE);
560  }

563  #[test]
564  fn time_axis_is_correct() {
565    let mut template =
566        TofPulseTemplate::new();
567
568    template.time_start = -7.5;
569    template.time_step = 1.0;
570
571    assert_eq!(
572        template.time_at(0),
573        Some(-7.5)
574    );
575
576    assert_eq!(
577        template.time_at(1),
578        Some(-6.5)
579    );
580
581    assert_eq!(
582        template.time_at(107),
583        Some(99.5)
584    );
585
586    assert_eq!(
587        template.time_at(108),
588        None
589    );
590  }

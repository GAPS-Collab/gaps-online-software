34enum MapStrategy {
35    // Join by key, this the most expensive
36    // for reduced aggregations
37    Join,
38    // explode now
39    Explode,
40    // Use an arg_sort to map the values back
41    Map,
42    Nothing,
43}

46    fn map_list_agg_by_arg_sort(
47        &self,
48        out_column: Column,
49        flattened: &Column,
50        mut ac: AggregationContext,
51        gb: GroupBy,
52    ) -> PolarsResult<IdxCa> {
53        // idx (new-idx, original-idx)
54        let mut idx_mapping = Vec::with_capacity(out_column.len());
55
56        // we already set this buffer so we can reuse the `original_idx` buffer
57        // that saves an allocation
58        let mut take_idx = vec![];
59
60        // groups are not changed, we can map by doing a standard arg_sort.
61        if std::ptr::eq(ac.groups().as_ref(), gb.get_groups()) {
62            let mut iter = 0..flattened.len() as IdxSize;
63            match ac.groups().as_ref().as_ref() {
64                GroupsType::Idx(groups) => {
65                    for g in groups.all() {
66                        idx_mapping.extend(g.iter().copied().zip(&mut iter));
67                    }
68                },
69                GroupsType::Slice { groups, .. } => {
70                    for &[first, len] in groups {
71                        idx_mapping.extend((first..first + len).zip(&mut iter));
72                    }
73                },
74            }
75        }
76        // groups are changed, we use the new group indexes as arguments of the arg_sort
77        // and sort by the old indexes
78        else {
79            let mut original_idx = Vec::with_capacity(out_column.len());
80            match gb.get_groups().as_ref() {
81                GroupsType::Idx(groups) => {
82                    for g in groups.all() {
83                        original_idx.extend_from_slice(g)
84                    }
85                },
86                GroupsType::Slice { groups, .. } => {
87                    for &[first, len] in groups {
88                        original_idx.extend(first..first + len)
89                    }
90                },
91            };
92
93            let mut original_idx_iter = original_idx.iter().copied();
94
95            match ac.groups().as_ref().as_ref() {
96                GroupsType::Idx(groups) => {
97                    for g in groups.all() {
98                        idx_mapping.extend(g.iter().copied().zip(&mut original_idx_iter));
99                    }
100                },
101                GroupsType::Slice { groups, .. } => {
102                    for &[first, len] in groups {
103                        idx_mapping.extend((first..first + len).zip(&mut original_idx_iter));
104                    }
105                },
106            }
107            original_idx.clear();
108            take_idx = original_idx;
109        }
110        // SAFETY:
111        // we only have unique indices ranging from 0..len
112        unsafe { perfect_sort(&POOL, &idx_mapping, &mut take_idx) };
113        Ok(IdxCa::from_vec(PlSmallStr::EMPTY, take_idx))
114    }

116    #[allow(clippy::too_many_arguments)]
117    fn map_by_arg_sort(
118        &self,
119        df: &DataFrame,
120        out_column: Column,
121        flattened: &Column,
122        mut ac: AggregationContext,
123        group_by_columns: &[Column],
124        gb: GroupBy,
125        cache_key: String,
126        state: &ExecutionState,
127    ) -> PolarsResult<Column> {
128        // we use an arg_sort to map the values back
129
130        // This is a bit more complicated because the final group tuples may differ from the original
131        // so we use the original indices as idx values to arg_sort the original column
132        //
133        // The example below shows the naive version without group tuple mapping
134
135        // columns
136        // a b a a
137        //
138        // agg list
139        // [0, 2, 3]
140        // [1]
141        //
142        // flatten
143        //
144        // [0, 2, 3, 1]
145        //
146        // arg_sort
147        //
148        // [0, 3, 1, 2]
149        //
150        // take by arg_sorted indexes and voila groups mapped
151        // [0, 1, 2, 3]
152
153        if flattened.len() != df.height() {
154            let ca = out_column.list().unwrap();
155            let non_matching_group =
156                ca.into_iter()
157                    .zip(ac.groups().iter())
158                    .find(|(output, group)| {
159                        if let Some(output) = output {
160                            output.as_ref().len() != group.len()
161                        } else {
162                            false
163                        }
164                    });
165
166            if let Some((output, group)) = non_matching_group {
167                let first = group.first();
168                let group = group_by_columns
169                    .iter()
170                    .map(|s| format!("{}", s.get(first as usize).unwrap()))
171                    .collect::<Vec<_>>();
172                polars_bail!(
173                    expr = self.expr, ComputeError:
174                    "the length of the window expression did not match that of the group\
175                    \n> group: {}\n> group length: {}\n> output: '{:?}'",
176                    comma_delimited(String::new(), &group), group.len(), output.unwrap()
177                );
178            } else {
179                polars_bail!(
180                    expr = self.expr, ComputeError:
181                    "the length of the window expression did not match that of the group"
182                );
183            };
184        }
185
186        let idx = if state.cache_window() {
187            if let Some(idx) = state.window_cache.get_map(&cache_key) {
188                idx
189            } else {
190                let idx = Arc::new(self.map_list_agg_by_arg_sort(out_column, flattened, ac, gb)?);
191                state.window_cache.insert_map(cache_key, idx.clone());
192                idx
193            }
194        } else {
195            Arc::new(self.map_list_agg_by_arg_sort(out_column, flattened, ac, gb)?)
196        };
197
198        // SAFETY:
199        // groups should always be in bounds.
200        unsafe { Ok(flattened.take_unchecked(&idx)) }
201    }

215    fn is_explicit_list_agg(&self) -> bool {
216        // col("foo").implode()
217        // col("foo").implode().alias()
218        // ..
219        // col("foo").implode().alias().alias()
220        //
221        // but not:
222        // col("foo").implode().sum().alias()
223        // ..
224        // col("foo").min()
225        let mut explicit_list = false;
226        for e in &self.expr {
227            if let Expr::Window { function, .. } = e {
228                // or list().alias
229                let mut finishes_list = false;
230                for e in &**function {
231                    match e {
232                        Expr::Agg(AggExpr::Implode(_)) => {
233                            finishes_list = true;
234                        },
235                        Expr::Alias(_, _) => {},
236                        _ => break,
237                    }
238                }
239                explicit_list = finishes_list;
240            }
241        }
242
243        explicit_list
244    }

288    /// Check if the branches have an aggregation
289    /// when(a > sum)
290    /// then (foo)
291    /// otherwise(bar - sum)
292    fn has_different_group_sources(&self) -> bool {
293        let mut has_arity = false;
294        let mut agg_col = false;
295        for e in &self.expr {
296            if let Expr::Window { function, .. } = e {
297                // or list().alias
298                for e in &**function {
299                    match e {
300                        Expr::Ternary { .. } | Expr::BinaryExpr { .. } => {
301                            has_arity = true;
302                        },
303                        Expr::Alias(_, _) => {},
304                        Expr::Agg(_) => {
305                            agg_col = true;
306                        },
307                        Expr::Function { options, .. }
308                        | Expr::AnonymousFunction { options, .. } => {
309                            if options.flags.contains(FunctionFlags::RETURNS_SCALAR)
310                                && matches!(options.collect_groups, ApplyOptions::GroupWise)
311                            {
312                                agg_col = true;
313                            }
314                        },
315                        _ => {},
316                    }
317                }
318            }
319        }
320        has_arity && agg_col
321    }

323    fn determine_map_strategy(
324        &self,
325        agg_state: &AggState,
326        gb: &GroupBy,
327    ) -> PolarsResult<MapStrategy> {
328        match (self.mapping, agg_state) {
329            // Explode
330            // `(col("x").sum() * col("y")).list().over("groups").flatten()`
331            (WindowMapping::Explode, _) => Ok(MapStrategy::Explode),
332            // // explicit list
333            // // `(col("x").sum() * col("y")).list().over("groups")`
334            // (false, false, _) => Ok(MapStrategy::Join),
335            // aggregations
336            //`sum("foo").over("groups")`
337            (_, AggState::AggregatedScalar(_)) => Ok(MapStrategy::Join),
338            // no explicit aggregations, map over the groups
339            //`(col("x").sum() * col("y")).over("groups")`
340            (WindowMapping::Join, AggState::AggregatedList(_)) => Ok(MapStrategy::Join),
341            // no explicit aggregations, map over the groups
342            //`(col("x").sum() * col("y")).over("groups")`
343            (WindowMapping::GroupsToRows, AggState::AggregatedList(_)) => {
344                if let GroupsType::Slice { .. } = gb.get_groups().as_ref() {
345                    // Result can be directly exploded if the input was sorted.
346                    Ok(MapStrategy::Explode)
347                } else {
348                    Ok(MapStrategy::Map)
349                }
350            },
351            // no aggregations, just return column
352            // or an aggregation that has been flattened
353            // we have to check which one
354            //`col("foo").over("groups")`
355            (WindowMapping::GroupsToRows, AggState::NotAggregated(_)) => {
356                // col()
357                // or col().alias()
358                if self.is_simple_column_expr() {
359                    Ok(MapStrategy::Nothing)
360                } else {
361                    Ok(MapStrategy::Map)
362                }
363            },
364            (WindowMapping::Join, AggState::NotAggregated(_)) => Ok(MapStrategy::Join),
365            // literals, do nothing and let broadcast
366            (_, AggState::Literal(_)) => Ok(MapStrategy::Nothing),
367        }
368    }

376impl PhysicalExpr for WindowExpr {
377    // Note: this was first implemented with expression evaluation but this performed really bad.
378    // Therefore we choose the group_by -> apply -> self join approach
379
380    // This first cached the group_by and the join tuples, but rayon under a mutex leads to deadlocks:
381    // https://github.com/rayon-rs/rayon/issues/592
382    fn evaluate(&self, df: &DataFrame, state: &ExecutionState) -> PolarsResult<Column> {
383        // This method does the following:
384        // 1. determine group_by tuples based on the group_column
385        // 2. apply an aggregation function
386        // 3. join the results back to the original dataframe
387        //    this stores all group values on the original df size
388        //
389        //      we have several strategies for this
390        //      - 3.1 JOIN
391        //          Use a join for aggregations like
392        //              `sum("foo").over("groups")`
393        //          and explicit `list` aggregations
394        //              `(col("x").sum() * col("y")).list().over("groups")`
395        //
396        //      - 3.2 EXPLODE
397        //          Explicit list aggregations that are followed by `over().flatten()`
398        //          # the fastest method to do things over groups when the groups are sorted.
399        //          # note that it will require an explicit `list()` call from now on.
400        //              `(col("x").sum() * col("y")).list().over("groups").flatten()`
401        //
402        //      - 3.3. MAP to original locations
403        //          This will be done for list aggregations that are not explicitly aggregated as list
404        //              `(col("x").sum() * col("y")).over("groups")
405        //          This can be used to reverse, sort, shuffle etc. the values in a group
406
407        // 4. select the final column and return
408
409        if df.is_empty() {
410            let field = self.phys_function.to_field(df.schema())?;
411            return Ok(Column::full_null(field.name().clone(), 0, field.dtype()));
412        }
413
414        let group_by_columns = self
415            .group_by
416            .iter()
417            .map(|e| e.evaluate(df, state))
418            .collect::<PolarsResult<Vec<_>>>()?;
419
420        // if the keys are sorted
421        let sorted_keys = group_by_columns.iter().all(|s| {
422            matches!(
423                s.is_sorted_flag(),
424                IsSorted::Ascending | IsSorted::Descending
425            )
426        });
427        let explicit_list_agg = self.is_explicit_list_agg();
428
429        // if we flatten this column we need to make sure the groups are sorted.
430        let mut sort_groups = matches!(self.mapping, WindowMapping::Explode) ||
431            // if not
432            //      `col().over()`
433            // and not
434            //      `col().list().over`
435            // and not
436            //      `col().sum()`
437            // and keys are sorted
438            //  we may optimize with explode call
439            (!self.is_simple_column_expr() && !explicit_list_agg && sorted_keys && !self.is_aggregation());
440
441        // overwrite sort_groups for some expressions
442        // TODO: fully understand the rationale is here.
443        if self.has_different_group_sources() {
444            sort_groups = true
445        }
446
447        let create_groups = || {
448            let gb = df.group_by_with_series(group_by_columns.clone(), true, sort_groups)?;
449            let mut groups = gb.take_groups();
450
451            if let Some((order_by, options)) = &self.order_by {
452                let order_by = order_by.evaluate(df, state)?;
453                polars_ensure!(order_by.len() == df.height(), ShapeMismatch: "the order by expression evaluated to a length: {} that doesn't match the input DataFrame: {}", order_by.len(), df.height());
454                groups = update_groups_sort_by(&groups, order_by.as_materialized_series(), options)?
455                    .into_sliceable()
456            }
457
458            let out: PolarsResult<GroupPositions> = Ok(groups);
459            out
460        };
461
462        // Try to get cached grouptuples
463        let (mut groups, cache_key) = if state.cache_window() {
464            let mut cache_key = String::with_capacity(32 * group_by_columns.len());
465            write!(&mut cache_key, "{}", state.branch_idx).unwrap();
466            for s in &group_by_columns {
467                cache_key.push_str(s.name());
468            }
469            if let Some((e, options)) = &self.order_by {
470                let e = match e.as_expression() {
471                    Some(e) => e,
472                    None => {
473                        polars_bail!(InvalidOperation: "cannot order by this expression in window function")
474                    },
475                };
476                window_function_format_order_by(&mut cache_key, e, options)
477            }
478
479            let groups = match state.window_cache.get_groups(&cache_key) {
480                Some(groups) => groups,
481                None => create_groups()?,
482            };
483            (groups, cache_key)
484        } else {
485            (create_groups()?, "".to_string())
486        };
487
488        // 2. create GroupBy object and apply aggregation
489        let apply_columns = self.apply_columns.clone();
490
491        // some window expressions need sorted groups
492        // to make sure that the caches align we sort
493        // the groups, so that the cached groups and join keys
494        // are consistent among all windows
495        if sort_groups || state.cache_window() {
496            groups.sort();
497            state
498                .window_cache
499                .insert_groups(cache_key.clone(), groups.clone());
500        }
501        let gb = GroupBy::new(df, group_by_columns.clone(), groups, Some(apply_columns));
502
503        // If the aggregation creates categoricals and `MapStrategy` is `Join`,
504        // the string cache was needed. So we hold it for that case.
505        // Worst case is that a categorical is created with indexes from the string
506        // cache which is fine, as the physical representation is undefined.
507        #[cfg(feature = "dtype-categorical")]
508        let _sc = polars_core::StringCacheHolder::hold();
509        let mut ac = self.run_aggregation(df, state, &gb)?;
510
511        use MapStrategy::*;
512        match self.determine_map_strategy(ac.agg_state(), &gb)? {
513            Nothing => {
514                let mut out = ac.flat_naive().into_owned();
515
516                if ac.is_literal() {
517                    out = out.new_from_index(0, df.height())
518                }
519                if let Some(name) = &self.out_name {
520                    out.rename(name.clone());
521                }
522                Ok(out.into_column())
523            },
524            Explode => {
525                let mut out = ac.aggregated().explode()?;
526                if let Some(name) = &self.out_name {
527                    out.rename(name.clone());
528                }
529                Ok(out.into_column())
530            },
531            Map => {
532                // TODO!
533                // investigate if sorted arrays can be return directly
534                let out_column = ac.aggregated();
535                let flattened = out_column.explode()?;
536                // we extend the lifetime as we must convince the compiler that ac lives
537                // long enough. We drop `GrouBy` when we are done with `ac`.
538                let ac = unsafe {
539                    std::mem::transmute::<AggregationContext<'_>, AggregationContext<'static>>(ac)
540                };
541                self.map_by_arg_sort(
542                    df,
543                    out_column,
544                    &flattened,
545                    ac,
546                    &group_by_columns,
547                    gb,
548                    cache_key,
549                    state,
550                )
551            },
552            Join => {
553                let out_column = ac.aggregated();
554                // we try to flatten/extend the array by repeating the aggregated value n times
555                // where n is the number of members in that group. That way we can try to reuse
556                // the same map by arg_sort logic as done for listed aggregations
557                let update_groups = !matches!(&ac.update_groups, UpdateGroups::No);
558                match (
559                    &ac.update_groups,
560                    set_by_groups(&out_column, &ac.groups, df.height(), update_groups),
561                ) {
562                    // for aggregations that reduce like sum, mean, first and are numeric
563                    // we take the group locations to directly map them to the right place
564                    (UpdateGroups::No, Some(out)) => Ok(out.into_column()),
565                    (_, _) => {
566                        let keys = gb.keys();
567
568                        let get_join_tuples = || {
569                            if group_by_columns.len() == 1 {
570                                let mut left = group_by_columns[0].clone();
571                                // group key from right column
572                                let mut right = keys[0].clone();
573
574                                let (left, right) = if left.dtype().is_nested() {
575                                    (
576                                        ChunkedArray::<BinaryOffsetType>::with_chunk(
577                                            "".into(),
578                                            row_encode::_get_rows_encoded_unordered(&[
579                                                left.clone()
580                                            ])?
581                                            .into_array(),
582                                        )
583                                        .into_series(),
584                                        ChunkedArray::<BinaryOffsetType>::with_chunk(
585                                            "".into(),
586                                            row_encode::_get_rows_encoded_unordered(&[
587                                                right.clone()
588                                            ])?
589                                            .into_array(),
590                                        )
591                                        .into_series(),
592                                    )
593                                } else {
594                                    (
595                                        left.into_materialized_series().clone(),
596                                        right.into_materialized_series().clone(),
597                                    )
598                                };
599
600                                PolarsResult::Ok(Arc::new(
601                                    left.hash_join_left(&right, JoinValidation::ManyToMany, true)
602                                        .unwrap()
603                                        .1,
604                                ))
605                            } else {
606                                let df_right =
607                                    unsafe { DataFrame::new_no_checks_height_from_first(keys) };
608                                let df_left = unsafe {
609                                    DataFrame::new_no_checks_height_from_first(group_by_columns)
610                                };
611                                Ok(Arc::new(
612                                    private_left_join_multiple_keys(&df_left, &df_right, true)?.1,
613                                ))
614                            }
615                        };
616
617                        // try to get cached join_tuples
618                        let join_opt_ids = if state.cache_window() {
619                            if let Some(jt) = state.window_cache.get_join(&cache_key) {
620                                jt
621                            } else {
622                                let jt = get_join_tuples()?;
623                                state
624                                    .window_cache
625                                    .insert_join(cache_key.clone(), jt.clone());
626                                jt
627                            }
628                        } else {
629                            get_join_tuples()?
630                        };
631
632                        let mut out = materialize_column(&join_opt_ids, &out_column);
633
634                        if let Some(name) = &self.out_name {
635                            out.rename(name.clone());
636                        }
637
638                        Ok(out.into_column())
639                    },
640                }
641            },
642        }
643    }

678fn materialize_column(join_opt_ids: &ChunkJoinOptIds, out_column: &Column) -> Column {
679    {
680        use arrow::Either;
681        use polars_ops::chunked_array::TakeChunked;
682
683        match join_opt_ids {
684            Either::Left(ids) => unsafe {
685                IdxCa::with_nullable_idx(ids, |idx| out_column.take_unchecked(idx))
686            },
687            Either::Right(ids) => unsafe { out_column.take_opt_chunked_unchecked(ids) },
688        }
689    }
690}
691
692/// Simple reducing aggregation can be set by the groups
693fn set_by_groups(
694    s: &Column,
695    groups: &GroupsType,
696    len: usize,
697    update_groups: bool,
698) -> Option<Column> {
699    if update_groups {
700        return None;
701    }
702    if s.dtype().to_physical().is_primitive_numeric() {
703        let dtype = s.dtype();
704        let s = s.to_physical_repr();
705
706        macro_rules! dispatch {
707            ($ca:expr) => {{
708                Some(set_numeric($ca, groups, len))
709            }};
710        }
711        downcast_as_macro_arg_physical!(&s, dispatch)
712            .map(|s| s.cast(dtype).unwrap())
713            .map(Column::from)
714    } else {
715        None
716    }
717}
718
719fn set_numeric<T>(ca: &ChunkedArray<T>, groups: &GroupsType, len: usize) -> Series
720where
721    T: PolarsNumericType,
722    ChunkedArray<T>: IntoSeries,
723{
724    let mut values = Vec::with_capacity(len);
725    let ptr: *mut T::Native = values.as_mut_ptr();
726    // SAFETY:
727    // we will write from different threads but we will never alias.
728    let sync_ptr_values = unsafe { SyncPtr::new(ptr) };
729
730    if ca.null_count() == 0 {
731        let ca = ca.rechunk();
732        match groups {
733            GroupsType::Idx(groups) => {
734                let agg_vals = ca.cont_slice().expect("rechunked");
735                POOL.install(|| {
736                    agg_vals
737                        .par_iter()
738                        .zip(groups.all().par_iter())
739                        .for_each(|(v, g)| {
740                            let ptr = sync_ptr_values.get();
741                            for idx in g.as_slice() {
742                                debug_assert!((*idx as usize) < len);
743                                unsafe { *ptr.add(*idx as usize) = *v }
744                            }
745                        })
746                })
747            },
748            GroupsType::Slice { groups, .. } => {
749                let agg_vals = ca.cont_slice().expect("rechunked");
750                POOL.install(|| {
751                    agg_vals
752                        .par_iter()
753                        .zip(groups.par_iter())
754                        .for_each(|(v, [start, g_len])| {
755                            let ptr = sync_ptr_values.get();
756                            let start = *start as usize;
757                            let end = start + *g_len as usize;
758                            for idx in start..end {
759                                debug_assert!(idx < len);
760                                unsafe { *ptr.add(idx) = *v }
761                            }
762                        })
763                });
764            },
765        }
766
767        // SAFETY: we have written all slots
768        unsafe { values.set_len(len) }
769        ChunkedArray::new_vec(ca.name().clone(), values).into_series()
770    } else {
771        // We don't use a mutable bitmap as bits will have race conditions!
772        // A single byte might alias if we write from single threads.
773        let mut validity: Vec<bool> = vec![false; len];
774        let validity_ptr = validity.as_mut_ptr();
775        let sync_ptr_validity = unsafe { SyncPtr::new(validity_ptr) };
776
777        let n_threads = POOL.current_num_threads();
778        let offsets = _split_offsets(ca.len(), n_threads);
779
780        match groups {
781            GroupsType::Idx(groups) => offsets.par_iter().for_each(|(offset, offset_len)| {
782                let offset = *offset;
783                let offset_len = *offset_len;
784                let ca = ca.slice(offset as i64, offset_len);
785                let groups = &groups.all()[offset..offset + offset_len];
786                let values_ptr = sync_ptr_values.get();
787                let validity_ptr = sync_ptr_validity.get();
788
789                ca.iter().zip(groups.iter()).for_each(|(opt_v, g)| {
790                    for idx in g.as_slice() {
791                        let idx = *idx as usize;
792                        debug_assert!(idx < len);
793                        unsafe {
794                            match opt_v {
795                                Some(v) => {
796                                    *values_ptr.add(idx) = v;
797                                    *validity_ptr.add(idx) = true;
798                                },
799                                None => {
800                                    *values_ptr.add(idx) = T::Native::default();
801                                    *validity_ptr.add(idx) = false;
802                                },
803                            };
804                        }
805                    }
806                })
807            }),
808            GroupsType::Slice { groups, .. } => {
809                offsets.par_iter().for_each(|(offset, offset_len)| {
810                    let offset = *offset;
811                    let offset_len = *offset_len;
812                    let ca = ca.slice(offset as i64, offset_len);
813                    let groups = &groups[offset..offset + offset_len];
814                    let values_ptr = sync_ptr_values.get();
815                    let validity_ptr = sync_ptr_validity.get();
816
817                    for (opt_v, [start, g_len]) in ca.iter().zip(groups.iter()) {
818                        let start = *start as usize;
819                        let end = start + *g_len as usize;
820                        for idx in start..end {
821                            debug_assert!(idx < len);
822                            unsafe {
823                                match opt_v {
824                                    Some(v) => {
825                                        *values_ptr.add(idx) = v;
826                                        *validity_ptr.add(idx) = true;
827                                    },
828                                    None => {
829                                        *values_ptr.add(idx) = T::Native::default();
830                                        *validity_ptr.add(idx) = false;
831                                    },
832                                };
833                            }
834                        }
835                    }
836                })
837            },
838        }
839        // SAFETY: we have written all slots
840        unsafe { values.set_len(len) }
841        let validity = Bitmap::from(validity);
842        let arr = PrimitiveArray::new(
843            T::get_dtype().to_physical().to_arrow(CompatLevel::newest()),
844            values.into(),
845            Some(validity),
846        );
847        Series::try_from((ca.name().clone(), arr.boxed())).unwrap()
848    }
849}

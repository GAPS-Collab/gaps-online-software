8/// Implements the
9/// [Log-normal](https://en.wikipedia.org/wiki/Log-normal_distribution)
10/// distribution
11///
12/// # Examples
13///
14/// ```
15/// use statrs::distribution::{LogNormal, Continuous};
16/// use statrs::statistics::Distribution;
17/// use statrs::prec;
18///
19/// let n = LogNormal::new(0.0, 1.0).unwrap();
20/// assert_eq!(n.mean().unwrap(), (0.5f64).exp());
21/// assert!(prec::almost_eq(n.pdf(1.0), 0.3989422804014326779399, 1e-16));
22/// ```
23#[derive(Debug, Copy, Clone, PartialEq)]
24pub struct LogNormal {
25    location: f64,
26    scale: f64,
27}
28
29impl LogNormal {
30    /// Constructs a new log-normal distribution with a location of `location`
31    /// and a scale of `scale`
32    ///
33    /// # Errors
34    ///
35    /// Returns an error if `location` or `scale` are `NaN`.
36    /// Returns an error if `scale <= 0.0`
37    ///
38    /// # Examples
39    ///
40    /// ```
41    /// use statrs::distribution::LogNormal;
42    ///
43    /// let mut result = LogNormal::new(0.0, 1.0);
44    /// assert!(result.is_ok());
45    ///
46    /// result = LogNormal::new(0.0, 0.0);
47    /// assert!(result.is_err());
48    /// ```
49    pub fn new(location: f64, scale: f64) -> Result<LogNormal> {
50        if location.is_nan() || scale.is_nan() || scale <= 0.0 {
51            Err(StatsError::BadParams)
52        } else {
53            Ok(LogNormal { location, scale })
54        }
55    }
56}

59    fn sample<R: Rng + ?Sized>(&self, rng: &mut R) -> f64 {
60        super::normal::sample_unchecked(rng, self.location, self.scale).exp()
61    }

64impl ContinuousCDF<f64, f64> for LogNormal {
65    /// Calculates the cumulative distribution function for the log-normal
66    /// distribution
67    /// at `x`
68    ///
69    /// # Formula
70    ///
71    /// ```ignore
72    /// (1 / 2) + (1 / 2) * erf((ln(x) - μ) / sqrt(2) * σ)
73    /// ```
74    ///
75    /// where `μ` is the location, `σ` is the scale, and `erf` is the
76    /// error function
77    fn cdf(&self, x: f64) -> f64 {
78        if x <= 0.0 {
79            0.0
80        } else if x.is_infinite() {
81            1.0
82        } else {
83            0.5 * erf::erfc((self.location - x.ln()) / (self.scale * f64::consts::SQRT_2))
84        }
85    }
86
87    /// Calculates the survival function for the log-normal
88    /// distribution at `x`
89    ///
90    /// # Formula
91    ///
92    /// ```ignore
93    /// (1 / 2) + (1 / 2) * erf(-(ln(x) - μ) / sqrt(2) * σ)
94    /// ```
95    ///
96    /// where `μ` is the location, `σ` is the scale, and `erf` is the
97    /// error function
98    ///
99    /// note that this calculates the complement due to flipping
100    /// the sign of the argument error function with respect to the cdf.
101    ///
102    /// the normal cdf Φ (and internal error function) as the following property:
103    /// ```ignore
104    ///  Φ(-x) + Φ(x) = 1
105    ///  Φ(-x)        = 1 - Φ(x) 
106    /// ```
107    fn sf(&self, x: f64) -> f64 {
108        if x <= 0.0 {
109            1.0
110        } else if x.is_infinite() {
111            0.0
112        } else {
113            0.5 * erf::erfc((x.ln() - self.location) / (self.scale * f64::consts::SQRT_2))
114        }
115    }
116}

146impl Distribution<f64> for LogNormal {
147    /// Returns the mean of the log-normal distribution
148    ///
149    /// # Formula
150    ///
151    /// ```ignore
152    /// e^(μ + σ^2 / 2)
153    /// ```
154    ///
155    /// where `μ` is the location and `σ` is the scale
156    fn mean(&self) -> Option<f64> {
157        Some((self.location + self.scale * self.scale / 2.0).exp())
158    }
159    /// Returns the variance of the log-normal distribution
160    ///
161    /// # Formula
162    ///
163    /// ```ignore
164    /// (e^(σ^2) - 1) * e^(2μ + σ^2)
165    /// ```
166    ///
167    /// where `μ` is the location and `σ` is the scale
168    fn variance(&self) -> Option<f64> {
169        let sigma2 = self.scale * self.scale;
170        Some((sigma2.exp() - 1.0) * (self.location + self.location + sigma2).exp())
171    }
172    /// Returns the entropy of the log-normal distribution
173    ///
174    /// # Formula
175    ///
176    /// ```ignore
177    /// ln(σe^(μ + 1 / 2) * sqrt(2π))
178    /// ```
179    ///
180    /// where `μ` is the location and `σ` is the scale
181    fn entropy(&self) -> Option<f64> {
182        Some(0.5 + self.scale.ln() + self.location + consts::LN_SQRT_2PI)
183    }
184    /// Returns the skewness of the log-normal distribution
185    ///
186    /// # Formula
187    ///
188    /// ```ignore
189    /// (e^(σ^2) + 2) * sqrt(e^(σ^2) - 1)
190    /// ```
191    ///
192    /// where `μ` is the location and `σ` is the scale
193    fn skewness(&self) -> Option<f64> {
194        let expsigma2 = (self.scale * self.scale).exp();
195        Some((expsigma2 + 2.0) * (expsigma2 - 1.0).sqrt())
196    }
197}

214impl Mode<Option<f64>> for LogNormal {
215    /// Returns the mode of the log-normal distribution
216    ///
217    /// # Formula
218    ///
219    /// ```ignore
220    /// e^(μ - σ^2)
221    /// ```
222    ///
223    /// where `μ` is the location and `σ` is the scale
224    fn mode(&self) -> Option<f64> {
225        Some((self.location - self.scale * self.scale).exp())

229impl Continuous<f64, f64> for LogNormal {
230    /// Calculates the probability density function for the log-normal
231    /// distribution at `x`
232    ///
233    /// # Formula
234    ///
235    /// ```ignore
236    /// (1 / xσ * sqrt(2π)) * e^(-((ln(x) - μ)^2) / 2σ^2)
237    /// ```
238    ///
239    /// where `μ` is the location and `σ` is the scale
240    fn pdf(&self, x: f64) -> f64 {
241        if x <= 0.0 || x.is_infinite() {
242            0.0
243        } else {
244            let d = (x.ln() - self.location) / self.scale;
245            (-0.5 * d * d).exp() / (x * consts::SQRT_2PI * self.scale)
246        }
247    }
248
249    /// Calculates the log probability density function for the log-normal
250    /// distribution at `x`
251    ///
252    /// # Formula
253    ///
254    /// ```ignore
255    /// ln((1 / xσ * sqrt(2π)) * e^(-((ln(x) - μ)^2) / 2σ^2))
256    /// ```
257    ///
258    /// where `μ` is the location and `σ` is the scale
259    fn ln_pdf(&self, x: f64) -> f64 {
260        if x <= 0.0 || x.is_infinite() {
261            f64::NEG_INFINITY
262        } else {
263            let d = (x.ln() - self.location) / self.scale;
264            (-0.5 * d * d) - consts::LN_SQRT_2PI - (x * self.scale).ln()
265        }
266    }
267}

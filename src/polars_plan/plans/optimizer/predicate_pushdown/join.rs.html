4// Information concerning individual sides of a join.
5#[derive(PartialEq, Eq)]
6struct LeftRight<T>(T, T);
7
8fn should_block_join_specific(
9    ae: &AExpr,
10    how: &JoinType,
11    on_names: &PlHashSet<PlSmallStr>,
12    expr_arena: &Arena<AExpr>,
13    schema_left: &Schema,
14    schema_right: &Schema,
15) -> LeftRight<bool> {
16    use AExpr::*;
17    match ae {
18        // joins can produce null values
19        Function {
20            function:
21                FunctionExpr::Boolean(BooleanFunction::IsNotNull)
22                | FunctionExpr::Boolean(BooleanFunction::IsNull)
23                | FunctionExpr::FillNull { .. },
24            ..
25        } => join_produces_null(how),
26        #[cfg(feature = "is_in")]
27        Function {
28            function: FunctionExpr::Boolean(BooleanFunction::IsIn),
29            ..
30        } => join_produces_null(how),
31        // joins can produce duplicates
32        #[cfg(feature = "is_unique")]
33        Function {
34            function:
35                FunctionExpr::Boolean(BooleanFunction::IsUnique)
36                | FunctionExpr::Boolean(BooleanFunction::IsDuplicated),
37            ..
38        } => LeftRight(true, true),
39        #[cfg(feature = "is_first_distinct")]
40        Function {
41            function: FunctionExpr::Boolean(BooleanFunction::IsFirstDistinct),
42            ..
43        } => LeftRight(true, true),
44        // any operation that checks for equality or ordering can be wrong because
45        // the join can produce null values
46        // TODO! check if we can be less conservative here
47        BinaryExpr {
48            op: Operator::Eq | Operator::NotEq,
49            left,
50            right,
51        } => {
52            let LeftRight(bleft, bright) = join_produces_null(how);
53
54            let l_name = aexpr_output_name(*left, expr_arena).unwrap();
55            let r_name = aexpr_output_name(*right, expr_arena).unwrap();
56
57            let is_in_on = on_names.contains(&l_name) || on_names.contains(&r_name);
58
59            let block_left =
60                is_in_on && (schema_left.contains(&l_name) || schema_left.contains(&r_name));
61            let block_right =
62                is_in_on && (schema_right.contains(&l_name) || schema_right.contains(&r_name));
63            LeftRight(block_left | bleft, block_right | bright)
64        },
65        _ => join_produces_null(how),
66    }
67}

69/// Returns a tuple indicating whether predicates should be blocked for either side based on the
70/// join type.
71///
72/// * `true` indicates that predicates must not be pushed to that side
73fn join_produces_null(how: &JoinType) -> LeftRight<bool> {
74    match how {
75        JoinType::Left => LeftRight(false, true),
76        JoinType::Right => LeftRight(true, false),
77
78        JoinType::Full { .. } => LeftRight(true, true),
79        JoinType::Cross => LeftRight(true, true),
80        #[cfg(feature = "asof_join")]
81        JoinType::AsOf(_) => LeftRight(true, true),
82
83        JoinType::Inner => LeftRight(false, false),
84        #[cfg(feature = "semi_anti_join")]
85        JoinType::Semi | JoinType::Anti => LeftRight(false, false),
86        #[cfg(feature = "iejoin")]
87        JoinType::IEJoin => LeftRight(false, false),
88    }
89}

91fn all_pred_cols_in_left_on(
92    predicate: &ExprIR,
93    expr_arena: &mut Arena<AExpr>,
94    left_on: &[ExprIR],
95) -> bool {
96    aexpr_to_leaf_names_iter(predicate.node(), expr_arena)
97        .all(|pred_column_name| left_on.iter().any(|e| e.output_name() == &pred_column_name))
98}

100// Checks if a predicate refers to columns in both tables
101fn predicate_applies_to_both_tables(
102    predicate: Node,
103    expr_arena: &Arena<AExpr>,
104    schema_left: &Schema,
105    schema_right: &Schema,
106    suffix: &str,
107) -> bool {
108    let mut left_used = false;
109    let mut right_used = false;
110    for name in aexpr_to_leaf_names_iter(predicate, expr_arena) {
111        if schema_left.contains(name.as_ref()) {
112            left_used |= true;
113        } else {
114            right_used |= schema_right.contains(name.as_ref())
115                || name.ends_with(suffix)
116                    && schema_right.contains(split_suffix(name.as_ref(), suffix))
117        }
118    }
119    left_used && right_used
120}

122#[allow(clippy::too_many_arguments)]
123pub(super) fn process_join(
124    opt: &mut PredicatePushDown,
125    lp_arena: &mut Arena<IR>,
126    expr_arena: &mut Arena<AExpr>,
127    input_left: Node,
128    input_right: Node,
129    left_on: Vec<ExprIR>,
130    right_on: Vec<ExprIR>,
131    schema: SchemaRef,
132    options: Arc<JoinOptions>,
133    acc_predicates: PlHashMap<PlSmallStr, ExprIR>,
134) -> PolarsResult<IR> {
135    use IR::*;
136    let schema_left = lp_arena.get(input_left).schema(lp_arena);
137    let schema_right = lp_arena.get(input_right).schema(lp_arena);
138
139    let on_names = left_on
140        .iter()
141        .flat_map(|e| aexpr_to_leaf_names_iter(e.node(), expr_arena))
142        .chain(
143            right_on
144                .iter()
145                .flat_map(|e| aexpr_to_leaf_names_iter(e.node(), expr_arena)),
146        )
147        .collect::<PlHashSet<_>>();
148
149    let mut pushdown_left = init_hashmap(Some(acc_predicates.len()));
150    let mut pushdown_right = init_hashmap(Some(acc_predicates.len()));
151    let mut local_predicates = Vec::with_capacity(acc_predicates.len());
152
153    for (_, predicate) in acc_predicates {
154        // Cross joins produce a cartesian product, so if a predicate combines columns from both tables, we should not push down.
155        // Inequality joins logically produce a cartesian product, so the same logic applies.
156        if (options.args.how.is_cross() || options.args.how.is_ie())
157            && predicate_applies_to_both_tables(
158                predicate.node(),
159                expr_arena,
160                &schema_left,
161                &schema_right,
162                options.args.suffix(),
163            )
164        {
165            local_predicates.push(predicate);
166            continue;
167        }
168
169        // check if predicate can pass the joins node
170        let block_pushdown_left = has_aexpr(predicate.node(), expr_arena, |ae| {
171            should_block_join_specific(
172                ae,
173                &options.args.how,
174                &on_names,
175                expr_arena,
176                &schema_left,
177                &schema_right,
178            )
179            .0
180        });
181        let block_pushdown_right = has_aexpr(predicate.node(), expr_arena, |ae| {
182            should_block_join_specific(
183                ae,
184                &options.args.how,
185                &on_names,
186                expr_arena,
187                &schema_left,
188                &schema_right,
189            )
190            .1
191        });
192
193        // these indicate to which tables we are going to push down the predicate
194        let mut filter_left = false;
195        let mut filter_right = false;
196
197        if !block_pushdown_left && check_input_node(predicate.node(), &schema_left, expr_arena) {
198            insert_and_combine_predicate(&mut pushdown_left, &predicate, expr_arena);
199            filter_left = true;
200            // If we push down to the left and all predicate columns are also
201            // join columns, we also push down right for inner, left or semi join
202            if all_pred_cols_in_left_on(&predicate, expr_arena, &left_on) {
203                filter_right = match &options.args.how {
204                    // TODO! if join_on right has a different name
205                    // we can set this to `true` IFF we rename the predicate
206                    JoinType::Inner | JoinType::Left => {
207                        check_input_node(predicate.node(), &schema_right, expr_arena)
208                    },
209                    #[cfg(feature = "semi_anti_join")]
210                    JoinType::Semi => check_input_node(predicate.node(), &schema_right, expr_arena),
211                    _ => false,
212                }
213            }
214        // this is `else if` because if the predicate is in the left hand side
215        // the right hand side should be renamed with the suffix.
216        // in that case we should not push down as the user wants to filter on `x`
217        // not on `x_rhs`.
218        } else if !block_pushdown_right
219            && check_input_node(predicate.node(), &schema_right, expr_arena)
220        {
221            filter_right = true
222        }
223        if filter_right {
224            insert_and_combine_predicate(&mut pushdown_right, &predicate, expr_arena);
225        }
226
227        match (filter_left, filter_right, &options.args.how) {
228            // if not pushed down on one of the tables we have to do it locally.
229            (false, false, _) |
230            // if left join and predicate only available in right table,
231            // 'we should not filter right, because that would lead to
232            // invalid results.
233            // see: #2057
234            (false, true, JoinType::Left)
235            => {
236                local_predicates.push(predicate);
237                continue;
238            },
239            // business as usual
240            _ => {}
241        }
242    }
243
244    opt.pushdown_and_assign(input_left, pushdown_left, lp_arena, expr_arena)?;
245    opt.pushdown_and_assign(input_right, pushdown_right, lp_arena, expr_arena)?;
246
247    let lp = Join {
248        input_left,
249        input_right,
250        left_on,
251        right_on,
252        schema,
253        options,
254    };
255
256    Ok(opt.optional_apply_predicate(lp, local_predicates, lp_arena, expr_arena))
257}

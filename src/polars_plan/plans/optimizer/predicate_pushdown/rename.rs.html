6fn remove_any_key_referencing_renamed(
7    new: &str,
8    acc_predicates: &mut PlHashMap<PlSmallStr, ExprIR>,
9    local_predicates: &mut Vec<ExprIR>,
10) {
11    let mut move_to_local = vec![];
12    for key in acc_predicates.keys() {
13        if key_has_name(key, new) {
14            move_to_local.push(key.clone())
15        }
16    }
17
18    for key in move_to_local {
19        local_predicates.push(acc_predicates.remove(&key).unwrap())
20    }
21}

23pub(super) fn process_rename(
24    acc_predicates: &mut PlHashMap<PlSmallStr, ExprIR>,
25    expr_arena: &mut Arena<AExpr>,
26    existing: &[PlSmallStr],
27    new: &[PlSmallStr],
28) -> PolarsResult<Vec<ExprIR>> {
29    let mut local_predicates = vec![];
30    for (existing, new) in existing.iter().zip(new.iter()) {
31        let has_existing = acc_predicates.contains_key(existing.as_str());
32        let has_new = acc_predicates.contains_key(new.as_str());
33        let has_both = has_existing && has_new;
34
35        // swapping path add to local for now
36        if has_both {
37            // Search for the key and add it to local because swapping is more complicated
38            if let Some(to_local) = acc_predicates.remove(new.as_str()) {
39                local_predicates.push(to_local);
40            } else {
41                // The keys can be combined eg. `a AND b AND c` in this case replacing/finding
42                // the key that should be renamed is more complicated, so for now
43                // we just move it to local.
44                remove_any_key_referencing_renamed(new, acc_predicates, &mut local_predicates)
45            }
46            continue;
47        }
48        // simple new name path
49        else {
50            // Find the key and update the predicate as well as the key
51            // This ensure the optimization is pushed down.
52            if let Some(mut e) = acc_predicates.remove(new.as_str()) {
53                let new_node =
54                    rename_matching_aexpr_leaf_names(e.node(), expr_arena, new, existing.clone());
55                e.set_node(new_node);
56                acc_predicates.insert(predicate_to_key(new_node, expr_arena), e);
57            } else {
58                // The keys can be combined eg. `a AND b AND c` in this case replacing/finding
59                // the key that should be renamed is more complicated, so for now
60                // we just move it to local.
61                remove_any_key_referencing_renamed(new, acc_predicates, &mut local_predicates)
62            }
63        }
64    }
65    Ok(local_predicates)
66}

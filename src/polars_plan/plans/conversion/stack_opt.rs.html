1use std::borrow::Borrow;
2
3use self::type_check::TypeCheckRule;
4use super::*;
5
6/// Applies expression simplification and type coercion during conversion to IR.
7pub(super) struct ConversionOptimizer {
8    scratch: Vec<Node>,
9
10    simplify: Option<SimplifyExprRule>,
11    coerce: Option<TypeCoercionRule>,
12    check: Option<TypeCheckRule>,
13    // IR's can be cached in the DSL.
14    // But if they are used multiple times in DSL (e.g. concat/join)
15    // then it can occur that we take a slot multiple times.
16    // So we keep track of the arena versions used and allow only
17    // one unique IR cache to be reused.
18    pub(super) used_arenas: PlHashSet<u32>,
19}
20
21impl ConversionOptimizer {
22    pub(super) fn new(simplify: bool, type_coercion: bool, type_check: bool) -> Self {
23        let simplify = if simplify {
24            Some(SimplifyExprRule {})
25        } else {
26            None
27        };
28
29        let coerce = if type_coercion {
30            Some(TypeCoercionRule {})
31        } else {
32            None
33        };
34
35        let check = if type_check {
36            Some(TypeCheckRule)
37        } else {
38            None
39        };
40
41        ConversionOptimizer {
42            scratch: Vec::with_capacity(8),
43            simplify,
44            coerce,
45            check,
46            used_arenas: Default::default(),
47        }
48    }
49
50    pub(super) fn push_scratch(&mut self, expr: Node, expr_arena: &Arena<AExpr>) {
51        self.scratch.push(expr);
52        // traverse all subexpressions and add to the stack
53        let expr = unsafe { expr_arena.get_unchecked(expr) };
54        expr.inputs_rev(&mut self.scratch);
55    }
56
57    pub(super) fn fill_scratch<N: Borrow<Node>>(&mut self, exprs: &[N], expr_arena: &Arena<AExpr>) {
58        for e in exprs {
59            let node = *e.borrow();
60            self.push_scratch(node, expr_arena);
61        }
62    }
63
64    pub(super) fn coerce_types(
65        &mut self,
66        expr_arena: &mut Arena<AExpr>,
67        ir_arena: &mut Arena<IR>,
68        current_node: Node,
69    ) -> PolarsResult<()> {
70        // Different from the stack-opt in the optimizer phase, this does a single pass until fixed point per expression.
71
72        if let Some(rule) = &mut self.check {
73            while let Some(x) = rule.optimize_plan(ir_arena, expr_arena, current_node)? {
74                ir_arena.replace(current_node, x);
75            }
76        }
77
78        // process the expressions on the stack and apply optimizations.
79        while let Some(current_expr_node) = self.scratch.pop() {
80            let expr = unsafe { expr_arena.get_unchecked(current_expr_node) };
81
82            if expr.is_leaf() {
83                continue;
84            }
85
86            if let Some(rule) = &mut self.simplify {
87                while let Some(x) =
88                    rule.optimize_expr(expr_arena, current_expr_node, ir_arena, current_node)?
89                {
90                    expr_arena.replace(current_expr_node, x);
91                }
92            }
93            if let Some(rule) = &mut self.coerce {
94                while let Some(x) =
95                    rule.optimize_expr(expr_arena, current_expr_node, ir_arena, current_node)?
96                {
97                    expr_arena.replace(current_expr_node, x);
98                }
99            }
100
101            let expr = unsafe { expr_arena.get_unchecked(current_expr_node) };
102            // traverse subexpressions and add to the stack
103            expr.inputs_rev(&mut self.scratch)
104        }
105
106        Ok(())
107    }
108}

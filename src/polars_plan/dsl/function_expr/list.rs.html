91            ArgMax => mapper.with_dtype(IDX_DTYPE),
92            #[cfg(feature = "diff")]

115fn map_list_dtype_to_array_dtype(datatype: &DataType, width: usize) -> PolarsResult<DataType> {
116    if let DataType::List(inner) = datatype {

119        polars_bail!(ComputeError: "expected List dtype")

212            Get(null_on_oob) => wrap!(get, null_on_oob),

215            #[cfg(feature = "list_gather")]

217            #[cfg(feature = "list_count")]

219            Sum => map!(sum),
220            Length => map!(length),

222            Min => map!(min),

224            Median => map!(median),

230            Diff { n, null_behavior } => map!(diff, n, null_behavior),

236            #[cfg(feature = "list_any_all")]

261        },

319pub(super) fn slice(args: &mut [Column]) -> PolarsResult<Option<Column>> {
320    let s = &args[0];
321    let list_ca = s.list()?;
322    let offset_s = &args[1];
323    let length_s = &args[2];
324
325    let mut out: ListChunked = match (offset_s.len(), length_s.len()) {
326        (1, 1) => {
327            let offset = offset_s.get(0).unwrap().try_extract::<i64>()?;
328            let slice_len = length_s
329                .get(0)
330                .unwrap()
331                .extract::<usize>()
332                .unwrap_or(usize::MAX);
333            return Ok(Some(list_ca.lst_slice(offset, slice_len).into_column()));
334        },
335        (1, length_slice_len) => {
336            check_slice_arg_shape(length_slice_len, list_ca.len(), "length")?;
337            let offset = offset_s.get(0).unwrap().try_extract::<i64>()?;
338            // cast to i64 as it is more likely that it is that dtype
339            // instead of usize/u64 (we never need that max length)
340            let length_ca = length_s.cast(&DataType::Int64)?;
341            let length_ca = length_ca.i64().unwrap();
342
343            list_ca
344                .amortized_iter()
345                .zip(length_ca)
346                .map(|(opt_s, opt_length)| match (opt_s, opt_length) {
347                    (Some(s), Some(length)) => Some(s.as_ref().slice(offset, length as usize)),
348                    _ => None,
349                })
350                .collect_trusted()
351        },
352        (offset_len, 1) => {
353            check_slice_arg_shape(offset_len, list_ca.len(), "offset")?;
354            let length_slice = length_s
355                .get(0)
356                .unwrap()
357                .extract::<usize>()
358                .unwrap_or(usize::MAX);
359            let offset_ca = offset_s.cast(&DataType::Int64)?;
360            let offset_ca = offset_ca.i64().unwrap();
361            list_ca
362                .amortized_iter()
363                .zip(offset_ca)
364                .map(|(opt_s, opt_offset)| match (opt_s, opt_offset) {
365                    (Some(s), Some(offset)) => Some(s.as_ref().slice(offset, length_slice)),
366                    _ => None,
367                })
368                .collect_trusted()
369        },
370        _ => {
371            check_slice_arg_shape(offset_s.len(), list_ca.len(), "offset")?;
372            check_slice_arg_shape(length_s.len(), list_ca.len(), "length")?;
373            let offset_ca = offset_s.cast(&DataType::Int64)?;
374            let offset_ca = offset_ca.i64()?;
375            // cast to i64 as it is more likely that it is that dtype
376            // instead of usize/u64 (we never need that max length)
377            let length_ca = length_s.cast(&DataType::Int64)?;
378            let length_ca = length_ca.i64().unwrap();
379
380            list_ca
381                .amortized_iter()
382                .zip(offset_ca)
383                .zip(length_ca)
384                .map(
385                    |((opt_s, opt_offset), opt_length)| match (opt_s, opt_offset, opt_length) {
386                        (Some(s), Some(offset), Some(length)) => {
387                            Some(s.as_ref().slice(offset, length as usize))
388                        },
389                        _ => None,
390                    },
391                )
392                .collect_trusted()
393        },
394    };
395    out.rename(s.name().clone());
396    Ok(Some(out.into_column()))
397}

399pub(super) fn concat(s: &mut [Column]) -> PolarsResult<Option<Column>> {
400    let mut first = std::mem::take(&mut s[0]);
401    let other = &s[1..];
402
403    let mut first_ca = match first.try_list() {
404        Some(ca) => ca,
405        None => {
406            first = first
407                .reshape_list(&[ReshapeDimension::Infer, ReshapeDimension::new_dimension(1)])
408                .unwrap();
409            first.list().unwrap()
410        },
411    }
412    .clone();
413
414    if first_ca.len() == 1 && !other.is_empty() {
415        let max_len = other.iter().map(|s| s.len()).max().unwrap();
416        if max_len > 1 {
417            first_ca = first_ca.new_from_index(0, max_len)
418        }
419    }
420
421    first_ca.lst_concat(other).map(|ca| Some(ca.into_column()))
422}

424pub(super) fn get(s: &mut [Column], null_on_oob: bool) -> PolarsResult<Option<Column>> {
425    let ca = s[0].list()?;
426    let index = s[1].cast(&DataType::Int64)?;
427    let index = index.i64().unwrap();
428
429    match index.len() {
430        1 => {
431            let index = index.get(0);
432            if let Some(index) = index {
433                ca.lst_get(index, null_on_oob).map(Column::from).map(Some)
434            } else {
435                Ok(Some(Column::full_null(
436                    ca.name().clone(),
437                    ca.len(),
438                    ca.inner_dtype(),
439                )))
440            }
441        },
442        len if len == ca.len() => {
443            let ca = ca.rechunk();
444            let arr = ca.downcast_iter().next().unwrap();
445            let offsets = arr.offsets().as_slice();
446            let take_by = if ca.null_count() == 0 {
447                index
448                    .iter()
449                    .enumerate()
450                    .map(|(i, opt_idx)| match opt_idx {
451                        Some(idx) => {
452                            let (start, end) = unsafe {
453                                (*offsets.get_unchecked(i), *offsets.get_unchecked(i + 1))
454                            };
455                            let offset = if idx >= 0 { start + idx } else { end + idx };
456                            if offset >= end || offset < start || start == end {
457                                if null_on_oob {
458                                    Ok(None)
459                                } else {
460                                    polars_bail!(ComputeError: "get index is out of bounds");
461                                }
462                            } else {
463                                Ok(Some(offset as IdxSize))
464                            }
465                        },
466                        None => Ok(None),
467                    })
468                    .collect::<Result<IdxCa, _>>()?
469            } else {
470                index
471                    .iter()
472                    .zip(arr.validity().unwrap())
473                    .enumerate()
474                    .map(|(i, (opt_idx, valid))| match (valid, opt_idx) {
475                        (true, Some(idx)) => {
476                            let (start, end) = unsafe {
477                                (*offsets.get_unchecked(i), *offsets.get_unchecked(i + 1))
478                            };
479                            let offset = if idx >= 0 { start + idx } else { end + idx };
480                            if offset >= end || offset < start || start == end {
481                                if null_on_oob {
482                                    Ok(None)
483                                } else {
484                                    polars_bail!(ComputeError: "get index is out of bounds");
485                                }
486                            } else {
487                                Ok(Some(offset as IdxSize))
488                            }
489                        },
490                        _ => Ok(None),
491                    })
492                    .collect::<Result<IdxCa, _>>()?
493            };
494            let s = Series::try_from((ca.name().clone(), arr.values().clone())).unwrap();
495            unsafe { s.take_unchecked(&take_by) }
496                .cast(ca.inner_dtype())
497                .map(Column::from)
498                .map(Some)
499        },
500        len => polars_bail!(
501            ComputeError:
502            "`list.get` expression got an index array of length {} while the list has {} elements",
503            len, ca.len()
504        ),
505    }
506}

508#[cfg(feature = "list_gather")]
509pub(super) fn gather(args: &[Column], null_on_oob: bool) -> PolarsResult<Column> {
510    let ca = &args[0];
511    let idx = &args[1];
512    let ca = ca.list()?;
513
514    if idx.len() == 1 && idx.dtype().is_primitive_numeric() && null_on_oob {
515        // fast path
516        let idx = idx.get(0)?.try_extract::<i64>()?;
517        let out = ca.lst_get(idx, null_on_oob).map(Column::from)?;
518        // make sure we return a list
519        out.reshape_list(&[ReshapeDimension::Infer, ReshapeDimension::new_dimension(1)])
520    } else {
521        ca.lst_gather(idx.as_materialized_series(), null_on_oob)
522            .map(Column::from)
523    }
524}

611#[cfg(feature = "list_sets")]
612pub(super) fn set_operation(s: &[Column], set_type: SetOperation) -> PolarsResult<Column> {
613    let s0 = &s[0];
614    let s1 = &s[1];
615
616    if s0.is_empty() || s1.is_empty() {
617        return match set_type {
618            SetOperation::Intersection => {
619                if s0.is_empty() {
620                    Ok(s0.clone())
621                } else {
622                    Ok(s1.clone().with_name(s0.name().clone()))
623                }
624            },
625            SetOperation::Difference => Ok(s0.clone()),
626            SetOperation::Union | SetOperation::SymmetricDifference => {
627                if s0.is_empty() {
628                    Ok(s1.clone().with_name(s0.name().clone()))
629                } else {
630                    Ok(s0.clone())
631                }
632            },
633        };
634    }
635
636    list_set_operation(s0.list()?, s1.list()?, set_type).map(|ca| ca.into_column())
637}

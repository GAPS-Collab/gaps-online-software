45/// Get all filenames in the current path sorted by timestamp if available
46/// If the given path is a file and not a directory, return only that 
47/// file instead
48///
49/// # Arguments:
50///
51///    * input   : name of the target directory
52///    * pattern : the regex pattern to look for. That the sorting works,
53///                the pattern needs to return a date for the first
54///                captured argument and a time for the second captured argument
55pub fn list_path_contents_sorted(input: &str, pattern: Option<Regex>) -> Result<Vec<String>, io::Error> {
56  let path = Path::new(input);
57  match fs::metadata(path) {
58    Ok(metadata) => {
59      if metadata.is_file() {
60        let fname = String::from(input);
61        return Ok(vec![fname]);
62      } 
63      if metadata.is_dir() {
64        let re : Regex;
65        match pattern {
66          None => {
67            // use a default pattern which matches mmost cases  
68            //re = Regex::new(r"Run\d+_\d+\.(\d{6})_(\d{6})UTC(\.tof)?\.gaps$").unwrap();
69            re = Regex::new(GENERIC_ONLINE_FILE_PATTERH).unwrap();
70          }
71          Some(_re) => {
72            re = _re;
73          }
74        }
75        let mut entries: Vec<(u32, u32, String)> = fs::read_dir(path)?
76          .filter_map(Result::ok) // Ignore unreadable entries
77          .filter_map(|entry| {
78            let filename = format!("{}/{}", path.display(), entry.file_name().into_string().ok()?);
79            re.captures(&filename.clone()).map(|caps| {
80              let date = caps.get(1)?.as_str().parse::<u32>().ok()?;
81              let time = caps.get(2)?.as_str().parse::<u32>().ok()?;
82              Some((date, time, filename))
83            })?
84          })
85          .collect();
86
87        // Sort by (date, time)
88        entries.sort_by(|a, b| (a.0, a.1).cmp(&(b.0, b.1)));
89        // Return only filenames
90        return Ok(entries.into_iter().map(|(_, _, name)| name).collect());
91      } 
92      Err(io::Error::new(io::ErrorKind::Other, "Path exists but is neither a file nor a directory"))
93    }
94    Err(e) => Err(e),
95  }
96}

100/// Get all filenames in the current path sorted by timestamp if available
101/// If the given path is a file and not a directory, return only that 
102/// file instead
103///
104/// # Arguments:
105///
106///    * input   : name of the target directory
107///    * pattern : the regex pattern to look for. That the sorting works,
108///                the pattern needs to return a date for the first
109///                captured argument and a time for the second captured argument
110#[cfg(feature="pybindings")]
111#[pyfunction]
112#[pyo3(name="list_path_contents_sorted")]
113#[pyo3(signature = ( input, pattern = None ))]
114pub fn list_path_contents_sorted_py(input: &str, pattern: Option<String>) -> PyResult<Option<Vec<String>>> {
115  let mut regex_pattern : Option<Regex> = None;
116  if let Some(pat_str) = pattern {
117    match Regex::new(&pat_str) {
118      Err(err) => {
119        let msg = format!("Unable to compile regex {}! {}. Check your regex syntax! Also try a raw string.", &pat_str, err); 
120        return Err(PyValueError::new_err(msg));
121      }
122      Ok(re) => {
123        regex_pattern = Some(re);
124      }
125    }
126  }
127  match list_path_contents_sorted(input, regex_pattern) {
128    Err(err) => {  
129      error!("Unable to get files! {err}");
130      return Err(PyValueError::new_err(err.to_string()));
131    }
132    Ok(files) => {
133      return Ok(Some(files));
134    }
135  }
136}

149/// Retrieve the utc timestamp from any telemetry (binary) file 
150#[cfg_attr(feature="pybindings", pyfunction)]
151pub fn get_unix_timestamp_from_telemetry(fname : &str) -> Option<u64> { 
152  let tformat_re = Regex::new(GENERIC_TELEMETRY_FILE_PATTERN_CAPUTRE).unwrap();
153  let res = tformat_re.captures(fname).and_then(|caps| {
154    let map : HashMap<String, String> = tformat_re.capture_names()
155      .filter_map(|name| name)
156      .filter_map(|name| { 
157        //caps.name(name).map(|m| (m.as_str().to_string(), m.as_str().to_string()))
158        caps.name(name).map(|m| (name.to_string(), m.as_str().to_string()))
159      })
160      .collect();
161    Some(map)
162  });
163  return get_unix_timestamp(&res.unwrap()["utctime"], None); 
164}

180/// the liftof suite
181///
182/// # Arguments

185///             Readoutboard (1-50)
186/// * default : if default, just add

200/// A standardized name for regular run files saved by
201/// the liftof suite
202///
203/// # Arguments
204///
205/// * run       : run id (identifier)
206/// * subrun    :  subrun id (identifier of file # within
207///                the run
208/// * rb_id     :  in case this should be used on the rb, 
209///                a rb id can be specified as well
210/// * timestamp :  substitute the current time with this timestamp
211///                (or basically any other string) instead.
212/// * tof_only  :  if true, the filename will end with the suffix 
213///                .tof.gaps, if false it will end simply with .gaps 
214#[cfg_attr(feature="pybindings", pyfunction)]
215pub fn get_runfilename(run : u32, subrun : u64, rb_id : Option<u8>, timestamp : Option<String>, tof_only : bool) -> String {
216  let ts : String;
217  match timestamp {
218    Some(_ts) => {
219      ts = _ts;
220    }
221    None => {
222      ts = get_utc_timestamp();
223    }
224  }
225  let fname : String;
226  match rb_id {
227    None => {
228      if tof_only {
229        fname = format!("Run{run}_{subrun}.{ts}.tof.gaps"); 
230      } else {
231        fname = format!("Run{run}_{subrun}.{ts}.gaps");
232      }
233    }
234    Some(rbid) => {
235      fname = format!("Run{run}_{subrun}.{ts}.RB{rbid:02}.tof.gaps");
236    }
237  }
238  fname
239}

288/// Retrieve the DateTime object from a string as used in 
289/// the names of the run files
290///
291/// # Arguments:
292///
293///   * input  : The input string the datetime shall be extracted
294///              from 
295///   * format : The format of the date string. Something like 
296///              %y%m%d_%H%M%S
297#[cfg_attr(feature="pybindings", pyfunction)]
298#[cfg_attr(feature="pybindings", pyo3(signature = (input , tformat = None )))]
299pub fn get_datetime(input : &str, tformat : Option<String>) -> Option<DateTime<Utc>> {
300  // this is the default format 
301  let mut date_time_format = String::from("%y%m%d_%H%M%S");
302  if let Some(tform) = tformat {
303    date_time_format = tform.to_string(); 
304  }
305  if let Ok(ndtime) = NaiveDateTime::parse_from_str(input, &date_time_format) {
306    //let dt_utc : DateTime<Utc> = DateTime::<Utc>::from_utc(ndtime, Utc); 
307    let dt_utc : DateTime<Utc> = DateTime::<Utc>::from_naive_utc_and_offset(ndtime, Utc); 
308    return Some(dt_utc);
309  } else { 
310    error!("Unable to parse {} for format {}! You can specify formats trhough the tformat keyword", input, date_time_format);
311    return None;
312  }
313}

317/// Retrieve the UNIX timestamp from a string as used in 
318/// the names of the run files
319///
320/// # Arguments:
321///
322///   * input  : The input string the datetime shall be extracted
323///              from 
324///   * format : The format of the date string. Something like 
325///              %y%m%d_%H%M%S
326#[cfg_attr(feature="pybindings", pyfunction)]
327#[cfg_attr(feature="pybindings", pyo3(signature = (input , tformat = None )))]
328pub fn get_unix_timestamp(input : &str, tformat : Option<String>) -> Option<u64> {
329  let dt = get_datetime(input, tformat);
330  if let Some(dt_) = dt {
331    // FIXME - we are only supporting times later than 
332    //         the UNIX epoch!
333    return Some(dt_.timestamp() as u64);
334  } else {
335    return None;
336  }

391/// Implement the Reader trait and necessary getters/setters to 
392/// make a struct an actual reader
393#[macro_export]
394macro_rules! reader {
395  ($struct_name:ident, $element_type:ident) => {
396 
397    use crate::io::DataReader; 
398    use crate::io::Serialization;
399
400    impl Iterator for $struct_name {
401      type Item = $element_type;
402      fn next(&mut self) -> Option<Self::Item> {
403        self.read_next()
404      }
405    }
406
407    impl DataReader<$element_type> for $struct_name {
408      fn get_header0(&self) -> u8 {
409        ($element_type::HEAD & 0x1) as u8 
410      }
411
412      fn get_header1(&self) -> u8 {
413        ($element_type::HEAD & 0x2) as u8
414      }
415
416      fn get_file_idx(&self) -> usize {
417        self.file_idx // Setting the specified field
418      }
419    
420      fn set_file_idx(&mut self, file_idx : usize) {
421        self.file_idx = file_idx;
422      }
423      
424      fn get_filenames(&self) -> &Vec<String> {
425          &self.filenames
426      }
427      
428      fn set_cursor(&mut self, pos : usize) {
429        self.cursor = pos;
430      }
431 
432      fn set_file_reader(&mut self, reader : BufReader<File>) {
433        self.file_reader = reader;
434      }
435    
436      fn read_next(&mut self) -> Option<$element_type> {
437        self.read_next_item()
438      }
439    
440      /// Get the next file ready
441      fn prime_next_file(&mut self) -> Option<usize> {
442        if self.file_idx == self.filenames.len() -1 {
443          return None;
444        } else {
445          self.file_idx += 1;
446          let nextfilename : &str = self.filenames[self.file_idx].as_str();
447          let nextfile     = OpenOptions::new().create(false).append(false).read(true).open(nextfilename).expect("Unable to open file {nextfilename}");
448          self.file_reader = BufReader::new(nextfile);
449          self.cursor      = 0;
450          return Some(self.file_idx);
451        }
452      }
453    }
454  }
455}

457/// Generics for packet reading (TofPacket, Telemetry packet,...)
458/// FIXME - not implemented yet
459pub trait DataReader<T> 
460  where T : Default + Serialization {
461  ///// header bytes, e.g. 0xAAAA for TofPackets, first byte
462  //const HEADER0 : u8 = 0;
463  ///// header bytes, e.g. 0xAAAA for TofPackets, second byte
464  //const HEADER1 : u8 = 0;
465
466  fn get_header0(&self) -> u8;
467  fn get_header1(&self) -> u8;
468
469  /// Return all filenames the reader is primed with   
470  fn get_filenames(&self) -> &Vec<String>;
471
472  /// The current index corresponding to the file the 
473  /// reader is currently working on
474  fn get_file_idx(&self) -> usize;
475
476  /// Set a new file idx corresponding to a file the reader 
477  /// is currently working on
478  fn set_file_idx(&mut self, idx : usize);
479
480  /// reset a new reader
481  fn set_file_reader(&mut self, freader : BufReader<File>);
482  
483  /// Get the next file ready
484  fn prime_next_file(&mut self) -> Option<usize>;
485
486  /// The name of the file the reader is currently 
487  /// working on
488  fn get_current_filename(&self) -> Option<&str> {
489    // should only happen when it is empty
490    if self.get_filenames().len() <= self.get_file_idx() {
491      return None;
492    }
493    Some(self.get_filenames()[self.get_file_idx()].as_str())
494  }
495
496  /// Manage the internal cursor attribute
497  fn set_cursor(&mut self, pos : usize);
498
499  /// Get the next frame/packet from the stream. Can be used to 
500  /// implement iterators
501  fn read_next(&mut self) -> Option<T>; 
502
503  /// Get the first entry in all of the files the reader is 
504  /// primed with
505  fn first(&mut self)     -> Option<T> {
506      match self.rewind() {
507      Err(err) => {
508        error!("Error when rewinding files! {err}");
509        return None;
510      }
511      Ok(_) => ()
512    }
513    let pack = self.read_next();
514    match self.rewind() {
515      Err(err) => {
516        error!("Error when rewinding files! {err}");
517      }
518      Ok(_) => ()
519    }
520    return pack;
521  }
522
523  /// Get the last entry in all of the files the reader is 
524  /// primed with
525  fn last(&mut self)      -> Option<T> {
526    self.set_file_idx(self.get_filenames().len() - 1);
527    let lastfilename = self.get_filenames()[self.get_file_idx()].as_str();
528    let lastfile     = OpenOptions::new().create(false).append(false).read(true).open(lastfilename).expect("Unable to open file {nextfilename}");
529    self.set_file_reader(BufReader::new(lastfile));
530    self.set_cursor(0);
531    let mut tp    = T::default();
532    let mut idx = 0;
533    loop {
534      match self.read_next() {
535        None => {
536          match self.rewind() {
537            Err(err) => {
538              error!("Error when rewinding files! {err}");
539            }
540            Ok(_) => ()
541          }
542          if idx == 0 {
543            return None;
544          } else {
545            return Some(tp);
546          }
547        }
548        Some(pack) => {
549          idx += 1;
550          tp = pack;
551          continue;
552        }
553      }
554    }
555  }
556
557  /// Rewind the current file and set the file index to the 
558  /// first file, so data can be read again from the 
559  /// beginning
560  fn rewind(&mut self) -> io::Result<()> {
561    let firstfile = &self.get_filenames()[0];
562    let file = OpenOptions::new().create(false).append(false).read(true).open(&firstfile)?;
563    self.set_file_reader(BufReader::new(file));
564    self.set_cursor(0);
565    self.set_file_idx(0);
566    Ok(())
567  }
568}

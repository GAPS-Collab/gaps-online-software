7/// Implements the [Multivariate Student's t-distribution](https://en.wikipedia.org/wiki/Multivariate_t-distribution)
8/// distribution using the "nalgebra" crate for matrix operations.
9///
10/// Assumes all the marginal distributions have the same degree of freedom, ν.
11///
12/// # Examples
13///
14/// ```
15/// use statrs::distribution::{MultivariateStudent, Continuous};
16/// use nalgebra::{DVector, DMatrix};
17/// use statrs::statistics::{MeanN, VarianceN};
18///
19/// let mvs = MultivariateStudent::new(vec![0., 0.], vec![1., 0., 0., 1.], 4.).unwrap();
20/// assert_eq!(mvs.mean().unwrap(), DVector::from_vec(vec![0., 0.]));
21/// assert_eq!(mvs.variance().unwrap(), DMatrix::from_vec(2, 2, vec![2., 0., 0., 2.]));
22/// assert_eq!(mvs.pdf(&DVector::from_vec(vec![1.,  1.])), 0.04715702017537655);
23/// ```
24#[derive(Debug, Clone, PartialEq)]
25pub struct MultivariateStudent<D>
26where
27    D: Dim,
28    nalgebra::DefaultAllocator:
29        nalgebra::allocator::Allocator<D> + nalgebra::allocator::Allocator<D, D>,
30{
31    scale_chol_decomp: OMatrix<f64, D, D>,
32    location: OVector<f64, D>,
33    scale: OMatrix<f64, D, D>,
34    freedom: f64,
35    precision: OMatrix<f64, D, D>,
36    ln_pdf_const: f64,
37}

39/// Represents the errors that can occur when creating a [`MultivariateStudent`].
40#[derive(Copy, Clone, PartialEq, Eq, Debug, Hash)]
41#[non_exhaustive]
42pub enum MultivariateStudentError {
43    /// The scale matrix is asymmetric or contains a NaN.
44    ScaleInvalid,
45
46    /// The location vector contains a NaN.
47    LocationInvalid,
48
49    /// The degrees of freedom are NaN, zero or less than zero.
50    FreedomInvalid,
51
52    /// The amount of rows in the location vector is not equal to the amount
53    /// of rows in the scale matrix.
54    DimensionMismatch,
55
56    /// After all other validation, computing the Cholesky decomposition failed.
57    /// This means that the scale matrix is not definite-positive.
58    CholeskyFailed,
59}
60
61impl std::fmt::Display for MultivariateStudentError {
62    #[cfg_attr(coverage_nightly, coverage(off))]
63    fn fmt(&self, f: &mut std::fmt::Formatter) -> std::fmt::Result {
64        match self {
65            MultivariateStudentError::ScaleInvalid => {
66                write!(f, "Scale matrix is asymmetric or contains a NaN")
67            }
68            MultivariateStudentError::LocationInvalid => {
69                write!(f, "Location vector contains a NaN")
70            }
71            MultivariateStudentError::FreedomInvalid => {
72                write!(f, "Degrees of freedom are NaN, zero or less than zero")
73            }
74            MultivariateStudentError::DimensionMismatch => write!(
75                f,
76                "Location vector and scale matrix do not have the same number of rows"
77            ),
78            MultivariateStudentError::CholeskyFailed => {
79                write!(f, "Computing the Cholesky decomposition failed")
80            }
81        }
82    }
83}

87impl MultivariateStudent<Dyn> {
88    /// Constructs a new multivariate students t distribution with a location of `location`,
89    /// scale matrix `scale` and `freedom` degrees of freedom.
90    ///
91    /// # Errors
92    ///
93    /// Returns `StatsError::BadParams` if the scale matrix is not symmetric-positive
94    /// definite and `StatsError::ArgMustBePositive` if freedom is non-positive.
95    pub fn new(
96        location: Vec<f64>,
97        scale: Vec<f64>,
98        freedom: f64,
99    ) -> Result<Self, MultivariateStudentError> {
100        let dim = location.len();
101        Self::new_from_nalgebra(location.into(), DMatrix::from_vec(dim, dim, scale), freedom)
102    }
103
104    /// Returns the dimension of the distribution.
105    pub fn dim(&self) -> usize {
106        self.location.len()
107    }
108}

110impl<D> MultivariateStudent<D>
111where
112    D: DimMin<D, Output = D>,
113    nalgebra::DefaultAllocator: nalgebra::allocator::Allocator<D>
114        + nalgebra::allocator::Allocator<D, D>
115        + nalgebra::allocator::Allocator<D>,
116{
117    pub fn new_from_nalgebra(
118        location: OVector<f64, D>,
119        scale: OMatrix<f64, D, D>,
120        freedom: f64,
121    ) -> Result<Self, MultivariateStudentError> {
122        let dim = location.len();
123
124        if location.iter().any(|f| f.is_nan()) {
125            return Err(MultivariateStudentError::LocationInvalid);
126        }
127
128        if !scale.is_square()
129            || scale.lower_triangle() != scale.upper_triangle().transpose()
130            || scale.iter().any(|f| f.is_nan())
131        {
132            return Err(MultivariateStudentError::ScaleInvalid);
133        }
134
135        if freedom.is_nan() || freedom <= 0.0 {
136            return Err(MultivariateStudentError::FreedomInvalid);
137        }
138
139        if location.nrows() != scale.nrows() {
140            return Err(MultivariateStudentError::DimensionMismatch);
141        }
142
143        let scale_det = scale.determinant();
144        let ln_pdf_const = gamma::ln_gamma(0.5 * (freedom + dim as f64))
145            - gamma::ln_gamma(0.5 * freedom)
146            - 0.5 * (dim as f64) * (freedom * PI).ln()
147            - 0.5 * scale_det.ln();
148
149        match Cholesky::new(scale.clone()) {
150            None => Err(MultivariateStudentError::CholeskyFailed),
151            Some(cholesky_decomp) => {
152                let precision = cholesky_decomp.inverse();
153                Ok(MultivariateStudent {
154                    scale_chol_decomp: cholesky_decomp.unpack(),
155                    location,
156                    scale,
157                    freedom,
158                    precision,
159                    ln_pdf_const,
160                })
161            }
162        }
163    }
164
165    /// Returns the cholesky decomposiiton matrix of the scale matrix.
166    ///
167    /// Returns A where Σ = AAᵀ.
168    pub fn scale_chol_decomp(&self) -> &OMatrix<f64, D, D> {
169        &self.scale_chol_decomp
170    }
171
172    /// Returns the location of the distribution.
173    pub fn location(&self) -> &OVector<f64, D> {
174        &self.location
175    }
176
177    /// Returns the scale matrix of the distribution.
178    pub fn scale(&self) -> &OMatrix<f64, D, D> {
179        &self.scale
180    }
181
182    /// Returns the degrees of freedom of the distribution.
183    pub fn freedom(&self) -> f64 {
184        self.freedom
185    }
186
187    /// Returns the inverse of the cholesky decomposition matrix.
188    pub fn precision(&self) -> &OMatrix<f64, D, D> {
189        &self.precision
190    }
191
192    /// Returns the logarithmed constant part of the probability
193    /// distribution function.
194    pub fn ln_pdf_const(&self) -> f64 {
195        self.ln_pdf_const
196    }
197}

201impl<D> ::rand::distributions::Distribution<OVector<f64, D>> for MultivariateStudent<D>
202where
203    D: Dim,
204    nalgebra::DefaultAllocator:
205        nalgebra::allocator::Allocator<D> + nalgebra::allocator::Allocator<D, D>,
206{
207    /// Samples from the multivariate student distribution
208    ///
209    /// # Formula
210    ///
211    /// ```math
212    /// W ⋅ L ⋅ Z + μ
213    /// ```
214    ///
215    /// where `W` has √(ν/Sν) distribution, Sν has Chi-squared
216    /// distribution with ν degrees of freedom,
217    /// `L` is the Cholesky decomposition of the scale matrix,
218    /// `Z` is a vector of normally distributed random variables, and
219    /// `μ` is the location vector
220    fn sample<R: ::rand::Rng + ?Sized>(&self, rng: &mut R) -> OVector<f64, D> {
221        use crate::distribution::{ChiSquared, Normal};
222
223        let d = Normal::new(0., 1.).unwrap();
224        let s = ChiSquared::new(self.freedom).unwrap();
225        let w = (self.freedom / s.sample(rng)).sqrt();
226        let (r, c) = self.location.shape_generic();
227        let z = OVector::<f64, D>::from_distribution_generic(r, c, &d, rng);
228        (w * &self.scale_chol_decomp * z) + &self.location
229    }
230}

283impl<D> VarianceN<OMatrix<f64, D, D>> for MultivariateStudent<D>
284where
285    D: Dim,
286    nalgebra::DefaultAllocator:
287        nalgebra::allocator::Allocator<D> + nalgebra::allocator::Allocator<D, D>,
288{
289    /// Returns the covariance matrix of the multivariate student distribution.
290    ///
291    /// # Formula
292    ///
293    /// ```math
294    /// Σ ⋅ ν / (ν - 2)
295    /// ```
296    ///
297    /// where `Σ` is the scale matrix and `ν` is the degrees of freedom.
298    /// Only defined if freedom is larger than 2.
299    fn variance(&self) -> Option<OMatrix<f64, D, D>> {
300        if self.freedom > 2. {
301            Some(self.scale.clone() * self.freedom / (self.freedom - 2.))
302        } else {
303            None
304        }
305    }
306}

328impl<D> Continuous<&OVector<f64, D>, f64> for MultivariateStudent<D>
329where
330    D: Dim + DimMin<D, Output = D>,
331    nalgebra::DefaultAllocator: nalgebra::allocator::Allocator<D>
332        + nalgebra::allocator::Allocator<D, D>
333        + nalgebra::allocator::Allocator<D>,
334{
335    /// Calculates the probability density function for the multivariate.
336    /// student distribution at `x`.
337    ///
338    /// # Formula
339    ///
340    /// ```math
341    /// [Γ(ν+p)/2] / [Γ(ν/2) ((ν * π)^p det(Σ))^(1 / 2)] * [1 + 1/ν (x - μ)ᵀ inv(Σ) (x - μ)]^(-(ν+p)/2)
342    /// ```
343    ///
344    /// where
345    /// - `ν` is the degrees of freedom,
346    /// - `μ` is the mean,
347    /// - `Γ` is the Gamma function,
348    /// - `inv(Σ)` is the precision matrix,
349    /// - `det(Σ)` is the determinant of the scale matrix, and
350    /// - `k` is the dimension of the distribution.
351    fn pdf(&self, x: &OVector<f64, D>) -> f64 {
352        if self.freedom.is_infinite() {
353            use super::multivariate_normal::density_normalization_and_exponential;
354            let (pdf_const, exp_arg) = density_normalization_and_exponential(
355                &self.location,
356                &self.scale,
357                &self.precision,
358                x,
359            )
360            .unwrap();
361            return pdf_const * exp_arg.exp();
362        }
363
364        let dv = x - &self.location;
365        let exp_arg: f64 = (&self.precision * &dv).dot(&dv);
366        let base_term = 1. + exp_arg / self.freedom;
367        self.ln_pdf_const.exp() * base_term.powf(-(self.freedom + self.location.len() as f64) / 2.)
368    }
369
370    /// Calculates the log probability density function for the multivariate
371    /// student distribution at `x`. Equivalent to pdf(x).ln().
372    fn ln_pdf(&self, x: &OVector<f64, D>) -> f64 {
373        if self.freedom.is_infinite() {
374            use super::multivariate_normal::density_normalization_and_exponential;
375            let (pdf_const, exp_arg) = density_normalization_and_exponential(
376                &self.location,
377                &self.scale,
378                &self.precision,
379                x,
380            )
381            .unwrap();
382            return pdf_const.ln() + exp_arg;
383        }
384
385        let dv = x - &self.location;
386        let exp_arg: f64 = (&self.precision * &dv).dot(&dv);
387        let base_term = 1. + exp_arg / self.freedom;
388        self.ln_pdf_const - (self.freedom + self.location.len() as f64) / 2. * base_term.ln()
389    }
390}

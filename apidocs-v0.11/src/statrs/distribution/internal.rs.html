3/// Implements univariate function bisection searching for criteria
4/// ```text
5/// smallest k such that f(k) >= z
6/// ```
7/// Evaluates to `None` if
8/// - provided interval has lower bound greater than upper bound
9/// - function found not semi-monotone on the provided interval containing `z`
10///
11/// Evaluates to `Some(k)`, where `k` satisfies the search criteria
12pub fn integral_bisection_search<K: Num + Clone, T: Num + PartialOrd>(
13    f: impl Fn(&K) -> T,
14    z: T,
15    lb: K,
16    ub: K,
17) -> Option<K> {
18    if !(f(&lb)..=f(&ub)).contains(&z) {
19        return None;
20    }
21    let two = K::one() + K::one();
22    let mut lb = lb;
23    let mut ub = ub;
24    loop {
25        let mid = (lb.clone() + ub.clone()) / two.clone();
26        if !(f(&lb)..=f(&ub)).contains(&f(&mid)) {
27            return None; // f found not monotone on interval
28        } else if f(&lb) == z {
29            return Some(lb);
30        } else if f(&ub) == z || (lb.clone() + K::one()) == ub {
31            return Some(ub); // found or no more integers between
32        } else if f(&mid) >= z {
33            ub = mid;
34        } else {
35            lb = mid;
36        }
37    }
38}

349    /// cdf should be the integral of the pdf
350    fn check_integrate_pdf_is_cdf<D: ContinuousCDF<f64, f64> + Continuous<f64, f64>>(
351        dist: &D,
352        x_min: f64,
353        x_max: f64,
354        step: f64,
355    ) {
356        let mut prev_x = x_min;
357        let mut prev_density = dist.pdf(x_min);
358        let mut sum = 0.0;
359
360        loop {
361            let x = prev_x + step;
362            let density = dist.pdf(x);
363
364            assert!(density >= 0.0);
365
366            let ln_density = dist.ln_pdf(x);
367
368            assert_almost_eq!(density.ln(), ln_density, 1e-10);
369
370            // triangle rule
371            sum += (prev_density + density) * step / 2.0;
372
373            let cdf = dist.cdf(x);
374            if (sum - cdf).abs() > 1e-3 {
375                println!("Integral of pdf doesn't equal cdf!");
376                println!("Integration from {x_min} by {step} to {x} = {sum}");
377                println!("cdf = {cdf}");
378                panic!();
379            }
380
381            if x >= x_max {
382                break;
383            } else {
384                prev_x = x;
385                prev_density = density;
386            }
387        }
388
389        assert!(sum > 0.99);
390        assert!(sum <= 1.001);
391    }

393    /// cdf should be the sum of the pmf
394    fn check_sum_pmf_is_cdf<D: DiscreteCDF<u64, f64> + Discrete<u64, f64>>(dist: &D, x_max: u64) {
395        let mut sum = 0.0;
396
397        // go slightly beyond x_max to test for off-by-one errors
398        for i in 0..x_max + 3 {
399            let prob = dist.pmf(i);
400
401            assert!(prob >= 0.0);
402            assert!(prob <= 1.0);
403
404            sum += prob;
405
406            if i == x_max {
407                assert!(sum > 0.99);
408            }
409
410            assert_almost_eq!(sum, dist.cdf(i), 1e-10);
411            // assert_almost_eq!(sum, dist.cdf(i as f64), 1e-10);
412            // assert_almost_eq!(sum, dist.cdf(i as f64 + 0.1), 1e-10);
413            // assert_almost_eq!(sum, dist.cdf(i as f64 + 0.5), 1e-10);
414            // assert_almost_eq!(sum, dist.cdf(i as f64 + 0.9), 1e-10);
415        }
416
417        assert!(sum > 0.99);
418        assert!(sum <= 1.0 + 1e-10);
419    }

421    /// Does a series of checks that all continuous distributions must obey.
422    /// 99% of the probability mass should be between x_min and x_max.
423    pub fn check_continuous_distribution<D: ContinuousCDF<f64, f64> + Continuous<f64, f64>>(
424        dist: &D,
425        x_min: f64,
426        x_max: f64,
427    ) {
428        assert_eq!(dist.pdf(f64::NEG_INFINITY), 0.0);
429        assert_eq!(dist.pdf(f64::INFINITY), 0.0);
430        assert_eq!(dist.ln_pdf(f64::NEG_INFINITY), f64::NEG_INFINITY);
431        assert_eq!(dist.ln_pdf(f64::INFINITY), f64::NEG_INFINITY);
432        assert_eq!(dist.cdf(f64::NEG_INFINITY), 0.0);
433        assert_eq!(dist.cdf(f64::INFINITY), 1.0);
434
435        check_integrate_pdf_is_cdf(dist, x_min, x_max, (x_max - x_min) / 100000.0);
436    }

454    #[test]
455    fn test_integer_bisection() {
456        fn search(z: usize, data: &[usize]) -> Option<usize> {
457            integral_bisection_search(|idx: &usize| data[*idx], z, 0, data.len() - 1)
458        }
459
460        let needle = 3;
461        let data = (0..5)
462            .map(|n| if n >= needle { n + 1 } else { n })
463            .collect::<Vec<_>>();
464
465        for i in 0..(data.len()) {
466            assert_eq!(search(data[i], &data), Some(i),)
467        }
468        {
469            let infimum = search(needle, &data);
470            let found_element = search(needle + 1, &data); // 4 > needle && member of range
471            assert_eq!(found_element, Some(needle));
472            assert_eq!(infimum, found_element)
473        }
474    }

21    pub fn write_dac(&self, channel:u8, value: u16) -> Result<(), LinuxI2CError> {
22        let mut dev = LinuxI2CDevice::new(&format!("/dev/i2c-{}", self.bus), self.address)?;
23        
24        let dac_value_msb = (value & 0xFF00) >> 8;
25        let dac_value_lsb = (value & 0x00FF);
26        let dac_value = [dac_value_msb as u8, dac_value_lsb as u8];
27
28        dev.smbus_write_i2c_block_data(WRITE_AND_UPDATE_DAC+channel, &dac_value);
29        
30        Ok(())
31    }
32
33    pub fn read_dac(&self, channel: u8) -> Result<u16, LinuxI2CError> {
34        let mut dev = LinuxI2CDevice::new(&format!("/dev/i2c-{}", self.bus), self.address)?;
35        let mut read_data = [0; 2];
36        let mut msgs = [
37            LinuxI2CMessage::write(&[READBACK_ENABLE+channel, 0x00, 0x00]),
38            LinuxI2CMessage::read(&mut read_data)
39        ];
40        
41        dev.transfer(&mut msgs)?;
42
43        let dac_value = ((read_data[0] as u16 & 0xFF) << 8) | (read_data[1] as u16 & 0xFF);
44
45        Ok((dac_value))
46    }
47
48    pub fn read_dac_voltage(&self, channel: u8) -> Result<f32, LinuxI2CError> {
49        let dac = Self::read_dac(&self, channel)?;
50        let voltage = ((dac as f32) / 2f32.powf(16.0)) * RB_AD5675_REF_VOLTAGE;
51
52        Ok((voltage))
53    }
